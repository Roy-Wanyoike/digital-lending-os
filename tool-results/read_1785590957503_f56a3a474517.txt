    73→}

    98→
    99→export function WalletTab() {
   100→  const { data: businesses, loading: bLoading, error: bizError, refetch } = useApi<Business[]>('/api/businesses?limit=50')
   101→  const { data: rates, error: ratesError } = useApi<RatesData>('/api/wallets/rates')
   102→  const [selectedBizId, setSelectedBizId] = useState<string>('')
   103→  const safeBusinesses = Array.isArray(businesses) ? businesses : []
   104→  const bizId = selectedBizId || safeBusinesses?.[0]?.id || ''
   105→  const [walletKey, setWalletKey] = useState(0)
   106→  const { data: wallets, loading: wLoading, error: walletsError } = useApi<WalletData[]>(bizId ? `/api/wallets?businessId=${bizId}&k=${walletKey}` : '')
   107→  const [selectedWalletId, setSelectedWalletId] = useState<string | null>(null)
   108→  const [txns, setTxns] = useState<WalletTransaction[]>([])
   109→  const [txLoading, setTxLoading] = useState(false)
   110→
   111→  // Dialogs
   112→  const [createOpen, setCreateOpen] = useState(false)
   113→  const [depositOpen, setDepositOpen] = useState(false)
   114→  const [withdrawOpen, setWithdrawOpen] = useState(false)
   115→  const [convertOpen, setConvertOpen] = useState(false)
   116→  const [cryptoOpen, setCryptoOpen] = useState(false)
   117→
   118→  // History dialogs
   119→  const [historyTab, setHistoryTab] = useState('transactions')
   120→  const [deposits, setDeposits] = useState<DepositRecord[]>([])
   121→  const [withdrawals, setWithdrawals] = useState<WithdrawalRecord[]>([])
   122→  const [cryptoWdrs, setCryptoWdrs] = useState<CryptoWithdrawalRecord[]>([])
   123→  const [historyOpen, setHistoryOpen] = useState(false)
   124→
   125→
   126→
   127→  // Create wallet form
   128→  const [newCurrency, setNewCurrency] = useState('USD')
   129→  const [creating, setCreating] = useState(false)
   130→
   131→  // Deposit form
   132→  const [depAmount, setDepAmount] = useState('')
   133→  const [depMethod, setDepMethod] = useState('bank_transfer')
   134→  const [depNotes, setDepNotes] = useState('')
   135→  const [depSubmitting, setDepSubmitting] = useState(false)
   136→
   137→  // Withdraw form
   138→  const [wdrAmount, setWdrAmount] = useState('')
   139→  const [wdrMethod, setWdrMethod] = useState('bank_transfer')
   140→  const [wdrBankName, setWdrBankName] = useState('')
   141→  const [wdrBankAccount, setWdrBankAccount] = useState('')
   142→  const [wdrRecipientName, setWdrRecipientName] = useState('')
   143→  const [wdrNotes, setWdrNotes] = useState('')
   144→  const [wdrSubmitting, setWdrSubmitting] = useState(false)
   145→
   146→  // Convert form
   147→  const [cvtFromWalletId, setCvtFromWalletId] = useState('')
   148→  const [cvtToWalletId, setCvtToWalletId] = useState('')
   149→  const [cvtAmount, setCvtAmount] = useState('')
   150→  const [cvtSubmitting, setCvtSubmitting] = useState(false)
   151→
   152→  // Crypto form
   153→  const [crCrypto, setCrCrypto] = useState('USDT')
   154→  const [crNetwork, setCrNetwork] = useState('trc20')
   155→  const [crAddress, setCrAddress] = useState('')
   156→  const [crAmount, setCrAmount] = useState('')
   157→  const [crNotes, setCrNotes] = useState('')
   158→  const [crSubmitting, setCrSubmitting] = useState(false)
   159→
   160→  const showToast = useCallback((msg: string) => { toast(msg) }, [])
   161→
   162→  const loadTransactions = useCallback(async (walletId: string) => {
   163→    setSelectedWalletId(walletId)
   164→    setTxLoading(true)
   165→    try {
   166→      const res = await fetch(`/api/wallets/${walletId}/transactions?limit=50`)
   167→      const json = await res.json()
   168→      setTxns(json.data || [])
   169→    } catch { setTxns([]) } finally { setTxLoading(false) }
   170→  }, [])
   171→
   172→  const loadHistory = useCallback(async (walletId: string, tab: string) => {
   173→    setHistoryTab(tab)
   174→    setHistoryOpen(true)
   175→    try {
   176→      if (tab === 'deposits') {
   177→        const r = await fetch(`/api/wallets/deposit?walletId=${walletId}&limit=50`)
   178→        setDeposits((await r.json()).data || [])
   179→      } else if (tab === 'withdrawals') {
   180→        const r = await fetch(`/api/wallets/withdrawal?walletId=${walletId}&limit=50`)
   181→        setWithdrawals((await r.json()).data || [])
   182→      } else if (tab === 'crypto') {
   183→        const r = await fetch(`/api/wallets/crypto-withdrawal?walletId=${walletId}&limit=50`)
   184→        setCryptoWdrs((await r.json()).data || [])
   185→      }
   186→    } catch { /* ignore */ }
   187→  }, [])
   188→
   189→  const allWallets = Array.isArray(wallets) ? wallets : []
   190→  const totalPortfolio = allWallets.reduce((sum, w) => {
   191→    try {
   192→      const toUsd = rates?.fiatToUsd?.[w.currency] || 1
   193→      return sum + w.balance * toUsd
   194→    } catch { return sum }
   195→  }, 0)
   196→  const totalBalance = allWallets.reduce((s, w) => s + w.balance, 0)
   197→  const totalPending = allWallets.reduce((s, w) => s + w.pendingBalance, 0)
   198→  const totalFrozen = allWallets.reduce((s, w) => s + w.frozenBalance, 0)
   199→
   200→  const selectedWallet = allWallets.find(w => w.id === selectedWalletId)
   201→
   202→  // Conversion preview
   203→  const conversionPreview = (() => {
   204→    if (!cvtAmount || !cvtFromWalletId || !cvtToWalletId || !rates) return null
   205→    const from = allWallets.find(w => w.id === cvtFromWalletId)
   206→    const to = allWallets.find(w => w.id === cvtToWalletId)
   207→    if (!from || !to) return null
   208→    const amount = parseFloat(cvtAmount)
   209→    if (isNaN(amount) || amount <= 0) return null
   210→    const direct = rates.fiatRates[from.currency]?.[to.currency]
   211→    if (!direct) return null
   212→    const gross = amount * direct
   213→    const fee = Math.round(gross * 0.005 * 100) / 100
   214→    const net = Math.round((gross - fee) * 100) / 100
   215→    return { rate: direct, gross, fee, net }
   216→  })()
   217→
   218→  // Crypto withdrawal preview
   219→  const cryptoPreview = (() => {
   220→    if (!crAmount || !selectedWallet || !rates) return null
   221→    const amount = parseFloat(crAmount)
   222→    if (isNaN(amount) || amount <= 0) return null
   223→    const toUsd = rates.fiatToUsd[selectedWallet.currency] || 1
   224→    const usdAmount = amount * toUsd
   225→    const cryptoPrice = rates.cryptoPrices[crCrypto]
   226→    if (!cryptoPrice) return null
   227→    const cryptoAmt = usdAmount / cryptoPrice
   228→    const netFee = rates.networkFees[crNetwork] || 0
   229→    const netCrypto = Math.max(0, cryptoAmt - netFee)
   230→    const procFee = Math.max(amount * (rates.cryptoWithdrawalFeePercent / 100), rates.cryptoWithdrawalMinFee)
   231→    const totalDebit = amount + procFee
   232→    return { usdAmount, cryptoAmt, netCrypto, netFee, procFee, totalDebit, cryptoPrice }
   233→  })()
   234→
   235→  const handleCreateWallet = async () => {
   236→    if (!bizId) return
   237→    setCreating(true)
   238→    try {
   239→      const res = await fetch('/api/wallets', {
   240→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   241→        body: JSON.stringify({ businessId: bizId, currency: newCurrency }),
   242→      })
   243→      const json = await res.json()
   244→      if (!res.ok) throw new Error(json.error || 'Failed')
   245→      showToast('Wallet created!')
   246→      setCreateOpen(false)
   247→      setWalletKey(k => k + 1)
   248→    } catch (e: any) { showToast(e.message) } finally { setCreating(false) }
   249→  }
   250→
   251→  const handleDeposit = async () => {
   252→    if (!selectedWalletId || !depAmount) return
   253→    setDepSubmitting(true)
   254→    try {
   255→      const res = await fetch('/api/wallets/deposit', {
   256→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   257→        body: JSON.stringify({
   258→          walletId: selectedWalletId, amount: parseFloat(depAmount),
   259→          paymentMethod: depMethod, provider: 'demo', notes: depNotes || undefined,
   260→        }),
   261→      })
   262→      const json = await res.json()
   263→      if (!res.ok) throw new Error(json.error || 'Failed')
   264→      showToast('Deposit completed!')
   265→      setDepositOpen(false); setDepAmount(''); setDepNotes('')
   266→      setWalletKey(k => k + 1)
   267→      loadTransactions(selectedWalletId)
   268→    } catch (e: any) { showToast(e.message) } finally { setDepSubmitting(false) }
   269→  }
   270→
   271→  const handleWithdraw = async () => {
   272→    if (!selectedWalletId || !wdrAmount) return
   273→    setWdrSubmitting(true)
   274→    try {
   275→      const res = await fetch('/api/wallets/withdrawal', {
   276→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   277→        body: JSON.stringify({
   278→          walletId: selectedWalletId, amount: parseFloat(wdrAmount),
   279→          paymentMethod: wdrMethod, provider: 'demo',
   280→          bankName: wdrBankName || undefined, bankAccount: wdrBankAccount || undefined,
   281→          recipientName: wdrRecipientName || undefined, notes: wdrNotes || undefined,
   282→        }),
   283→      })
   284→      const json = await res.json()
   285→      if (!res.ok) throw new Error(json.error || 'Failed')
   286→      showToast('Withdrawal completed!')
   287→      setWithdrawOpen(false); setWdrAmount(''); setWdrBankName(''); setWdrBankAccount(''); setWdrRecipientName(''); setWdrNotes('')
   288→      setWalletKey(k => k + 1)
   289→      loadTransactions(selectedWalletId)
   290→    } catch (e: any) { showToast(e.message) } finally { setWdrSubmitting(false) }
   291→  }
   292→
   293→  const handleConvert = async () => {
   294→    if (!cvtFromWalletId || !cvtToWalletId || !cvtAmount) return
   295→    setCvtSubmitting(true)
   296→    try {
   297→      const res = await fetch('/api/wallets/convert', {
   298→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   299→        body: JSON.stringify({ fromWalletId: cvtFromWalletId, toWalletId: cvtToWalletId, fromAmount: parseFloat(cvtAmount) }),
   300→      })
   301→      const json = await res.json()
   302→      if (!res.ok) throw new Error(json.error || 'Failed')
   303→      showToast('Conversion completed!')
   304→      setConvertOpen(false); setCvtAmount('')
   305→      setWalletKey(k => k + 1)
   306→      if (selectedWalletId) loadTransactions(selectedWalletId)
   307→    } catch (e: any) { showToast(e.message) } finally { setCvtSubmitting(false) }
   308→  }
   309→
   310→  const handleCryptoWithdraw = async () => {
   311→    if (!selectedWalletId || !crAmount || !crAddress) return
   312→    setCrSubmitting(true)
   313→    try {
   314→      const res = await fetch('/api/wallets/crypto-withdrawal', {
   315→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   316→        body: JSON.stringify({
   317→          walletId: selectedWalletId, amount: parseFloat(crAmount),
   318→          cryptoCurrency: crCrypto, network: crNetwork, walletAddress: crAddress.trim(),
   319→          notes: crNotes || undefined,
   320→        }),
   321→      })
   322→      const json = await res.json()
   323→      if (!res.ok) throw new Error(json.error || 'Failed')
   324→      showToast('Crypto withdrawal submitted!')
   325→      setCryptoOpen(false); setCrAmount(''); setCrAddress(''); setCrNotes('')
   326→      setWalletKey(k => k + 1)
   327→      loadTransactions(selectedWalletId)
   328→    } catch (e: any) { showToast(e.message) } finally { setCrSubmitting(false) }
   329→  }
   330→
   331→  // When crypto changes, reset network to first valid
   332→  useEffect(() => {
   333→    if (rates) {
   334→      const nets = rates.cryptoNetworks[crCrypto]
   335→      if (nets && nets.length > 0) setCrNetwork(nets[0])
   336→    }
   337→  }, [crCrypto, rates])
   338→
   339→  // When convert source wallet changes, clear target if same
   340→  useEffect(() => {
   341→    if (cvtFromWalletId === cvtToWalletId) setCvtToWalletId('')
   342→  }, [cvtFromWalletId])
   343→
   344→  if (bLoading || (bizId && wLoading)) return <LoadingSkeleton />
   345→  if (bizError || ratesError || walletsError) return <ErrorState message={bizError || ratesError || walletsError || ''} onRetry={refetch} />
   346→
   347→  return (
   348→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="space-y-6">
   349→      {/* Business selector */}
   350→      <div className="flex items-center justify-between flex-wrap gap-3">
   351→        <div className="flex items-center gap-3">
   352→          <span className="text-sm text-muted-foreground">Business:</span>
   353→          <Select value={selectedBizId || businesses?.[0]?.id} onValueChange={setSelectedBizId}>
   354→            <SelectTrigger className="w-64"><SelectValue placeholder="Select business" /></SelectTrigger>
   355→            <SelectContent>{safeBusinesses.map(b => (<SelectItem key={b.id} value={b.id}>{b.name}</SelectItem>))}</SelectContent>
   356→          </Select>
   357→        </div>
   358→        <Button size="sm" onClick={() => setCreateOpen(true)}><Plus className="h-4 w-4 mr-1" /> New Wallet</Button>
   359→      </div>
   360→
   361→      {/* KPIs */}
   362→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   363→        <Card className="border-emerald-200 bg-gradient-to-br from-emerald-50/50 to-white">
   364→          <CardContent className="p-4 sm:p-6 text-center">
   365→            <p className="text-xs text-muted-foreground mb-1">Portfolio (USD)</p>
   366→            <p className="text-2xl font-bold text-emerald-700">{formatCurrency(totalPortfolio)}</p>
   367→          </CardContent>
   368→        </Card>
   369→        <KPICard title="Total Balance" value={formatCurrency(totalBalance)} icon={WalletIcon} />
   370→        <KPICard title="Pending" value={formatCurrency(totalPending)} icon={RefreshCw} />
   371→        <KPICard title="Frozen" value={formatCurrency(totalFrozen)} icon={WalletIcon} />
   372→      </div>
   373→
   374→      {/* Wallet Cards */}
   375→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   376→        {allWallets.map(w => (
   377→          <Card key={w.id} className={`hover:shadow-md transition-shadow cursor-pointer ${selectedWalletId === w.id ? 'ring-2 ring-emerald-500' : ''}`} onClick={() => loadTransactions(w.id)}>
   378→            <CardContent className="p-4 sm:p-6">
   379→              <div className="flex items-center justify-between mb-3">
   380→                <div className="flex items-center gap-2">
   381→                  <span className="text-xl">{CURRENCY_FLAGS[w.currency] || '💰'}</span>
   382→                  <span className="font-semibold">{w.currency}</span>
   383→                  {w.isDefault && <Badge variant="outline" className="text-[10px]">Default</Badge>}
   384→                </div>
   385→                <Badge variant={w.status === 'active' ? 'default' : 'destructive'}>{w.status}</Badge>
   386→              </div>
   387→              <p className="text-2xl sm:text-3xl font-bold">{formatCurrency(w.balance, w.currency)}</p>
   388→              <div className="mt-3 space-y-1 text-xs text-muted-foreground">
   389→                <div className="flex justify-between"><span>Available</span><span className="text-emerald-600 font-medium">{formatCurrency(w.availableBalance, w.currency)}</span></div>
   390→                <div className="flex justify-between"><span>Pending</span><span className="text-amber-600 font-medium">{formatCurrency(w.pendingBalance, w.currency)}</span></div>
   391→                <div className="flex justify-between"><span>Frozen</span><span className="text-red-600 font-medium">{formatCurrency(w.frozenBalance, w.currency)}</span></div>
   392→              </div>
   393→            </CardContent>
   394→          </Card>
   395→        ))}
   396→      </div>
   397→
   398→      {allWallets.length === 0 && bizId && !wLoading && (
   399→        <Card className="border-dashed"><CardContent className="p-12 text-center"><p className="text-muted-foreground">No wallets for this business.</p></CardContent></Card>
   400→      )}
   401→
   402→      {/* Action Buttons + Transaction History */}
   403→      <AnimatePresence>
   404→        {selectedWalletId && selectedWallet && (
   405→          <motion.div initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: 'auto' }} exit={{ opacity: 0, height: 0 }}>
   406→            <div className="flex items-center justify-between mb-3 flex-wrap gap-2">
   407→              <h3 className="font-semibold">{selectedWallet.currency} Wallet — {CURRENCY_FLAGS[selectedWallet.currency] || ''}</h3>
   408→              <div className="flex gap-2 flex-wrap">
   409→                <Button size="sm" onClick={() => setDepositOpen(true)}><Download className="h-3.5 w-3.5 mr-1 text-emerald-500" /> Deposit</Button>
   410→                <Button size="sm" variant="outline" onClick={() => setWithdrawOpen(true)}><Upload className="h-3.5 w-3.5 mr-1 text-red-500" /> Withdraw</Button>
   411→                <Button size="sm" variant="outline" onClick={() => setConvertOpen(true)}><ArrowLeftRight className="h-3.5 w-3.5 mr-1 text-amber-500" /> Convert</Button>
   412→                <Button size="sm" variant="outline" onClick={() => setCryptoOpen(true)}><Bitcoin className="h-3.5 w-3.5 mr-1 text-orange-500" /> Crypto</Button>
   413→                <Button size="sm" variant="ghost" onClick={() => loadHistory(selectedWalletId, 'transactions')}>History</Button>
   414→              </div>
   415→            </div>
   416→            <Card>
   417→              <CardContent className="p-0">
   418→                <div className="overflow-x-auto">
   419→                  <Table>
   420→                    <TableHeader>
   421→                      <TableRow>
   422→                        <TableHead>Reference</TableHead>
   423→                        <TableHead>Type</TableHead>
   424→                        <TableHead>Description</TableHead>
   425→                        <TableHead className="text-right">Amount</TableHead>
   426→                        <TableHead className="text-right">Balance After</TableHead>
   427→                        <TableHead>Status</TableHead>
   428→                        <TableHead>Date</TableHead>
   429→                      </TableRow>
