   571→// Helper Components
   572→function ApiEndpoint({ 
   573→  method, 
   574→  path, 
   575→  description, 
   576→  body, 
   577→  params, 
   578→  response 
   579→}: { 
   580→  method: string; 
   581→  path: string; 
   582→  description: string;
   583→  body?: Record<string, string>;
   584→  params?: Record<string, string>;
   585→  response?: Record<string, string>;
   586→}) {
   587→  const getMethodColor = (m: string) => {
   588→    if (m.includes('POST')) return 'bg-emerald-500'
   589→    if (m.includes('PUT')) return 'bg-blue-500'
   590→    if (m.includes('DELETE')) return 'bg-red-500'
   591→    return 'bg-amber-500'
   592→  }
   593→
   594→  return (
   595→    <div className="space-y-3">
   596→      <div className="flex items-center gap-3">
   597→        <Badge className={`${getMethodColor(method)} text-white`}>
   598→          {method}
   599→        </Badge>
   600→        <code className="text-sm bg-muted px-3 py-1 rounded font-mono">{path}</code>
   601→      </div>
   602→      <p className="text-sm text-muted-foreground ml-20">{description}</p>
   603→      
   604→      {body && (
   605→        <div className="ml-20 space-y-1">
   606→          <p className="text-xs font-medium text-muted-foreground uppercase">Request Body:</p>
   607→          <pre className="text-xs bg-muted p-3 rounded overflow-x-auto">
   608→            {JSON.stringify(body, null, 2)}
   609→          </pre>
   610→        </div>
   611→      )}
   612→      
   613→      {params && (
   614→        <div className="ml-20 space-y-1">
   615→          <p className="text-xs font-medium text-muted-foreground uppercase">Query Params:</p>
   616→          <pre className="text-xs bg-muted p-3 rounded overflow-x-auto">
   617→            {JSON.stringify(params, null, 2)}
   618→          </pre>
   619→        </div>
   620→      )}
   621→      
   622→      {response && (
   623→        <div className="ml-20 space-y-1">
   624→          <p className="text-xs font-medium text-muted-foreground uppercase">Response:</p>
   625→          <pre className="text-xs bg-muted p-3 rounded overflow-x-auto">
   626→            {JSON.stringify(response, null, 2)}
   627→          </pre>
   628→        </div>
   629→      )}
   630→    </div>
   631→  )
   632→}
