   579→// Helper Components
   580→function ApiEndpoint({ 
   581→  method, 
   582→  path, 
   583→  description, 
   584→  body, 
   585→  params, 
   586→  response 
   587→}: { 
   588→  method: string; 
   589→  path: string; 
   590→  description: string;
   591→  body?: Record<string, string>;
   592→  params?: Record<string, string>;
   593→  response?: Record<string, string>;
   594→}) {
   595→  const getMethodColor = (m: string) => {
   596→    if (m.includes('POST')) return 'bg-emerald-500'
   597→    if (m.includes('PUT')) return 'bg-blue-500'
   598→    if (m.includes('DELETE')) return 'bg-red-500'
   599→    return 'bg-amber-500'
   600→  }
   601→
   602→  return (
   603→    <div className="space-y-3">
   604→      <div className="flex items-center gap-3">
   605→        <Badge className={`${getMethodColor(method)} text-white`}>
   606→          {method}
   607→        </Badge>
   608→        <code className="text-sm bg-muted px-3 py-1 rounded font-mono">{path}</code>
   609→      </div>
   610→      <p className="text-sm text-muted-foreground ml-20">{description}</p>
   611→      
   612→      {body && (
   613→        <div className="ml-20 space-y-1">
   614→          <p className="text-xs font-medium text-muted-foreground uppercase">Request Body:</p>
   615→          <pre className="text-xs bg-muted p-3 rounded overflow-x-auto">
   616→            {JSON.stringify(body, null, 2)}
   617→          </pre>
   618→        </div>
   619→      )}
   620→      
   621→      {params && (
   622→        <div className="ml-20 space-y-1">
   623→          <p className="text-xs font-medium text-muted-foreground uppercase">Query Params:</p>
   624→          <pre className="text-xs bg-muted p-3 rounded overflow-x-auto">
   625→            {JSON.stringify(params, null, 2)}
   626→          </pre>
   627→        </div>
   628→      )}
   629→      
   630→      {response && (
   631→        <div className="ml-20 space-y-1">
   632→          <p className="text-xs font-medium text-muted-foreground uppercase">Response:</p>
   633→          <pre className="text-xs bg-muted p-3 rounded overflow-x-auto">
   634→            {JSON.stringify(response, null, 2)}
   635→          </pre>
   636→        </div>
   637→      )}
   638→    </div>
   639→  )
   640→}
