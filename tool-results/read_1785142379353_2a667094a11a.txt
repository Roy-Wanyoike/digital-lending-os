     1→'use client'
     2→
     3→import { useState, useCallback } from 'react'
     4→import { motion, AnimatePresence } from 'framer-motion'
     5→import { Shield, Plus, Eye, DollarSign, Play, ChevronRight, AlertTriangle, CheckCircle, Clock, X, ExternalLink, CreditCard } from 'lucide-react'
     6→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
     7→import { Badge } from '@/components/ui/badge'
     8→import { Button } from '@/components/ui/button'
     9→import { Input } from '@/components/ui/input'
    10→import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table'
    11→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
    12→import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter } from '@/components/ui/dialog'
    13→import { Label } from '@/components/ui/label'
    14→import { Separator } from '@/components/ui/separator'
    15→import { ScrollArea } from '@/components/ui/scroll-area'
    16→import {
    17→  useApi, formatCurrency, getStatusBadgeVariant, getStatusColor,
    18→  getRiskBg, getRiskColor, formatDate, ESCROW_STATUSES, PipelineCard,
    19→  LoadingSkeleton, KPICard, CURRENCY_FLAGS, ScoreBar, Toast,
    20→  type EscrowTransaction, type Business,
    21→} from '@/lib/dashboard-helpers'
    22→
    23→// ─── Types ────────────────────────────────────────────────────
    24→
    25→interface Milestone { id: string; sequence: number; title: string; amount: number; status: string; releasedAt?: string | null }
    26→interface Disbursement { id: string; amount: number; currency: string; status: string; paymentRef?: string | null }
    27→interface Dispute { id: string; raisedBy: string; reason: string; description?: string | null; status: string; resolution?: string | null; aiRecommendation?: string | null; createdAt: string }
    28→interface AuditEntry { id: string; action: string; actor?: string | null; details: string; createdAt: string }
    29→interface EscrowDetail extends EscrowTransaction {
    30→  milestones: Milestone[]
    31→  disbursements: Disbursement[]
    32→  disputes: Dispute[]
    33→  auditLog: AuditEntry[]
    34→}
    35→
    36→// ─── Component ────────────────────────────────────────────────
    37→
    38→export function EscrowTab() {
    39→  const [refreshKey, setRefreshKey] = useState(0)
    40→  const { data: transactions, loading } = useApi<EscrowTransaction[]>(`/api/escrow/transactions?limit=50&k=${refreshKey}`)
    41→  const { data: businesses } = useApi<Business[]>('/api/businesses?limit=50')
    42→  const [statusFilter, setStatusFilter] = useState('all')
    43→  const [selectedId, setSelectedId] = useState<string | null>(null)
    44→  const [createOpen, setCreateOpen] = useState(false)
    45→  const [disputeOpen, setDisputeOpen] = useState(false)
    46→  const [fundOpen, setFundOpen] = useState(false)
    47→  const [fundEscrowId, setFundEscrowId] = useState<string | null>(null)
    48→  const [toastMsg, setToastMsg] = useState('')
    49→  const [toastVis, setToastVis] = useState(false)
    50→
    51→  // Fund form state
    52→  const [fundEmail, setFundEmail] = useState('')
    53→  const [fundFirstName, setFundFirstName] = useState('')
    54→  const [fundLastName, setFundLastName] = useState('')
    55→  const [fundProvider, setFundProvider] = useState('')
    56→  const [fundProviders, setFundProviders] = useState<Array<{code: string; name: string}>>([])
    57→  const [funding, setFunding] = useState(false)
    58→
    59→  // Create form state
    60→  const [formBuyer, setFormBuyer] = useState('')
    61→  const [formSeller, setFormSeller] = useState('')
    62→  const [formAmount, setFormAmount] = useState('')
    63→  const [formCurrency, setFormCurrency] = useState('USD')
    64→  const [formDesc, setFormDesc] = useState('')
    65→  const [formMilestones, setFormMilestones] = useState('1')
    66→  const [creating, setCreating] = useState(false)
    67→
    68→  // Dispute form state
    69→  const [disputeRaisedBy, setDisputeRaisedBy] = useState('buyer')
    70→  const [disputeReason, setDisputeReason] = useState('')
    71→  const [disputeDesc, setDisputeDesc] = useState('')
    72→  const [disputing, setDisputing] = useState(false)
    73→
    74→  const showToast = useCallback((msg: string) => { setToastMsg(msg); setToastVis(true); setTimeout(() => setToastVis(false), 3000) }, [])
    75→
    76→  const doAction = useCallback(async (id: string, action: string, body?: Record<string, string>) => {
    77→    try {
    78→      const res = await fetch(`/api/escrow/transactions/${id}/${action}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: body ? JSON.stringify(body) : undefined })
    79→      const json = await res.json()
    80→      if (!res.ok) throw new Error(json.error || 'Action failed')
    81→      showToast(`${action} successful!`)
    82→      setRefreshKey(k => k + 1)
    83→      return json.data
    84→    } catch (e: any) { showToast(e.message); return null }
    85→  }, [showToast])
    86→
    87→  const openFundDialog = useCallback(async (escrowId: string, currency: string) => {
    88→    setFundEscrowId(escrowId)
    89→    setFundOpen(true)
    90→    try {
    91→      const res = await fetch(`/api/payments/providers?currency=${currency}`)
    92→      const json = await res.json()
    93→      const providers = (json.data || []).map((p: any) => ({ code: p.code, name: p.name }))
    94→      setFundProviders(providers)
    95→      if (providers.length > 0) setFundProvider(providers[0].code)
    96→    } catch { setFundProviders([]) }
    97→  }, [])
    98→
    99→  if (loading) return <LoadingSkeleton />
   100→
   101→  const allTxns = transactions || []
   102→  const filtered = statusFilter === 'all' ? allTxns : allTxns.filter(t => t.status?.toLowerCase() === statusFilter.toLowerCase())
   103→
   104→  const pipelineCounts = ESCROW_STATUSES.map(s => ({
   105→    status: s,
   106→    count: allTxns.filter(t => t.status?.toLowerCase() === s.toLowerCase().replace(/\s/g, '_') || t.status?.toLowerCase().replace(/\s/g, '') === s.toLowerCase().replace(/\s/g, '')).length,
   107→  }))
   108→  const pipelineColors = ['#94a3b8', '#3b82f6', '#f59e0b', '#10b981', '#ef4444']
   109→
   110→  const totalVolume = allTxns.reduce((s, t) => s + t.amount, 0)
   111→  const activeEscrows = allTxns.filter(t => ['funded', 'in_escrow'].includes(t.status?.toLowerCase())).length
   112→  const disputed = allTxns.filter(t => t.status?.toLowerCase() === 'disputed').length
   113→
   114→  // ─── Actions ─────────────────────────────────────────────
   115→  const doAction = useCallback(async (id: string, action: string, body?: Record<string, string>) => {
   116→    try {
   117→      const res = await fetch(`/api/escrow/transactions/${id}/${action}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: body ? JSON.stringify(body) : undefined })
   118→      const json = await res.json()
   119→      if (!res.ok) throw new Error(json.error || 'Action failed')
   120→      showToast(`${action} successful!`)
   121→      setRefreshKey(k => k + 1)
   122→      return json.data
   123→    } catch (e: any) { showToast(e.message); return null }
   124→  }, [showToast])
   125→
   126→  const openFundDialog = useCallback(async (escrowId: string, currency: string) => {
   127→    setFundEscrowId(escrowId)
   128→    setFundOpen(true)
   129→    try {
   130→      const res = await fetch(`/api/payments/providers?currency=${currency}`)
   131→      const json = await res.json()
   132→      const providers = (json.data || []).map((p: any) => ({ code: p.code, name: p.name }))
   133→      setFundProviders(providers)
   134→      if (providers.length > 0) setFundProvider(providers[0].code)
   135→    } catch { setFundProviders([]) }
   136→  }, [])
   137→
   138→  const handleFund = async () => {
   139→    if (!fundEscrowId || !fundEmail) return
   140→    setFunding(true)
   141→    try {
   142→      const body: Record<string, string> = { email: fundEmail }
   143→      if (fundFirstName) body.firstName = fundFirstName
   144→      if (fundLastName) body.lastName = fundLastName
   145→      if (fundProvider) body.provider = fundProvider
   146→      const res = await fetch(`/api/escrow/transactions/${fundEscrowId}/fund`, {
   147→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   148→        body: JSON.stringify(body),
   149→      })
   150→      const json = await res.json()
   151→      if (!res.ok) throw new Error(json.error || 'Funding failed')
   152→      const data = json.data
   153→      if (data?.checkoutUrl) {
   154→        window.open(data.checkoutUrl, '_blank', 'noopener,noreferrer')
   155→        showToast(`Redirecting to ${data.providerName}...`)
   156→      } else if (data?.status === 'awaiting_payment') {
   157→        showToast('Payment initiated! Complete it via the provider.')
   158→      } else {
   159→        showToast('Escrow funding initiated!')
   160→      }
   161→      setFundOpen(false)
   162→      setFundEmail(''); setFundFirstName(''); setFundLastName(''); setFundProvider('')
   163→      setRefreshKey(k => k + 1)
   164→    } catch (e: any) { showToast(e.message) } finally { setFunding(false) }
   165→  }
   166→
   167→  const handleCreate = async () => {
   168→    setCreating(true)
   169→    try {
   170→      const milestones = Array.from({ length: parseInt(formMilestones) }, (_, i) => ({
   171→        title: `Milestone ${i + 1}`,
   172→        amount: Math.round((parseFloat(formAmount) / parseInt(formMilestones)) * 100) / 100,
   173→      }))
   174→      const res = await fetch('/api/escrow/transactions', {
   175→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   176→        body: JSON.stringify({ buyerId: formBuyer, sellerId: formSeller, amount: parseFloat(formAmount), currency: formCurrency, description: formDesc || undefined, milestones }),
   177→      })
   178→      const json = await res.json()
   179→      if (!res.ok) throw new Error(json.error || json.details?.[0]?.message || 'Create failed')
   180→      showToast('Escrow created!')
   181→      setCreateOpen(false)
   182→      setFormBuyer(''); setFormSeller(''); setFormAmount(''); setFormDesc(''); setFormMilestones('1')
   183→      setRefreshKey(k => k + 1)
   184→    } catch (e: any) { showToast(e.message) } finally { setCreating(false) }
   185→  }
   186→
   187→  const handleDispute = async () => {
   188→    if (!selectedId || !disputeReason) return
   189→    setDisputing(true)
   190→    try {
   191→      const res = await fetch(`/api/escrow/transactions/${selectedId}/disputes`, {
   192→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   193→        body: JSON.stringify({ raisedBy: disputeRaisedBy, reason: disputeReason, description: disputeDesc || undefined }),
   194→      })
   195→      const json = await res.json()
   196→      if (!res.ok) throw new Error(json.error || 'Dispute failed')
   197→      showToast('Dispute raised!')
   198→      setDisputeOpen(false); setDisputeReason(''); setDisputeDesc('')
   199→      setRefreshKey(k => k + 1)
   200→    } catch (e: any) { showToast(e.message) } finally { setDisputing(false) }
   201→  }
   202→
   203→  const selectedTxn = allTxns.find(t => t.id === selectedId)
   204→
   205→  return (
   206→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="space-y-6">
   207→      {/* KPIs */}
   208→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   209→        <KPICard title="Total Volume" value={formatCurrency(totalVolume)} icon={DollarSign} />
   210→        <KPICard title="Active Escrows" value={activeEscrows.toString()} icon={Shield} />
   211→        <KPICard title="Disputed" value={disputed.toString()} icon={AlertTriangle} />
   212→        <KPICard title="Total Deals" value={allTxns.length.toString()} icon={ChevronRight} />
   213→      </div>
   214→
   215→      {/* Pipeline */}
   216→      <div className="flex gap-2 sm:gap-3 overflow-x-auto pb-2">
   217→        {pipelineCounts.map((p, i) => (
   218→          <PipelineCard key={p.status} label={p.status} count={p.count} color={pipelineColors[i]} />
   219→        ))}
   220→      </div>
   221→
   222→      {/* Toolbar */}
   223→      <div className="flex flex-wrap items-center justify-between gap-3">
   224→        <div className="flex items-center gap-3">
   225→          <span className="text-sm text-muted-foreground">Status:</span>
   226→          <Select value={statusFilter} onValueChange={setStatusFilter}>
   227→            <SelectTrigger className="w-40"><SelectValue /></SelectTrigger>
   228→            <SelectContent>
   229→              <SelectItem value="all">All Statuses</SelectItem>
   230→              {ESCROW_STATUSES.map(s => <SelectItem key={s} value={s.toLowerCase().replace(/\s/g, '_')}>{s}</SelectItem>)}
   231→            </SelectContent>
   232→          </Select>
   233→        </div>
   234→        <Button size="sm" onClick={() => setCreateOpen(true)}><Plus className="h-4 w-4 mr-1" /> New Escrow</Button>
   235→      </div>
   236→
   237→      {/* Table */}
   238→      <Card>
   239→        <CardContent className="p-0">
   240→          <div className="overflow-x-auto">
   241→            <Table>
   242→              <TableHeader>
   243→                <TableRow>
   244→                  <TableHead>Reference</TableHead>
   245→                  <TableHead>Buyer → Seller</TableHead>
   246→                  <TableHead className="text-right">Amount</TableHead>
   247→                  <TableHead>Status</TableHead>
   248→                  <TableHead>AI Risk</TableHead>
   249→                  <TableHead>Milestones</TableHead>
   250→                  <TableHead>Actions</TableHead>
   251→                  <TableHead>Date</TableHead>
   252→                </TableRow>
   253→              </TableHeader>
   254→              <TableBody>
   255→                {filtered.slice(0, 20).map(txn => {
   256→                  const st = txn.status?.toLowerCase() || ''
   257→                  const canFund = st === 'created'
   258→                  const canActivate = st === 'funded'
   259→                  const canRelease = st === 'in_escrow'
   260→                  const canDispute = ['funded', 'in_escrow'].includes(st)
   261→                  return (
   262→                    <TableRow key={txn.id} className="even:bg-muted/50">
   263→                      <TableCell className="font-mono text-xs">{txn.txRef}</TableCell>
   264→                      <TableCell className="max-w-[160px] truncate text-xs">
   265→                        <span className="text-foreground">{txn.buyer?.name}</span>
   266→                        <span className="text-muted-foreground mx-1">→</span>
   267→                        <span className="text-foreground">{txn.seller?.name}</span>
   268→                      </TableCell>
   269→                      <TableCell className="text-right font-medium">{formatCurrency(txn.amount, txn.currency)}</TableCell>
   270→                      <TableCell><Badge variant={getStatusBadgeVariant(txn.status)} className={getStatusColor(txn.status)}>{txn.status}</Badge></TableCell>
   271→                      <TableCell>
   272→                        <div className="flex items-center gap-2">
   273→                          <div className="w-10 h-1.5 bg-slate-100 rounded-full overflow-hidden">
   274→                            <div className={`h-full rounded-full ${getRiskBg(txn.aiRiskScore || 0)}`} style={{ width: `${txn.aiRiskScore || 0}%` }} />
   275→                          </div>
   276→                          <span className={`text-xs font-medium ${getRiskColor(txn.aiRiskScore || 0)}`}>{txn.aiRiskLevel}</span>
   277→                        </div>
   278→                      </TableCell>
   279→                      <TableCell className="text-xs">{txn.currentMilestone}/{txn.totalMilestones}</TableCell>
   280→                      <TableCell>
   281→                        <div className="flex items-center gap-1">
   282→                          <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => setSelectedId(txn.id)} title="View details"><Eye className="h-3.5 w-3.5" /></Button>
   283→                          {canFund && <Button variant="outline" size="sm" className="h-7 text-xs" onClick={() => openFundDialog(txn.id, txn.currency)}><CreditCard className="h-3 w-3 mr-1" />Fund</Button>}
   284→                          {canActivate && <Button variant="outline" size="sm" className="h-7 text-xs" onClick={() => doAction(txn.id, 'activate')}><Play className="h-3 w-3 mr-1" />Activate</Button>}
   285→                          {canRelease && <Button variant="outline" size="sm" className="h-7 text-xs" onClick={() => {
   286→                            const ms = txn.milestones?.find((m: any) => m.status === 'ready' || m.status === 'pending')
   287→                            if (ms) doAction(txn.id, 'release', { milestoneId: ms.id })
   288→                            else showToast('No pending milestones')
   289→                          }}><CheckCircle className="h-3 w-3 mr-1" />Release</Button>}
   290→                          {canDispute && <Button variant="outline" size="sm" className="h-7 text-xs text-red-600" onClick={() => { setSelectedId(txn.id); setDisputeOpen(true) }}><AlertTriangle className="h-3 w-3 mr-1" />Dispute</Button>}
   291→                        </div>
   292→                      </TableCell>
   293→                      <TableCell className="text-xs text-muted-foreground whitespace-nowrap">{formatDate(txn.createdAt)}</TableCell>
   294→                    </TableRow>
   295→                  )
   296→                })}
   297→              </TableBody>
   298→            </Table>
   299→          </div>
   300→        </CardContent>
   301→      </Card>
   302→
   303→      {/* Detail Drawer */}
   304→      <AnimatePresence>
   305→        {selectedTxn && (
   306→          <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} className="fixed inset-0 z-50 flex justify-end">
   307→            <div className="absolute inset-0 bg-black/30" onClick={() => setSelectedId(null)} />
   308→            <motion.div initial={{ x: '100%' }} animate={{ x: 0 }} exit={{ x: '100%' }} transition={{ type: 'spring', damping: 25, stiffness: 300 }} className="relative w-full max-w-xl bg-background shadow-xl h-full overflow-y-auto border-l">
   309→              <div className="sticky top-0 z-10 bg-background border-b px-6 py-4 flex items-center justify-between">
   310→                <div>
   311→                  <h3 className="font-semibold">{selectedTxn.txRef}</h3>
   312→                  <p className="text-xs text-muted-foreground">{formatDate(selectedTxn.createdAt)}</p>
   313→                </div>
   314→                <Button variant="ghost" size="icon" onClick={() => setSelectedId(null)}><X className="h-4 w-4" /></Button>
   315→              </div>
   316→              <div className="p-6 space-y-6">
   317→                {/* Status & Amount */}
   318→                <div className="flex items-center justify-between">
   319→                  <Badge variant={getStatusBadgeVariant(selectedTxn.status)} className={getStatusColor(selectedTxn.status)}>{selectedTxn.status}</Badge>
   320→                  <p className="text-2xl font-bold">{formatCurrency(selectedTxn.amount, selectedTxn.currency)}</p>
   321→                </div>
   322→                <div className="grid grid-cols-2 gap-4 text-sm">
   323→                  <div><p className="text-muted-foreground">Buyer</p><p className="font-medium">{selectedTxn.buyer?.name}</p></div>
   324→                  <div><p className="text-muted-foreground">Seller</p><p className="font-medium">{selectedTxn.seller?.name}</p></div>
   325→                  <div><p className="text-muted-foreground">Funded</p><p className="font-medium text-blue-600 dark:text-blue-400">{formatCurrency(selectedTxn.fundedAmount, selectedTxn.currency)}</p></div>
   326→                  <div><p className="text-muted-foreground">Released</p><p className="font-medium text-emerald-600 dark:text-emerald-400">{formatCurrency(selectedTxn.releasedAmount, selectedTxn.currency)}</p></div>
   327→                </div>
   328→
   329→                {/* Risk Score */}
   330→                <div>
   331→                  <p className="text-sm font-medium mb-2">AI Risk Assessment</p>
   332→                  <ScoreBar score={selectedTxn.aiRiskScore || 0} label="Risk Score" />
   333→                </div>
   334→
   335→                {/* Milestones */}
   336→                {selectedTxn.milestones && selectedTxn.milestones.length > 0 && (
   337→                  <div>
   338→                    <p className="text-sm font-medium mb-2">Milestones</p>
   339→                    <div className="space-y-2">
   340→                      {selectedTxn.milestones.map((m: any) => (
   341→                        <div key={m.id} className="flex items-center justify-between p-3 rounded-lg bg-muted">
   342→                          <div className="flex items-center gap-2">
   343→                            {m.status === 'released' ? <CheckCircle className="h-4 w-4 text-emerald-500" /> : <Clock className="h-4 w-4 text-muted-foreground" />}
   344→                            <span className="text-sm font-medium">{m.title}</span>
   345→                          </div>
   346→                          <span className="text-sm font-medium">{formatCurrency(m.amount, selectedTxn.currency)}</span>
   347→                        </div>
   348→                      ))}
   349→                    </div>
   350→                  </div>
   351→                )}
   352→
   353→                {/* Disputes */}
   354→                {selectedTxn.disputes && selectedTxn.disputes.length > 0 && (
   355→                  <div>
   356→                    <p className="text-sm font-medium mb-2">Disputes</p>
   357→                    {selectedTxn.disputes.map((d: any) => (
   358→                      <div key={d.id} className="p-3 rounded-lg bg-red-50 dark:bg-red-950/30 border border-red-100 dark:border-red-900 mb-2">
   359→                        <div className="flex items-center justify-between mb-1">
   360→                          <Badge variant={d.status === 'open' ? 'destructive' : 'secondary'}>{d.status}</Badge>
   361→                          <span className="text-xs text-muted-foreground">{formatDate(d.createdAt)}</span>
   362→                        </div>
   363→                        <p className="text-sm font-medium">{d.reason}</p>
   364→                        {d.aiRecommendation && <p className="text-xs text-muted-foreground mt-1">AI: {d.aiRecommendation}</p>
   365→                        {d.resolution && <p className="text-xs text-emerald-600 mt-1">Resolution: {d.resolution}</p>}
   366→                      </div>
   367→                    ))}
   368→                  </div>
   369→                )}
   370→
   371→                {/* Description */}
   372→                {selectedTxn.description && (
   373→                  <div>
   374→                    <p className="text-sm font-medium mb-1">Description</p>
   375→                    <p className="text-sm text-muted-foreground">{selectedTxn.description}</p>
   376→                  </div>
   377→                )}
   378→
   379→                {/* Fees */}
   380→                <div className="grid grid-cols-2 gap-4 text-sm">
   381→                  <div><p className="text-muted-foreground">Fee</p><p>{formatCurrency(selectedTxn.feeAmount, selectedTxn.feeCurrency)}</p></div>
   382→                  <div><p className="text-muted-foreground">Refunded</p><p>{formatCurrency(selectedTxn.refundedAmount, selectedTxn.currency)}</p></div>
   383→                </div>
   384→              </div>
   385→            </motion.div>
   386→          </motion.div>
   387→        )}
   388→      </AnimatePresence>
   389→
   390→      {/* Fund Dialog */}
   391→      <Dialog open={fundOpen} onOpenChange={setFundOpen}>
   392→        <DialogContent className="max-w-md">
   393→          <DialogHeader>
   394→            <DialogTitle className="flex items-center gap-2"><CreditCard className="h-5 w-5" /> Fund Escrow</DialogTitle>
   395→            <DialogDescription>Pay via a payment provider to fund this escrow. You will be redirected to complete payment.</DialogDescription>
   396→          </DialogHeader>
   397→          <div className="space-y-4">
   398→            {fundProviders.length > 0 && (
   399→              <div className="space-y-2"><Label>Payment Provider</Label>
   400→                <Select value={fundProvider} onValueChange={setFundProvider}>
   401→                  <SelectTrigger><SelectValue placeholder="Select provider" /></SelectTrigger>
