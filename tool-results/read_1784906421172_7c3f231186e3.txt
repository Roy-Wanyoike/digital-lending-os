    71→}

    96→
    97→export function WalletTab() {
    98→  const { data: businesses, loading: bLoading } = useApi<Business[]>('/api/businesses?limit=50')
    99→  const { data: rates } = useApi<RatesData>('/api/wallets/rates')
   100→  const [selectedBizId, setSelectedBizId] = useState<string>('')
   101→  const bizId = selectedBizId || businesses?.[0]?.id || ''
   102→  const [walletKey, setWalletKey] = useState(0)
   103→  const { data: wallets, loading: wLoading } = useApi<WalletData[]>(bizId ? `/api/wallets?businessId=${bizId}&k=${walletKey}` : '')
   104→  const [selectedWalletId, setSelectedWalletId] = useState<string | null>(null)
   105→  const [txns, setTxns] = useState<WalletTransaction[]>([])
   106→  const [txLoading, setTxLoading] = useState(false)
   107→
   108→  // Dialogs
   109→  const [createOpen, setCreateOpen] = useState(false)
   110→  const [depositOpen, setDepositOpen] = useState(false)
   111→  const [withdrawOpen, setWithdrawOpen] = useState(false)
   112→  const [convertOpen, setConvertOpen] = useState(false)
   113→  const [cryptoOpen, setCryptoOpen] = useState(false)
   114→
   115→  // History dialogs
   116→  const [historyTab, setHistoryTab] = useState('transactions')
   117→  const [deposits, setDeposits] = useState<DepositRecord[]>([])
   118→  const [withdrawals, setWithdrawals] = useState<WithdrawalRecord[]>([])
   119→  const [cryptoWdrs, setCryptoWdrs] = useState<CryptoWithdrawalRecord[]>([])
   120→  const [historyOpen, setHistoryOpen] = useState(false)
   121→
   122→  const [toastMsg, setToastMsg] = useState('')
   123→  const [toastVis, setToastVis] = useState(false)
   124→
   125→  // Create wallet form
   126→  const [newCurrency, setNewCurrency] = useState('USD')
   127→  const [creating, setCreating] = useState(false)
   128→
   129→  // Deposit form
   130→  const [depAmount, setDepAmount] = useState('')
   131→  const [depMethod, setDepMethod] = useState('bank_transfer')
   132→  const [depNotes, setDepNotes] = useState('')
   133→  const [depSubmitting, setDepSubmitting] = useState(false)
   134→
   135→  // Withdraw form
   136→  const [wdrAmount, setWdrAmount] = useState('')
   137→  const [wdrMethod, setWdrMethod] = useState('bank_transfer')
   138→  const [wdrBankName, setWdrBankName] = useState('')
   139→  const [wdrBankAccount, setWdrBankAccount] = useState('')
   140→  const [wdrRecipientName, setWdrRecipientName] = useState('')
   141→  const [wdrNotes, setWdrNotes] = useState('')
   142→  const [wdrSubmitting, setWdrSubmitting] = useState(false)
   143→
   144→  // Convert form
   145→  const [cvtFromWalletId, setCvtFromWalletId] = useState('')
   146→  const [cvtToWalletId, setCvtToWalletId] = useState('')
   147→  const [cvtAmount, setCvtAmount] = useState('')
   148→  const [cvtSubmitting, setCvtSubmitting] = useState(false)
   149→
   150→  // Crypto form
   151→  const [crCrypto, setCrCrypto] = useState('USDT')
   152→  const [crNetwork, setCrNetwork] = useState('trc20')
   153→  const [crAddress, setCrAddress] = useState('')
   154→  const [crAmount, setCrAmount] = useState('')
   155→  const [crNotes, setCrNotes] = useState('')
   156→  const [crSubmitting, setCrSubmitting] = useState(false)
   157→
   158→  const showToast = useCallback((msg: string) => { setToastMsg(msg); setToastVis(true); setTimeout(() => setToastVis(false), 3000) }, [])
   159→
   160→  const loadTransactions = useCallback(async (walletId: string) => {
   161→    setSelectedWalletId(walletId)
   162→    setTxLoading(true)
   163→    try {
   164→      const res = await fetch(`/api/wallets/${walletId}/transactions?limit=50`)
   165→      const json = await res.json()
   166→      setTxns(json.data || [])
   167→    } catch { setTxns([]) } finally { setTxLoading(false) }
   168→  }, [])
   169→
   170→  const loadHistory = useCallback(async (walletId: string, tab: string) => {
   171→    setHistoryTab(tab)
   172→    setHistoryOpen(true)
   173→    try {
   174→      if (tab === 'deposits') {
   175→        const r = await fetch(`/api/wallets/deposit?walletId=${walletId}&limit=50`)
   176→        setDeposits((await r.json()).data || [])
   177→      } else if (tab === 'withdrawals') {
   178→        const r = await fetch(`/api/wallets/withdrawal?walletId=${walletId}&limit=50`)
   179→        setWithdrawals((await r.json()).data || [])
   180→      } else if (tab === 'crypto') {
   181→        const r = await fetch(`/api/wallets/crypto-withdrawal?walletId=${walletId}&limit=50`)
   182→        setCryptoWdrs((await r.json()).data || [])
   183→      }
   184→    } catch { /* ignore */ }
   185→  }, [])
   186→
   187→  const allWallets = wallets || []
   188→  const totalPortfolio = allWallets.reduce((sum, w) => {
   189→    try {
   190→      const toUsd = rates?.fiatToUsd?.[w.currency] || 1
   191→      return sum + w.balance * toUsd
   192→    } catch { return sum }
   193→  }, 0)
   194→  const totalBalance = allWallets.reduce((s, w) => s + w.balance, 0)
   195→  const totalPending = allWallets.reduce((s, w) => s + w.pendingBalance, 0)
   196→  const totalFrozen = allWallets.reduce((s, w) => s + w.frozenBalance, 0)
   197→
   198→  const selectedWallet = allWallets.find(w => w.id === selectedWalletId)
   199→
   200→  // Conversion preview
   201→  const conversionPreview = (() => {
   202→    if (!cvtAmount || !cvtFromWalletId || !cvtToWalletId || !rates) return null
   203→    const from = allWallets.find(w => w.id === cvtFromWalletId)
   204→    const to = allWallets.find(w => w.id === cvtToWalletId)
   205→    if (!from || !to) return null
   206→    const amount = parseFloat(cvtAmount)
   207→    if (isNaN(amount) || amount <= 0) return null
   208→    const direct = rates.fiatRates[from.currency]?.[to.currency]
   209→    if (!direct) return null
   210→    const gross = amount * direct
   211→    const fee = Math.round(gross * 0.005 * 100) / 100
   212→    const net = Math.round((gross - fee) * 100) / 100
   213→    return { rate: direct, gross, fee, net }
   214→  })()
   215→
   216→  // Crypto withdrawal preview
   217→  const cryptoPreview = (() => {
   218→    if (!crAmount || !selectedWallet || !rates) return null
   219→    const amount = parseFloat(crAmount)
   220→    if (isNaN(amount) || amount <= 0) return null
   221→    const toUsd = rates.fiatToUsd[selectedWallet.currency] || 1
   222→    const usdAmount = amount * toUsd
   223→    const cryptoPrice = rates.cryptoPrices[crCrypto]
   224→    if (!cryptoPrice) return null
   225→    const cryptoAmt = usdAmount / cryptoPrice
   226→    const netFee = rates.networkFees[crNetwork] || 0
   227→    const netCrypto = Math.max(0, cryptoAmt - netFee)
   228→    const procFee = Math.max(amount * (rates.cryptoWithdrawalFeePercent / 100), rates.cryptoWithdrawalMinFee)
   229→    const totalDebit = amount + procFee
   230→    return { usdAmount, cryptoAmt, netCrypto, netFee, procFee, totalDebit, cryptoPrice }
   231→  })()
   232→
   233→  const handleCreateWallet = async () => {
   234→    if (!bizId) return
   235→    setCreating(true)
   236→    try {
   237→      const res = await fetch('/api/wallets', {
   238→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   239→        body: JSON.stringify({ businessId: bizId, currency: newCurrency, isDefault: allWallets.length === 0 }),
   240→      })
   241→      const json = await res.json()
   242→      if (!res.ok) throw new Error(json.error || 'Failed')
   243→      showToast('Wallet created!')
   244→      setCreateOpen(false)
   245→      setWalletKey(k => k + 1)
   246→    } catch (e: any) { showToast(e.message) } finally { setCreating(false) }
   247→  }
   248→
   249→  const handleDeposit = async () => {
   250→    if (!selectedWalletId || !depAmount) return
   251→    setDepSubmitting(true)
   252→    try {
   253→      const res = await fetch('/api/wallets/deposit', {
   254→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   255→        body: JSON.stringify({
   256→          walletId: selectedWalletId, amount: parseFloat(depAmount),
   257→          paymentMethod: depMethod, provider: 'demo', notes: depNotes || undefined,
   258→        }),
   259→      })
   260→      const json = await res.json()
   261→      if (!res.ok) throw new Error(json.error || 'Failed')
   262→      showToast('Deposit completed!')
   263→      setDepositOpen(false); setDepAmount(''); setDepNotes('')
   264→      setWalletKey(k => k + 1)
   265→      loadTransactions(selectedWalletId)
   266→    } catch (e: any) { showToast(e.message) } finally { setDepSubmitting(false) }
   267→  }
   268→
   269→  const handleWithdraw = async () => {
   270→    if (!selectedWalletId || !wdrAmount) return
   271→    setWdrSubmitting(true)
   272→    try {
   273→      const res = await fetch('/api/wallets/withdrawal', {
   274→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   275→        body: JSON.stringify({
   276→          walletId: selectedWalletId, amount: parseFloat(wdrAmount),
   277→          paymentMethod: wdrMethod, provider: 'demo',
   278→          bankName: wdrBankName || undefined, bankAccount: wdrBankAccount || undefined,
   279→          recipientName: wdrRecipientName || undefined, notes: wdrNotes || undefined,
   280→        }),
   281→      })
   282→      const json = await res.json()
   283→      if (!res.ok) throw new Error(json.error || 'Failed')
   284→      showToast('Withdrawal completed!')
   285→      setWithdrawOpen(false); setWdrAmount(''); setWdrBankName(''); setWdrBankAccount(''); setWdrRecipientName(''); setWdrNotes('')
   286→      setWalletKey(k => k + 1)
   287→      loadTransactions(selectedWalletId)
   288→    } catch (e: any) { showToast(e.message) } finally { setWdrSubmitting(false) }
   289→  }
   290→
   291→  const handleConvert = async () => {
   292→    if (!cvtFromWalletId || !cvtToWalletId || !cvtAmount) return
   293→    setCvtSubmitting(true)
   294→    try {
   295→      const res = await fetch('/api/wallets/convert', {
   296→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   297→        body: JSON.stringify({ fromWalletId: cvtFromWalletId, toWalletId: cvtToWalletId, fromAmount: parseFloat(cvtAmount) }),
   298→      })
   299→      const json = await res.json()
   300→      if (!res.ok) throw new Error(json.error || 'Failed')
   301→      showToast('Conversion completed!')
   302→      setConvertOpen(false); setCvtAmount('')
   303→      setWalletKey(k => k + 1)
   304→      loadTransactions(selectedWalletId)
   305→    } catch (e: any) { showToast(e.message) } finally { setCvtSubmitting(false) }
   306→  }
   307→
   308→  const handleCryptoWithdraw = async () => {
   309→    if (!selectedWalletId || !crAmount || !crAddress) return
   310→    setCrSubmitting(true)
   311→    try {
   312→      const res = await fetch('/api/wallets/crypto-withdrawal', {
   313→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   314→        body: JSON.stringify({
   315→          walletId: selectedWalletId, amount: parseFloat(crAmount),
   316→          cryptoCurrency: crCrypto, network: crNetwork, walletAddress: crAddress.trim(),
   317→          notes: crNotes || undefined,
   318→        }),
   319→      })
   320→      const json = await res.json()
   321→      if (!res.ok) throw new Error(json.error || 'Failed')
   322→      showToast('Crypto withdrawal submitted!')
   323→      setCryptoOpen(false); setCrAmount(''); setCrAddress(''); setCrNotes('')
   324→      setWalletKey(k => k + 1)
   325→      loadTransactions(selectedWalletId)
   326→    } catch (e: any) { showToast(e.message) } finally { setCrSubmitting(false) }
   327→  }
   328→
   329→  // When crypto changes, reset network to first valid
   330→  useEffect(() => {
   331→    if (rates) {
   332→      const nets = rates.cryptoNetworks[crCrypto]
   333→      if (nets && nets.length > 0) setCrNetwork(nets[0])
   334→    }
   335→  }, [crCrypto, rates])
   336→
   337→  // When convert source wallet changes, clear target if same
   338→  useEffect(() => {
   339→    if (cvtFromWalletId === cvtToWalletId) setCvtToWalletId('')
   340→  }, [cvtFromWalletId])
   341→
   342→  if (bLoading || (bizId && wLoading)) return <LoadingSkeleton />
   343→
   344→  return (
   345→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="space-y-6">
   346→      {/* Business selector */}
   347→      <div className="flex items-center justify-between flex-wrap gap-3">
   348→        <div className="flex items-center gap-3">
   349→          <span className="text-sm text-slate-500">Business:</span>
   350→          <Select value={selectedBizId || businesses?.[0]?.id} onValueChange={setSelectedBizId}>
   351→            <SelectTrigger className="w-64"><SelectValue placeholder="Select business" /></SelectTrigger>
   352→            <SelectContent>{(businesses || []).map(b => (<SelectItem key={b.id} value={b.id}>{b.name}</SelectItem>))}</SelectContent>
   353→          </Select>
   354→        </div>
   355→        <Button size="sm" onClick={() => setCreateOpen(true)}><Plus className="h-4 w-4 mr-1" /> New Wallet</Button>
   356→      </div>
   357→
   358→      {/* KPIs */}
   359→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   360→        <Card className="border-emerald-200 bg-gradient-to-br from-emerald-50/50 to-white">
   361→          <CardContent className="p-4 sm:p-6 text-center">
   362→            <p className="text-xs text-slate-500 mb-1">Portfolio (USD)</p>
   363→            <p className="text-2xl font-bold text-emerald-700">{formatCurrency(totalPortfolio)}</p>
   364→          </CardContent>
   365→        </Card>
   366→        <KPICard title="Total Balance" value={formatCurrency(totalBalance)} icon={WalletIcon} />
   367→        <KPICard title="Pending" value={formatCurrency(totalPending)} icon={RefreshCw} />
   368→        <KPICard title="Frozen" value={formatCurrency(totalFrozen)} icon={WalletIcon} />
   369→      </div>
   370→
   371→      {/* Wallet Cards */}
   372→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   373→        {allWallets.map(w => (
   374→          <Card key={w.id} className={`hover:shadow-md transition-shadow cursor-pointer ${selectedWalletId === w.id ? 'ring-2 ring-emerald-500' : ''}`} onClick={() => loadTransactions(w.id)}>
   375→            <CardContent className="p-4 sm:p-6">
   376→              <div className="flex items-center justify-between mb-3">
   377→                <div className="flex items-center gap-2">
   378→                  <span className="text-xl">{CURRENCY_FLAGS[w.currency] || '💰'}</span>
   379→                  <span className="font-semibold">{w.currency}</span>
   380→                  {w.isDefault && <Badge variant="outline" className="text-[10px]">Default</Badge>}
   381→                </div>
   382→                <Badge variant={w.status === 'active' ? 'default' : 'destructive'}>{w.status}</Badge>
   383→              </div>
   384→              <p className="text-2xl sm:text-3xl font-bold">{formatCurrency(w.balance, w.currency)}</p>
   385→              <div className="mt-3 space-y-1 text-xs text-slate-500">
   386→                <div className="flex justify-between"><span>Available</span><span className="text-emerald-600 font-medium">{formatCurrency(w.availableBalance, w.currency)}</span></div>
   387→                <div className="flex justify-between"><span>Pending</span><span className="text-amber-600 font-medium">{formatCurrency(w.pendingBalance, w.currency)}</span></div>
   388→                <div className="flex justify-between"><span>Frozen</span><span className="text-red-600 font-medium">{formatCurrency(w.frozenBalance, w.currency)}</span></div>
   389→              </div>
   390→            </CardContent>
   391→          </Card>
   392→        ))}
   393→      </div>
   394→
   395→      {allWallets.length === 0 && bizId && !wLoading && (
   396→        <Card className="border-dashed"><CardContent className="p-12 text-center"><p className="text-slate-500">No wallets for this business.</p></CardContent></Card>
   397→      )}
   398→
   399→      {/* Action Buttons + Transaction History */}
   400→      <AnimatePresence>
   401→        {selectedWalletId && selectedWallet && (
   402→          <motion.div initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: 'auto' }} exit={{ opacity: 0, height: 0 }}>
   403→            <div className="flex items-center justify-between mb-3 flex-wrap gap-2">
   404→              <h3 className="font-semibold">{selectedWallet.currency} Wallet — {CURRENCY_FLAGS[selectedWallet.currency] || ''}</h3>
   405→              <div className="flex gap-2 flex-wrap">
   406→                <Button size="sm" onClick={() => setDepositOpen(true)}><Download className="h-3.5 w-3.5 mr-1 text-emerald-500" /> Deposit</Button>
   407→                <Button size="sm" variant="outline" onClick={() => setWithdrawOpen(true)}><Upload className="h-3.5 w-3.5 mr-1 text-red-500" /> Withdraw</Button>
   408→                <Button size="sm" variant="outline" onClick={() => setConvertOpen(true)}><ArrowLeftRight className="h-3.5 w-3.5 mr-1 text-amber-500" /> Convert</Button>
   409→                <Button size="sm" variant="outline" onClick={() => setCryptoOpen(true)}><Bitcoin className="h-3.5 w-3.5 mr-1 text-orange-500" /> Crypto</Button>
   410→                <Button size="sm" variant="ghost" onClick={() => loadHistory(selectedWalletId, 'transactions')}>History</Button>
   411→              </div>
   412→            </div>
   413→            <Card>
   414→              <CardContent className="p-0">
   415→                <div className="overflow-x-auto">
   416→                  <Table>
   417→                    <TableHeader>
   418→                      <TableRow>
   419→                        <TableHead>Reference</TableHead>
   420→                        <TableHead>Type</TableHead>
   421→                        <TableHead>Description</TableHead>
   422→                        <TableHead className="text-right">Amount</TableHead>
   423→                        <TableHead className="text-right">Balance After</TableHead>
   424→                        <TableHead>Status</TableHead>
   425→                        <TableHead>Date</TableHead>
   426→                      </TableRow>
   427→                    </TableHeader>
