   712→async function generateLoans(tenantId: string, customerIds: string[], productIds: string[]) {

   722→

   724→  const arrearsStatuses = ['CURRENT', 'DAYS_1_7', 'DAYS_8_30', 'DAYS_31_60', 'DAYS_61_90', 'DAYS_91_PLUS'] as const
   725→
   726→  const loans = []
   727→

   731→  for (const app of disbursedApps) {

   745→      applicationId: app.id,
   746→      productId: app.productId,
   747→      loanNumber: `LN-${new Date().getFullYear()}-${String(globalLoanCount).padStart(6, '0')}`,

   749→      approvedAmount: principal,
   750→      interestRate: 15,
   751→      interestType: 'FLAT_RATE' as const,

   762→      repaidInterest: status === 'FULLY_PAID' ? totalInterest : status === 'ACTIVE' ? Math.floor(totalInterest * Math.random() * 0.5) : 0,
   763→      repaidFees: status === 'FULLY_PAID' ? 330 : 0,

   766→      nextPaymentDue: new Date(Date.now() + 15 * 24 * 60 * 60 * 1000),
   767→      daysInArrears,

   784→    })
   785→  }

   787→  return { count: 0 }

   790→// Generate repayment schedule
   791→function generateSchedule(principal: number, rate: number, days: number): Array<object> {
   792→  const schedule = []
   793→  const installments = Math.max(1, Math.ceil(days / 30))
   794→  const perInstallment = principal / installments
   795→  const interestPerInstallment = (principal * rate / 100) / installments
   796→
   797→  for (let i = 1; i <= installments; i++) {
   798→    schedule.push({
   799→      installmentNo: i,
   800→      dueDate: new Date(Date.now() + i * 30 * 24 * 60 * 60 * 1000).toISOString().split('T')[0],
   801→      principal: Math.round(perInstallment),
   802→      interest: Math.round(interestPerInstallment),
   803→      fees: i === 1 ? 110 : 0,
   804→      total: Math.round(perInstallment + interestPerInstallment + (i === 1 ? 110 : 0)),
   805→      status: i === 1 ? 'PENDING' : 'SCHEDULED'
   806→    })
   807→  }
   808→
   809→  return schedule
   810→}

   812→// Generate repayments
   813→async function generateRepayments(tenantId: string) {
   814→  const activeLoans = await prisma.loan.findMany({
   815→    where: {
   816→      tenantId,
   817→      status: { in: ['ACTIVE', 'FULLY_PAID'] },
   818→      repaidPrincipal: { gt: 0 }
   819→    },
   820→    take: 100
   821→  })
   822→
   823→  const repayments = []
   824→
   825→  for (const loan of activeLoans) {
   826→    const numPayments = Math.floor(Math.random() * 3) + 1
   827→    const remainingToPay = loan.totalRepaid
   828→
   829→    for (let i = 0; i < numPayments && remainingToPay > 0; i++) {
   830→      const amount = Math.min(
   831→        Math.round(remainingToPay / (numPayments - i)),
   832→        remainingToPay
   833→      )
   834→
   835→      repayments.push({
   836→        tenantId,
   837→        loanId: loan.id,
   838→        customerId: loan.customerId,
   839→        amount,
   840→        principalPortion: Math.round(amount * 0.85),
   841→        interestPortion: Math.round(amount * 0.13),
   842→        feePortion: Math.round(amount * 0.02),
   843→        paymentMethod: 'MPESA' as const,
   844→        referenceNumber: `MPESA${Math.random().toString(36).substring(2, 11).toUpperCase()}`,
   845→        paidBy: null,
   846→        paymentDate: new Date(Date.now() - Math.random() * 30 * 24 * 60 * 60 * 1000),
   847→        dueDate: new Date(Date.now() - Math.random() * 35 * 24 * 60 * 60 * 1000),
   848→        status: 'COMPLETED' as const
   849→      })
   850→    }
   851→  }
   852→
   853→  if (repayments.length > 0) {
   854→    return await prisma.repayment.createMany({
   855→      data: repayments
   856→    })
   857→  }
   858→
   859→  return { count: 0 }
   860→}
