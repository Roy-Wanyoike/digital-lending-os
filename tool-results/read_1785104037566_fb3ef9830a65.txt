    96→
    97→export function WalletTab() {
    98→  const { data: businesses, loading: bLoading } = useApi<Business[]>('/api/businesses?limit=50')
    99→  const { data: rates } = useApi<RatesData>('/api/wallets/rates')
   100→  const [selectedBizId, setSelectedBizId] = useState<string>('')
   101→  const safeBusinesses = Array.isArray(businesses) ? businesses : []
   102→  const bizId = selectedBizId || safeBusinesses?.[0]?.id || ''
   103→  const [walletKey, setWalletKey] = useState(0)
   104→  const { data: wallets, loading: wLoading } = useApi<WalletData[]>(bizId ? `/api/wallets?businessId=${bizId}&k=${walletKey}` : '')
   105→  const [selectedWalletId, setSelectedWalletId] = useState<string | null>(null)
   106→  const [txns, setTxns] = useState<WalletTransaction[]>([])
   107→  const [txLoading, setTxLoading] = useState(false)
   108→
   109→  // Dialogs
   110→  const [createOpen, setCreateOpen] = useState(false)
   111→  const [depositOpen, setDepositOpen] = useState(false)
   112→  const [withdrawOpen, setWithdrawOpen] = useState(false)
   113→  const [convertOpen, setConvertOpen] = useState(false)
   114→  const [cryptoOpen, setCryptoOpen] = useState(false)
   115→
   116→  // History dialogs
   117→  const [historyTab, setHistoryTab] = useState('transactions')
   118→  const [deposits, setDeposits] = useState<DepositRecord[]>([])
   119→  const [withdrawals, setWithdrawals] = useState<WithdrawalRecord[]>([])
   120→  const [cryptoWdrs, setCryptoWdrs] = useState<CryptoWithdrawalRecord[]>([])
   121→  const [historyOpen, setHistoryOpen] = useState(false)
   122→
   123→  const [toastMsg, setToastMsg] = useState('')
   124→  const [toastVis, setToastVis] = useState(false)
   125→
   126→  // Create wallet form
   127→  const [newCurrency, setNewCurrency] = useState('USD')
   128→  const [creating, setCreating] = useState(false)
   129→
   130→  // Deposit form
   131→  const [depAmount, setDepAmount] = useState('')
   132→  const [depMethod, setDepMethod] = useState('bank_transfer')
   133→  const [depNotes, setDepNotes] = useState('')
   134→  const [depSubmitting, setDepSubmitting] = useState(false)
   135→
   136→  // Withdraw form
   137→  const [wdrAmount, setWdrAmount] = useState('')
   138→  const [wdrMethod, setWdrMethod] = useState('bank_transfer')
   139→  const [wdrBankName, setWdrBankName] = useState('')
   140→  const [wdrBankAccount, setWdrBankAccount] = useState('')
   141→  const [wdrRecipientName, setWdrRecipientName] = useState('')
   142→  const [wdrNotes, setWdrNotes] = useState('')
   143→  const [wdrSubmitting, setWdrSubmitting] = useState(false)
   144→
   145→  // Convert form
   146→  const [cvtFromWalletId, setCvtFromWalletId] = useState('')
   147→  const [cvtToWalletId, setCvtToWalletId] = useState('')
   148→  const [cvtAmount, setCvtAmount] = useState('')
   149→  const [cvtSubmitting, setCvtSubmitting] = useState(false)
   150→
   151→  // Crypto form
   152→  const [crCrypto, setCrCrypto] = useState('USDT')
   153→  const [crNetwork, setCrNetwork] = useState('trc20')
   154→  const [crAddress, setCrAddress] = useState('')
   155→  const [crAmount, setCrAmount] = useState('')
   156→  const [crNotes, setCrNotes] = useState('')
   157→  const [crSubmitting, setCrSubmitting] = useState(false)
   158→
   159→  const showToast = useCallback((msg: string) => { setToastMsg(msg); setToastVis(true); setTimeout(() => setToastVis(false), 3000) }, [])
   160→
   161→  const loadTransactions = useCallback(async (walletId: string) => {
   162→    setSelectedWalletId(walletId)
   163→    setTxLoading(true)
   164→    try {
   165→      const res = await fetch(`/api/wallets/${walletId}/transactions?limit=50`)
   166→      const json = await res.json()
   167→      setTxns(json.data || [])
   168→    } catch { setTxns([]) } finally { setTxLoading(false) }
   169→  }, [])
   170→
   171→  const loadHistory = useCallback(async (walletId: string, tab: string) => {
   172→    setHistoryTab(tab)
   173→    setHistoryOpen(true)
   174→    try {
   175→      if (tab === 'deposits') {
   176→        const r = await fetch(`/api/wallets/deposit?walletId=${walletId}&limit=50`)
   177→        setDeposits((await r.json()).data || [])
   178→      } else if (tab === 'withdrawals') {
   179→        const r = await fetch(`/api/wallets/withdrawal?walletId=${walletId}&limit=50`)
   180→        setWithdrawals((await r.json()).data || [])
   181→      } else if (tab === 'crypto') {
   182→        const r = await fetch(`/api/wallets/crypto-withdrawal?walletId=${walletId}&limit=50`)
   183→        setCryptoWdrs((await r.json()).data || [])
   184→      }
   185→    } catch { /* ignore */ }
   186→  }, [])
   187→
   188→  const allWallets = Array.isArray(wallets) ? wallets : []
   189→  const totalPortfolio = allWallets.reduce((sum, w) => {
   190→    try {
   191→      const toUsd = rates?.fiatToUsd?.[w.currency] || 1
   192→      return sum + w.balance * toUsd
   193→    } catch { return sum }
   194→  }, 0)
   195→  const totalBalance = allWallets.reduce((s, w) => s + w.balance, 0)
   196→  const totalPending = allWallets.reduce((s, w) => s + w.pendingBalance, 0)
   197→  const totalFrozen = allWallets.reduce((s, w) => s + w.frozenBalance, 0)
   198→
   199→  const selectedWallet = allWallets.find(w => w.id === selectedWalletId)
   200→
   201→  // Conversion preview
   202→  const conversionPreview = (() => {
   203→    if (!cvtAmount || !cvtFromWalletId || !cvtToWalletId || !rates) return null
   204→    const from = allWallets.find(w => w.id === cvtFromWalletId)
   205→    const to = allWallets.find(w => w.id === cvtToWalletId)
   206→    if (!from || !to) return null
   207→    const amount = parseFloat(cvtAmount)
   208→    if (isNaN(amount) || amount <= 0) return null
   209→    const direct = rates.fiatRates[from.currency]?.[to.currency]
   210→    if (!direct) return null
   211→    const gross = amount * direct
   212→    const fee = Math.round(gross * 0.005 * 100) / 100
   213→    const net = Math.round((gross - fee) * 100) / 100
   214→    return { rate: direct, gross, fee, net }
   215→  })()
   216→
   217→  // Crypto withdrawal preview
   218→  const cryptoPreview = (() => {
   219→    if (!crAmount || !selectedWallet || !rates) return null
   220→    const amount = parseFloat(crAmount)
   221→    if (isNaN(amount) || amount <= 0) return null
   222→    const toUsd = rates.fiatToUsd[selectedWallet.currency] || 1
   223→    const usdAmount = amount * toUsd
   224→    const cryptoPrice = rates.cryptoPrices[crCrypto]
   225→    if (!cryptoPrice) return null
   226→    const cryptoAmt = usdAmount / cryptoPrice
   227→    const netFee = rates.networkFees[crNetwork] || 0
   228→    const netCrypto = Math.max(0, cryptoAmt - netFee)
   229→    const procFee = Math.max(amount * (rates.cryptoWithdrawalFeePercent / 100), rates.cryptoWithdrawalMinFee)
   230→    const totalDebit = amount + procFee
   231→    return { usdAmount, cryptoAmt, netCrypto, netFee, procFee, totalDebit, cryptoPrice }
   232→  })()
   233→
   234→  const handleCreateWallet = async () => {
   235→    if (!bizId) return
   236→    setCreating(true)
   237→    try {
   238→      const res = await fetch('/api/wallets', {
   239→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   240→        body: JSON.stringify({ businessId: bizId, currency: newCurrency, isDefault: allWallets.length === 0 }),
   241→      })
   242→      const json = await res.json()
   243→      if (!res.ok) throw new Error(json.error || 'Failed')
   244→      showToast('Wallet created!')
   245→      setCreateOpen(false)
   246→      setWalletKey(k => k + 1)
   247→    } catch (e: any) { showToast(e.message) } finally { setCreating(false) }
   248→  }
   249→
   250→  const handleDeposit = async () => {
   251→    if (!selectedWalletId || !depAmount) return
   252→    setDepSubmitting(true)
   253→    try {
   254→      const res = await fetch('/api/wallets/deposit', {
   255→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   256→        body: JSON.stringify({
   257→          walletId: selectedWalletId, amount: parseFloat(depAmount),
   258→          paymentMethod: depMethod, provider: 'demo', notes: depNotes || undefined,
   259→        }),
   260→      })
   261→      const json = await res.json()
   262→      if (!res.ok) throw new Error(json.error || 'Failed')
   263→      showToast('Deposit completed!')
   264→      setDepositOpen(false); setDepAmount(''); setDepNotes('')
   265→      setWalletKey(k => k + 1)
   266→      loadTransactions(selectedWalletId)
   267→    } catch (e: any) { showToast(e.message) } finally { setDepSubmitting(false) }
   268→  }
   269→
   270→  const handleWithdraw = async () => {
   271→    if (!selectedWalletId || !wdrAmount) return
   272→    setWdrSubmitting(true)
   273→    try {
   274→      const res = await fetch('/api/wallets/withdrawal', {
   275→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   276→        body: JSON.stringify({
   277→          walletId: selectedWalletId, amount: parseFloat(wdrAmount),
   278→          paymentMethod: wdrMethod, provider: 'demo',
   279→          bankName: wdrBankName || undefined, bankAccount: wdrBankAccount || undefined,
   280→          recipientName: wdrRecipientName || undefined, notes: wdrNotes || undefined,
   281→        }),
   282→      })
   283→      const json = await res.json()
   284→      if (!res.ok) throw new Error(json.error || 'Failed')
   285→      showToast('Withdrawal completed!')
   286→      setWithdrawOpen(false); setWdrAmount(''); setWdrBankName(''); setWdrBankAccount(''); setWdrRecipientName(''); setWdrNotes('')
   287→      setWalletKey(k => k + 1)
   288→      loadTransactions(selectedWalletId)
   289→    } catch (e: any) { showToast(e.message) } finally { setWdrSubmitting(false) }
   290→  }
   291→
   292→  const handleConvert = async () => {
   293→    if (!cvtFromWalletId || !cvtToWalletId || !cvtAmount) return
   294→    setCvtSubmitting(true)
   295→    try {
   296→      const res = await fetch('/api/wallets/convert', {
   297→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   298→        body: JSON.stringify({ fromWalletId: cvtFromWalletId, toWalletId: cvtToWalletId, fromAmount: parseFloat(cvtAmount) }),
   299→      })
   300→      const json = await res.json()
   301→      if (!res.ok) throw new Error(json.error || 'Failed')
   302→      showToast('Conversion completed!')
   303→      setConvertOpen(false); setCvtAmount('')
   304→      setWalletKey(k => k + 1)
   305→      if (selectedWalletId) loadTransactions(selectedWalletId)
   306→    } catch (e: any) { showToast(e.message) } finally { setCvtSubmitting(false) }
   307→  }
   308→
   309→  const handleCryptoWithdraw = async () => {
   310→    if (!selectedWalletId || !crAmount || !crAddress) return
   311→    setCrSubmitting(true)
   312→    try {
   313→      const res = await fetch('/api/wallets/crypto-withdrawal', {
   314→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   315→        body: JSON.stringify({
   316→          walletId: selectedWalletId, amount: parseFloat(crAmount),
   317→          cryptoCurrency: crCrypto, network: crNetwork, walletAddress: crAddress.trim(),
   318→          notes: crNotes || undefined,
   319→        }),
   320→      })
   321→      const json = await res.json()
   322→      if (!res.ok) throw new Error(json.error || 'Failed')
   323→      showToast('Crypto withdrawal submitted!')
   324→      setCryptoOpen(false); setCrAmount(''); setCrAddress(''); setCrNotes('')
   325→      setWalletKey(k => k + 1)
   326→      loadTransactions(selectedWalletId)
   327→    } catch (e: any) { showToast(e.message) } finally { setCrSubmitting(false) }
   328→  }
   329→
   330→  // When crypto changes, reset network to first valid
   331→  useEffect(() => {
   332→    if (rates) {
   333→      const nets = rates.cryptoNetworks[crCrypto]
   334→      if (nets && nets.length > 0) setCrNetwork(nets[0])
   335→    }
   336→  }, [crCrypto, rates])
   337→
   338→  // When convert source wallet changes, clear target if same
   339→  useEffect(() => {
   340→    if (cvtFromWalletId === cvtToWalletId) setCvtToWalletId('')
   341→  }, [cvtFromWalletId])
   342→
   343→  if (bLoading || (bizId && wLoading)) return <LoadingSkeleton />
   344→
   345→  return (
   346→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="space-y-6">
   347→      {/* Business selector */}
   348→      <div className="flex items-center justify-between flex-wrap gap-3">
   349→        <div className="flex items-center gap-3">
   350→          <span className="text-sm text-slate-500">Business:</span>
   351→          <Select value={selectedBizId || businesses?.[0]?.id} onValueChange={setSelectedBizId}>
   352→            <SelectTrigger className="w-64"><SelectValue placeholder="Select business" /></SelectTrigger>
   353→            <SelectContent>{safeBusinesses.map(b => (<SelectItem key={b.id} value={b.id}>{b.name}</SelectItem>))}</SelectContent>
   354→          </Select>
   355→        </div>
   356→        <Button size="sm" onClick={() => setCreateOpen(true)}><Plus className="h-4 w-4 mr-1" /> New Wallet</Button>
   357→      </div>
   358→
   359→      {/* KPIs */}
   360→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   361→        <Card className="border-emerald-200 bg-gradient-to-br from-emerald-50/50 to-white">
   362→          <CardContent className="p-4 sm:p-6 text-center">
   363→            <p className="text-xs text-slate-500 mb-1">Portfolio (USD)</p>
   364→            <p className="text-2xl font-bold text-emerald-700">{formatCurrency(totalPortfolio)}</p>
   365→          </CardContent>
   366→        </Card>
   367→        <KPICard title="Total Balance" value={formatCurrency(totalBalance)} icon={WalletIcon} />
   368→        <KPICard title="Pending" value={formatCurrency(totalPending)} icon={RefreshCw} />
   369→        <KPICard title="Frozen" value={formatCurrency(totalFrozen)} icon={WalletIcon} />
   370→      </div>
   371→
   372→      {/* Wallet Cards */}
   373→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-4">
   374→        {allWallets.map(w => (
   375→          <Card key={w.id} className={`hover:shadow-md transition-shadow cursor-pointer ${selectedWalletId === w.id ? 'ring-2 ring-emerald-500' : ''}`} onClick={() => loadTransactions(w.id)}>
   376→            <CardContent className="p-4 sm:p-6">
   377→              <div className="flex items-center justify-between mb-3">
   378→                <div className="flex items-center gap-2">
   379→                  <span className="text-xl">{CURRENCY_FLAGS[w.currency] || '💰'}</span>
   380→                  <span className="font-semibold">{w.currency}</span>
   381→                  {w.isDefault && <Badge variant="outline" className="text-[10px]">Default</Badge>}
   382→                </div>
   383→                <Badge variant={w.status === 'active' ? 'default' : 'destructive'}>{w.status}</Badge>
   384→              </div>
   385→              <p className="text-2xl sm:text-3xl font-bold">{formatCurrency(w.balance, w.currency)}</p>
   386→              <div className="mt-3 space-y-1 text-xs text-slate-500">
   387→                <div className="flex justify-between"><span>Available</span><span className="text-emerald-600 font-medium">{formatCurrency(w.availableBalance, w.currency)}</span></div>
   388→                <div className="flex justify-between"><span>Pending</span><span className="text-amber-600 font-medium">{formatCurrency(w.pendingBalance, w.currency)}</span></div>
   389→                <div className="flex justify-between"><span>Frozen</span><span className="text-red-600 font-medium">{formatCurrency(w.frozenBalance, w.currency)}</span></div>
   390→              </div>
   391→            </CardContent>
   392→          </Card>
   393→        ))}
   394→      </div>
   395→
   396→      {allWallets.length === 0 && bizId && !wLoading && (
   397→        <Card className="border-dashed"><CardContent className="p-12 text-center"><p className="text-slate-500">No wallets for this business.</p></CardContent></Card>
   398→      )}
   399→
   400→      {/* Action Buttons + Transaction History */}
   401→      <AnimatePresence>
   402→        {selectedWalletId && selectedWallet && (
   403→          <motion.div initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: 'auto' }} exit={{ opacity: 0, height: 0 }}>
   404→            <div className="flex items-center justify-between mb-3 flex-wrap gap-2">
   405→              <h3 className="font-semibold">{selectedWallet.currency} Wallet — {CURRENCY_FLAGS[selectedWallet.currency] || ''}</h3>
   406→              <div className="flex gap-2 flex-wrap">
   407→                <Button size="sm" onClick={() => setDepositOpen(true)}><Download className="h-3.5 w-3.5 mr-1 text-emerald-500" /> Deposit</Button>
   408→                <Button size="sm" variant="outline" onClick={() => setWithdrawOpen(true)}><Upload className="h-3.5 w-3.5 mr-1 text-red-500" /> Withdraw</Button>
   409→                <Button size="sm" variant="outline" onClick={() => setConvertOpen(true)}><ArrowLeftRight className="h-3.5 w-3.5 mr-1 text-amber-500" /> Convert</Button>
   410→                <Button size="sm" variant="outline" onClick={() => setCryptoOpen(true)}><Bitcoin className="h-3.5 w-3.5 mr-1 text-orange-500" /> Crypto</Button>
   411→                <Button size="sm" variant="ghost" onClick={() => loadHistory(selectedWalletId, 'transactions')}>History</Button>
   412→              </div>
   413→            </div>
   414→            <Card>
   415→              <CardContent className="p-0">
   416→                <div className="overflow-x-auto">
   417→                  <Table>
   418→                    <TableHeader>
   419→                      <TableRow>
   420→                        <TableHead>Reference</TableHead>
   421→                        <TableHead>Type</TableHead>
   422→                        <TableHead>Description</TableHead>
   423→                        <TableHead className="text-right">Amount</TableHead>
   424→                        <TableHead className="text-right">Balance After</TableHead>
   425→                        <TableHead>Status</TableHead>
   426→                        <TableHead>Date</TableHead>
   427→                      </TableRow>
