    94→    94→function formatCurrency(n: number): string {
    95→    95→  if (n >= 1e9) return `$${(n / 1e9).toFixed(1)}B`
    96→    96→  if (n >= 1e6) return `$${(n / 1e6).toFixed(1)}M`
    97→    97→  if (n >= 1e3) return `$${(n / 1e3).toFixed(1)}K`
    98→    98→  return currencyFmt.format(n)
    99→    99→}
   100→   100→
   101→   101→function formatNumber(n: number): string {

   104→   104→  return n.toString()
   105→   105→}
