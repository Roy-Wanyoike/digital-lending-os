   414→   414→function useApi<T>(url: string) {
   415→   415→  const [data, setData] = useState<T | null>(null)
   416→   416→  const [loading, setLoading] = useState(true)
   417→   417→
   418→   418→  useEffect(() => {
   419→   419→    if (!url) return
   420→   420→    let cancelled = false
   421→   421→    fetch(url)
   422→   422→      .then(r => r.json())
   423→   423→      .then(d => { if (!cancelled) { setData(d); setLoading(false) } })
   424→   424→      .catch(() => { if (!cancelled) setLoading(false) })
   425→   425→    return () => { cancelled = true }
   426→   426→  }, [url])
   427→   427→
   428→   428→  return { data, loading }
   429→   429→}
