   114	export async function PUT(
   115	  request: NextRequest,
   116	  { params }: { params: Promise<{ id: string }> }
   117	) {
   118	  try {
   119	    const user = await getApiUser(request)
   120	    const { id } = await params
   121	    const body = await request.json()
   122	
   123	    const existing = await db.invoice.findUnique({
   124	      where: { id },
   125	      include: {
   126	        sender: { select: { tenantId: true } },
   127	        receiver: { select: { tenantId: true } },
   128	      },
   129	    })
   130	    if (!existing) {
   131	      return NextResponse.json({ error: 'Invoice not found' }, { status: 404 })
   132	    }
   133	    if (existing.sender?.tenantId !== user.tenantId && existing.receiver?.tenantId !== user.tenantId) {
   134	      return NextResponse.json({ error: 'Invoice not found' }, { status: 404 })
   135	    }
   136	
   137	    const allowedFields = [
   138	      'status',
   139	      'notes',
   140	      'paidAmount',
   141	      'dueDate',
   142	      'items',
   143	    ] as const
   144	
   145	    const updateData: Record<string, unknown> = {}
   146	    for (const field of allowedFields) {
   147	      if (body[field] !== undefined) {
   148	        if (field === 'dueDate' && typeof body[field] === 'string') {
   149	          updateData[field] = new Date(body[field])
   150	        } else if (field === 'items' && typeof body[field] !== 'string') {
   151	          updateData[field] = JSON.stringify(body[field])
   152	        } else {
   153	          updateData[field] = body[field]
   154	        }
   155	      }
   156	    }
   157	
   158	    // Auto-set paidAt when status changes to paid
   159	    if (updateData.status === 'paid' && existing.status !== 'paid') {
   160	      updateData.paidAt = new Date()
   161	      updateData.paidAmount = existing.amount
   162	    }
   163	
   164	    const invoice = await db.invoice.update({
   165	      where: { id },
   166	      data: updateData,
   167	      include: {
   168	        sender: {
   169	          select: { id: true, name: true },
   170	        },
   171	        receiver: {
   172	          select: { id: true, name: true },
   173	        },
   174	      },
   175	    })
   176	
   177	    return NextResponse.json({ data: invoice })
   178	  } catch (error) {
   179	    console.error('Error updating invoice:', error)
   180	    if (error instanceof AuthError) return NextResponse.json({ error: error.message }, { status: error.status })
   181	    return NextResponse.json({ error: 'Failed to update invoice' }, { status: 500 })
   182	  }

   188	// GET /api/deposits — List deposits for the tenant's wallets
   189	export async function GET(request: NextRequest) {
   190	  try {
   191	    const user = await getApiUser(request);
   192	    if (!user) {
   193	      return NextResponse.json({ error: 'Authentication required' }, { status: 401 });
   194	    }
   195	
   196	    const { searchParams } = new URL(request.url);
   197	    const page = parseInt(searchParams.get('page') || '1');
   198	    const limit = parseInt(searchParams.get('limit') || '20');
   199	    const status = searchParams.get('status');
   200	    const walletId = searchParams.get('walletId');
   201	
   202	    // Find wallet IDs belonging to businesses in this tenant
   203	    const tenantBusinessIds = (await db.business.findMany({
   204	      where: { tenantId: user.tenantId },
   205	      select: { id: true },
   206	    })).map(b => b.id);
   207	
   208	    const tenantWalletIds = (await db.wallet.findMany({
   209	      where: { businessId: { in: tenantBusinessIds } },
   210	      select: { id: true },
   211	    })).map(w => w.id);
   212	
   213	    // Build where clause — wallet-scoped
   214	    const where: any = {
   215	      walletId: { in: tenantWalletIds },
   216	    };
   217	    if (walletId) where.walletId = walletId;
   218	    if (status) where.status = status;
   219	
   220	    const [deposits, total] = await Promise.all([
   221	      db.deposit.findMany({
   222	        where,
   223	        include: {
   224	          wallet: { select: { id: true, currency: true, businessId: true } },
   225	        },
   226	        orderBy: { createdAt: 'desc' },
   227	        skip: (page - 1) * limit,
   228	        take: limit,
   229	      }),
   230	      db.deposit.count({ where }),
   231	    ]);
   232	
   233	    return NextResponse.json({
   234	      deposits,
   235	      pagination: { page, limit, total, pages: Math.ceil(total / limit) },
   236	    });
   237	  } catch (error) {
   238	    if (error instanceof AuthError) {
   239	      return NextResponse.json({ error: error.message }, { status: error.status });
   240	    }
   241	    console.error('Deposits GET error:', error);
   242	    return NextResponse.json({ error: 'Failed to fetch deposits' }, { status: 500 });
   243	  }
   244	}

   600	    const user = await getApiUser(request)
   601	    const { searchParams } = new URL(request.url)
   602	    const businessId = searchParams.get('businessId') || ''
   603	    const sortBy = searchParams.get('sortBy') || 'overallScore'
   604	    const sortOrder = searchParams.get('sortOrder') || 'desc'
   605	
   606	    const where: Record<string, unknown> = { business: { tenantId: user.tenantId } }
   607	    if (businessId) {
   608	      where.businessId = businessId
   609	    }
   610	
   611	    const orderBy: Record<string, string> = {}
   612	    if (['overallScore', 'paymentScore', 'deliveryScore', 'qualityScore', 'communicationScore', 'complianceScore'].includes(sortBy)) {
   613	      orderBy[sortBy] = sortOrder === 'asc' ? 'asc' : 'desc'
   614	    } else {
   615	      orderBy.overallScore = 'desc'
   616	    }
   617	
   618	    const scores = await db.trustScore.findMany({
   619	      where,
   620	      orderBy,
   621	      include: {
   622	        business: {
   623	          select: { id: true, name: true, country: true, industry: true, status: true },
   624	        },
   625	      },
   626	    })
   627	
   628	    return NextResponse.json({ data: scores })
   629	  } catch (error) {
   630	    console.error('Error listing trust scores:', error)
   631	    if (error instanceof AuthError) return NextResponse.json({ error: error.message }, { status: error.status })
   632	    return NextResponse.json({ error: 'Failed to list trust scores' }, { status: 500 })
   633	  }
   634	}

   636	export async function POST(request: NextRequest) {
   637	  try {
   638	    const user = await getApiUser(request)
   639	    const body = await request.json()
   640	    const parsed = recalculateSchema.safeParse(body)
   641	
   642	    if (!parsed.success) {
   643	      return NextResponse.json(
   644	        { error: parsed.error.issues.map((i) => i.message).join(', ') },
   645	        { status: 400 }
   646	      )
   647	    }
   648	
   649	    const { businessId } = parsed.data
   650	
   651	    // Verify business exists and belongs to tenant
   652	    const business = await db.business.findUnique({ where: { id: businessId } })
   653	    if (!business || business.tenantId !== user.tenantId) {
   654	      return NextResponse.json({ error: 'Business not found' }, { status: 404 })
   655	    }
   656	
   657	    // Find or create the trust score
   658	    let trustScore = await db.trustScore.findUnique({
   659	      where: { businessId },
   660	    })
   661	
   662	    if (!trustScore) {
   663	      trustScore = await db.trustScore.create({
   664	        data: { businessId },
   665	      })
   666	    }
   667	
   668	    // Fetch all reviews for this business
   669	    const reviews = await db.review.findMany({
   670	      where: { toBusinessId: businessId, status: 'published' },
   671	    })
   672	
   673	    // Fetch all reputation events
   674	    const events = await db.reputationEvent.findMany({
   675	      where: { trustScoreId: trustScore.id },
   676	    })
   677	
   678	    // Calculate scores based on reviews and events
   679	    let paymentScore = 50.0
   680	    let deliveryScore = 50.0
   681	    let qualityScore = 50.0
   682	    let communicationScore = 50.0
   683	
   684	    if (reviews.length > 0) {
   685	      const ratingsWithPayment = reviews.filter((r) => r.paymentRating !== null)
   686	      const ratingsWithDelivery = reviews.filter((r) => r.deliveryRating !== null)
   687	      const ratingsWithQuality = reviews.filter((r) => r.qualityRating !== null)
   688	      const ratingsWithComm = reviews.filter((r) => r.communicationRating !== null)
   689	
   690	      if (ratingsWithPayment.length > 0) {
   691	        const sum = ratingsWithPayment.reduce((acc, r) => acc + (r.paymentRating || 0), 0)
   692	        paymentScore = (sum / ratingsWithPayment.length) * 20
   693	      }
   694	
   695	      if (ratingsWithDelivery.length > 0) {
   696	        const sum = ratingsWithDelivery.reduce((acc, r) => acc + (r.deliveryRating || 0), 0)
   697	        deliveryScore = (sum / ratingsWithDelivery.length) * 20
   698	      }
   699	
   700	      if (ratingsWithQuality.length > 0) {
   701	        const sum = ratingsWithQuality.reduce((acc, r) => acc + (r.qualityRating || 0), 0)
   702	        qualityScore = (sum / ratingsWithQuality.length) * 20
   703	      }
   704	
   705	      if (ratingsWithComm.length > 0) {
   706	        const sum = ratingsWithComm.reduce((acc, r) => acc + (r.communicationRating || 0), 0)
   707	        communicationScore = (sum / ratingsWithComm.length) * 20
   708	      }
   709	
   710	      if (ratingsWithPayment.length === 0 && ratingsWithDelivery.length === 0) {
   711	        const avgRating = reviews.reduce((acc, r) => acc + r.rating, 0) / reviews.length
   712	        paymentScore = avgRating * 20
   713	        deliveryScore = avgRating * 20
   714	        qualityScore = avgRating * 20
   715	        communicationScore = avgRating * 20
   716	      }
   717	    }
   718	
   719	    let eventImpact = 0
   720	    for (const event of events) {
   721	      eventImpact += event.scoreImpact
   722	    }
   723	
   724	    const clamp = (v: number) => Math.max(0, Math.min(100, v))
   725	    paymentScore = clamp(paymentScore + eventImpact)
   726	    deliveryScore = clamp(deliveryScore + eventImpact)
   727	    qualityScore = clamp(qualityScore + eventImpact)
   728	    communicationScore = clamp(communicationScore + eventImpact)
   729	
   730	    const verifications = await db.verification.findMany({
   731	      where: { businessId },
   732	    })
   733	    const approvedVerifications = verifications.filter((v) => v.status === 'approved').length
   734	    const complianceScore = clamp(
   735	      verifications.length > 0 ? (approvedVerifications / verifications.length) * 100 : 50
   736	    )
   737	
   738	    const overallScore = clamp(
   739	      paymentScore * 0.25 +
   740	      deliveryScore * 0.2 +
   741	      qualityScore * 0.25 +
   742	      communicationScore * 0.15 +
   743	      complianceScore * 0.15
   744	    )
   745	
   746	    const updated = await db.trustScore.update({
   747	      where: { id: trustScore.id },
   748	      data: {
   749	        overallScore,
   750	        paymentScore,
   751	        deliveryScore,
   752	        qualityScore,
   753	        communicationScore,
   754	        complianceScore,
   755	        totalReviews: reviews.length,
   756	        totalTransactions: events.filter((e) => e.eventType === 'transaction_completed').length,
   757	        scoreVersion: { increment: 1 },
   758	        lastCalculated: new Date(),
   759	      },
   760	      include: {
   761	        business: {
   762	          select: { id: true, name: true, country: true, industry: true },
   763	        },
   764	      },
   765	    })
   766	
   767	    await db.reputationEvent.create({
   768	      data: {
   769	        trustScoreId: trustScore.id,
   770	        eventType: 'verification_upgraded',
   771	        scoreImpact: overallScore - trustScore.overallScore,
   772	        description: `Trust score recalculated (v${updated.scoreVersion})`,
   773	      },
   774	    })
   775	
   776	    return NextResponse.json({ data: updated })
   777	  } catch (error) {
   778	    console.error('Error recalculating trust score:', error)
   779	    if (error instanceof AuthError) return NextResponse.json({ error: error.message }, { status: error.status })
   780	    return NextResponse.json({ error: 'Failed to recalculate trust score' }, { status: 500 })
   781	  }
   782	}
