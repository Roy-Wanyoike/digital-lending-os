    64→function rand<T>(arr: T[]): T {
    65→  return arr[Math.floor(Math.random() * arr.length)]
    66→}

    68→function randFloat(min: number, max: number): number {
    69→  return Math.round((Math.random() * (max - min) + min) * 100) / 100
    70→}
    71→
    72→function randInt(min: number, max: number): number {
    73→  return Math.floor(Math.random() * (max - min + 1)) + min
    74→}

    76→function generateTxRef(): string {
    77→  const now = new Date()
    78→  const d = `${now.getFullYear()}${String(now.getMonth() + 1).padStart(2, '0')}${String(now.getDate()).padStart(2, '0')}`
    79→  return `ESC-${d}-${String(randInt(0, 99999)).padStart(5, '0')}`
    80→}

    82→function generateIntentRef(): string {
    83→  const now = new Date()
    84→  const d = `${now.getFullYear()}${String(now.getMonth() + 1).padStart(2, '0')}${String(now.getDate()).padStart(2, '0')}`
    85→  return `PAY-${d}-${String(randInt(0, 99999)).padStart(5, '0')}`
    86→}

    88→function generateLinkRef(): string {
    89→  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'
    90→  let result = 'PLINK-'
    91→  for (let i = 0; i < 6; i++) result += chars.charAt(Math.floor(Math.random() * chars.length))
    92→  return result
    93→}
