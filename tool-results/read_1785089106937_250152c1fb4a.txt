     1→'use client'
     2→
     3→import { useState, useCallback } from 'react'
     4→import { motion, AnimatePresence } from 'framer-motion'
     5→import { Shield, Plus, Eye, DollarSign, Play, ChevronRight, AlertTriangle, CheckCircle, Clock, X, ExternalLink, CreditCard } from 'lucide-react'
     6→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
     7→import { Badge } from '@/components/ui/badge'
     8→import { Button } from '@/components/ui/button'
     9→import { Input } from '@/components/ui/input'
    10→import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table'
    11→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
    12→import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter } from '@/components/ui/dialog'
    13→import { Label } from '@/components/ui/label'
    14→import { Separator } from '@/components/ui/separator'
    15→import { ScrollArea } from '@/components/ui/scroll-area'
    16→import {
    17→  useApi, formatCurrency, getStatusBadgeVariant, getStatusColor,
    18→  getRiskBg, getRiskColor, formatDate, ESCROW_STATUSES, PipelineCard,
    19→  LoadingSkeleton, KPICard, CURRENCY_FLAGS, ScoreBar, Toast,
    20→  type EscrowTransaction, type Business,
    21→} from '@/lib/dashboard-helpers'
    22→
    23→// ─── Types ────────────────────────────────────────────────────
    24→
    25→interface Milestone { id: string; sequence: number; title: string; amount: number; status: string; releasedAt?: string | null }
    26→interface Disbursement { id: string; amount: number; currency: string; status: string; paymentRef?: string | null }
    27→interface Dispute { id: string; raisedBy: string; reason: string; description?: string | null; status: string; resolution?: string | null; aiRecommendation?: string | null; createdAt: string }
    28→interface AuditEntry { id: string; action: string; actor?: string | null; details: string; createdAt: string }
    29→interface EscrowDetail extends EscrowTransaction {
    30→  milestones: Milestone[]
    31→  disbursements: Disbursement[]
    32→  disputes: Dispute[]
    33→  auditLog: AuditEntry[]
    34→}
    35→
    36→// ─── Component ────────────────────────────────────────────────
    37→
    38→export function EscrowTab() {
    39→  const [refreshKey, setRefreshKey] = useState(0)
    40→  const { data: transactions, loading } = useApi<EscrowTransaction[]>(`/api/escrow/transactions?limit=50&k=${refreshKey}`)
    41→  const { data: businesses } = useApi<Business[]>('/api/businesses?limit=50')
    42→  const [statusFilter, setStatusFilter] = useState('all')
    43→  const [selectedId, setSelectedId] = useState<string | null>(null)
    44→  const [createOpen, setCreateOpen] = useState(false)
    45→  const [disputeOpen, setDisputeOpen] = useState(false)
    46→  const [fundOpen, setFundOpen] = useState(false)
    47→  const [fundEscrowId, setFundEscrowId] = useState<string | null>(null)
    48→  const [toastMsg, setToastMsg] = useState('')
    49→  const [toastVis, setToastVis] = useState(false)
    50→
    51→  // Fund form state
    52→  const [fundEmail, setFundEmail] = useState('')
    53→  const [fundFirstName, setFundFirstName] = useState('')
    54→  const [fundLastName, setFundLastName] = useState('')
    55→  const [fundProvider, setFundProvider] = useState('')
    56→  const [fundProviders, setFundProviders] = useState<Array<{code: string; name: string}>>([])
    57→  const [funding, setFunding] = useState(false)
    58→
    59→  // Create form state
    60→  const [formBuyer, setFormBuyer] = useState('')
    61→  const [formSeller, setFormSeller] = useState('')
    62→  const [formAmount, setFormAmount] = useState('')
    63→  const [formCurrency, setFormCurrency] = useState('USD')
    64→  const [formDesc, setFormDesc] = useState('')
    65→  const [formMilestones, setFormMilestones] = useState('1')
    66→  const [creating, setCreating] = useState(false)
    67→
    68→  // Dispute form state
    69→  const [disputeRaisedBy, setDisputeRaisedBy] = useState('buyer')
    70→  const [disputeReason, setDisputeReason] = useState('')
    71→  const [disputeDesc, setDisputeDesc] = useState('')
    72→  const [disputing, setDisputing] = useState(false)
    73→
    74→  const showToast = useCallback((msg: string) => { setToastMsg(msg); setToastVis(true); setTimeout(() => setToastVis(false), 3000) }, [])
    75→
    76→  if (loading) return <LoadingSkeleton />
    77→
    78→  const allTxns = transactions || []
    79→  const filtered = statusFilter === 'all' ? allTxns : allTxns.filter(t => t.status?.toLowerCase() === statusFilter.toLowerCase())
    80→
    81→  const pipelineCounts = ESCROW_STATUSES.map(s => ({
    82→    status: s,
    83→    count: allTxns.filter(t => t.status?.toLowerCase() === s.toLowerCase().replace(/\s/g, '_') || t.status?.toLowerCase().replace(/\s/g, '') === s.toLowerCase().replace(/\s/g, '')).length,
    84→  }))
    85→  const pipelineColors = ['#94a3b8', '#3b82f6', '#f59e0b', '#10b981', '#ef4444']
    86→
    87→  const totalVolume = allTxns.reduce((s, t) => s + t.amount, 0)
    88→  const activeEscrows = allTxns.filter(t => ['funded', 'in_escrow'].includes(t.status?.toLowerCase())).length
    89→  const disputed = allTxns.filter(t => t.status?.toLowerCase() === 'disputed').length
    90→
    91→  // ─── Actions ─────────────────────────────────────────────
    92→  const doAction = useCallback(async (id: string, action: string, body?: Record<string, string>) => {
    93→    try {
    94→      const res = await fetch(`/api/escrow/transactions/${id}/${action}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: body ? JSON.stringify(body) : undefined })
    95→      const json = await res.json()
    96→      if (!res.ok) throw new Error(json.error || 'Action failed')
    97→      showToast(`${action} successful!`)
    98→      setRefreshKey(k => k + 1)
    99→      return json.data
   100→    } catch (e: any) { showToast(e.message); return null }
   101→  }, [showToast])
   102→
   103→  const openFundDialog = useCallback(async (escrowId: string, currency: string) => {
   104→    setFundEscrowId(escrowId)
   105→    setFundOpen(true)
   106→    try {
   107→      const res = await fetch(`/api/payments/providers?currency=${currency}`)
   108→      const json = await res.json()
   109→      const providers = (json.data || []).map((p: any) => ({ code: p.code, name: p.name }))
   110→      setFundProviders(providers)
   111→      if (providers.length > 0) setFundProvider(providers[0].code)
   112→    } catch { setFundProviders([]) }
   113→  }, [])
   114→
   115→  const handleFund = async () => {
   116→    if (!fundEscrowId || !fundEmail) return
   117→    setFunding(true)
   118→    try {
   119→      const body: Record<string, string> = { email: fundEmail }
   120→      if (fundFirstName) body.firstName = fundFirstName
   121→      if (fundLastName) body.lastName = fundLastName
   122→      if (fundProvider) body.provider = fundProvider
   123→      const res = await fetch(`/api/escrow/transactions/${fundEscrowId}/fund`, {
   124→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   125→        body: JSON.stringify(body),
   126→      })
   127→      const json = await res.json()
   128→      if (!res.ok) throw new Error(json.error || 'Funding failed')
   129→      const data = json.data
   130→      if (data?.checkoutUrl) {
   131→        window.open(data.checkoutUrl, '_blank', 'noopener,noreferrer')
   132→        showToast(`Redirecting to ${data.providerName}...`)
   133→      } else if (data?.status === 'awaiting_payment') {
   134→        showToast('Payment initiated! Complete it via the provider.')
   135→      } else {
   136→        showToast('Escrow funding initiated!')
   137→      }
   138→      setFundOpen(false)
   139→      setFundEmail(''); setFundFirstName(''); setFundLastName(''); setFundProvider('')
   140→      setRefreshKey(k => k + 1)
   141→    } catch (e: any) { showToast(e.message) } finally { setFunding(false) }
   142→  }
   143→
   144→  const handleCreate = async () => {
   145→    setCreating(true)
   146→    try {
   147→      const milestones = Array.from({ length: parseInt(formMilestones) }, (_, i) => ({
   148→        title: `Milestone ${i + 1}`,
   149→        amount: Math.round((parseFloat(formAmount) / parseInt(formMilestones)) * 100) / 100,
   150→      }))
   151→      const res = await fetch('/api/escrow/transactions', {
   152→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   153→        body: JSON.stringify({ buyerId: formBuyer, sellerId: formSeller, amount: parseFloat(formAmount), currency: formCurrency, description: formDesc || undefined, milestones }),
   154→      })
   155→      const json = await res.json()
   156→      if (!res.ok) throw new Error(json.error || json.details?.[0]?.message || 'Create failed')
   157→      showToast('Escrow created!')
   158→      setCreateOpen(false)
   159→      setFormBuyer(''); setFormSeller(''); setFormAmount(''); setFormDesc(''); setFormMilestones('1')
   160→      setRefreshKey(k => k + 1)
   161→    } catch (e: any) { showToast(e.message) } finally { setCreating(false) }
   162→  }
   163→
   164→  const handleDispute = async () => {
   165→    if (!selectedId || !disputeReason) return
   166→    setDisputing(true)
   167→    try {
   168→      const res = await fetch(`/api/escrow/transactions/${selectedId}/disputes`, {
   169→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   170→        body: JSON.stringify({ raisedBy: disputeRaisedBy, reason: disputeReason, description: disputeDesc || undefined }),
   171→      })
   172→      const json = await res.json()
   173→      if (!res.ok) throw new Error(json.error || 'Dispute failed')
   174→      showToast('Dispute raised!')
   175→      setDisputeOpen(false); setDisputeReason(''); setDisputeDesc('')
   176→      setRefreshKey(k => k + 1)
   177→    } catch (e: any) { showToast(e.message) } finally { setDisputing(false) }
   178→  }
   179→
   180→  const selectedTxn = allTxns.find(t => t.id === selectedId)
   181→
   182→  return (
   183→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="space-y-6">
   184→      {/* KPIs */}
   185→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   186→        <KPICard title="Total Volume" value={formatCurrency(totalVolume)} icon={DollarSign} />
   187→        <KPICard title="Active Escrows" value={activeEscrows.toString()} icon={Shield} />
   188→        <KPICard title="Disputed" value={disputed.toString()} icon={AlertTriangle} />
   189→        <KPICard title="Total Deals" value={allTxns.length.toString()} icon={ChevronRight} />
   190→      </div>
   191→
   192→      {/* Pipeline */}
   193→      <div className="flex gap-2 sm:gap-3 overflow-x-auto pb-2">
   194→        {pipelineCounts.map((p, i) => (
   195→          <PipelineCard key={p.status} label={p.status} count={p.count} color={pipelineColors[i]} />
   196→        ))}
   197→      </div>
   198→
   199→      {/* Toolbar */}
   200→      <div className="flex flex-wrap items-center justify-between gap-3">
   201→        <div className="flex items-center gap-3">
   202→          <span className="text-sm text-slate-500">Status:</span>
   203→          <Select value={statusFilter} onValueChange={setStatusFilter}>
   204→            <SelectTrigger className="w-40"><SelectValue /></SelectTrigger>
   205→            <SelectContent>
   206→              <SelectItem value="all">All Statuses</SelectItem>
   207→              {ESCROW_STATUSES.map(s => <SelectItem key={s} value={s.toLowerCase().replace(/\s/g, '_')}>{s}</SelectItem>)}
   208→            </SelectContent>
   209→          </Select>
   210→        </div>
   211→        <Button size="sm" onClick={() => setCreateOpen(true)}><Plus className="h-4 w-4 mr-1" /> New Escrow</Button>
   212→      </div>
   213→
   214→      {/* Table */}
   215→      <Card>
   216→        <CardContent className="p-0">
   217→          <div className="overflow-x-auto">
   218→            <Table>
   219→              <TableHeader>
   220→                <TableRow>
   221→                  <TableHead>Reference</TableHead>
   222→                  <TableHead>Buyer → Seller</TableHead>
   223→                  <TableHead className="text-right">Amount</TableHead>
   224→                  <TableHead>Status</TableHead>
   225→                  <TableHead>AI Risk</TableHead>
   226→                  <TableHead>Milestones</TableHead>
   227→                  <TableHead>Actions</TableHead>
   228→                  <TableHead>Date</TableHead>
   229→                </TableRow>
   230→              </TableHeader>
   231→              <TableBody>
   232→                {filtered.slice(0, 20).map(txn => {
   233→                  const st = txn.status?.toLowerCase() || ''
   234→                  const canFund = st === 'created'
   235→                  const canActivate = st === 'funded'
   236→                  const canRelease = st === 'in_escrow'
   237→                  const canDispute = ['funded', 'in_escrow'].includes(st)
   238→                  return (
   239→                    <TableRow key={txn.id} className="even:bg-muted/50">
   240→                      <TableCell className="font-mono text-xs">{txn.txRef}</TableCell>
   241→                      <TableCell className="max-w-[160px] truncate text-xs">
   242→                        <span className="text-slate-700">{txn.buyer?.name}</span>
   243→                        <span className="text-slate-400 mx-1">→</span>
   244→                        <span className="text-slate-700">{txn.seller?.name}</span>
   245→                      </TableCell>
   246→                      <TableCell className="text-right font-medium">{formatCurrency(txn.amount, txn.currency)}</TableCell>
   247→                      <TableCell><Badge variant={getStatusBadgeVariant(txn.status)} className={getStatusColor(txn.status)}>{txn.status}</Badge></TableCell>
   248→                      <TableCell>
   249→                        <div className="flex items-center gap-2">
   250→                          <div className="w-10 h-1.5 bg-slate-100 rounded-full overflow-hidden">
   251→                            <div className={`h-full rounded-full ${getRiskBg(txn.aiRiskScore || 0)}`} style={{ width: `${txn.aiRiskScore || 0}%` }} />
   252→                          </div>
   253→                          <span className={`text-xs font-medium ${getRiskColor(txn.aiRiskScore || 0)}`}>{txn.aiRiskLevel}</span>
   254→                        </div>
   255→                      </TableCell>
   256→                      <TableCell className="text-xs">{txn.currentMilestone}/{txn.totalMilestones}</TableCell>
   257→                      <TableCell>
   258→                        <div className="flex items-center gap-1">
   259→                          <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => setSelectedId(txn.id)} title="View details"><Eye className="h-3.5 w-3.5" /></Button>
   260→                          {canFund && <Button variant="outline" size="sm" className="h-7 text-xs" onClick={() => openFundDialog(txn.id, txn.currency)}><CreditCard className="h-3 w-3 mr-1" />Fund</Button>}
   261→                          {canActivate && <Button variant="outline" size="sm" className="h-7 text-xs" onClick={() => doAction(txn.id, 'activate')}><Play className="h-3 w-3 mr-1" />Activate</Button>}
   262→                          {canRelease && <Button variant="outline" size="sm" className="h-7 text-xs" onClick={() => {
   263→                            const ms = txn.milestones?.find((m: any) => m.status === 'ready' || m.status === 'pending')
   264→                            if (ms) doAction(txn.id, 'release', { milestoneId: ms.id })
   265→                            else showToast('No pending milestones')
   266→                          }}><CheckCircle className="h-3 w-3 mr-1" />Release</Button>}
   267→                          {canDispute && <Button variant="outline" size="sm" className="h-7 text-xs text-red-600" onClick={() => { setSelectedId(txn.id); setDisputeOpen(true) }}><AlertTriangle className="h-3 w-3 mr-1" />Dispute</Button>}
   268→                        </div>
   269→                      </TableCell>
   270→                      <TableCell className="text-xs text-slate-500 whitespace-nowrap">{formatDate(txn.createdAt)}</TableCell>
   271→                    </TableRow>
   272→                  )
   273→                })}
   274→              </TableBody>
   275→            </Table>
   276→          </div>
   277→        </CardContent>
   278→      </Card>
   279→
   280→      {/* Detail Drawer */}
   281→      <AnimatePresence>
   282→        {selectedTxn && (
   283→          <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} className="fixed inset-0 z-50 flex justify-end">
   284→            <div className="absolute inset-0 bg-black/30" onClick={() => setSelectedId(null)} />
   285→            <motion.div initial={{ x: '100%' }} animate={{ x: 0 }} exit={{ x: '100%' }} transition={{ type: 'spring', damping: 25, stiffness: 300 }} className="relative w-full max-w-xl bg-white shadow-xl h-full overflow-y-auto">
   286→              <div className="sticky top-0 z-10 bg-white border-b px-6 py-4 flex items-center justify-between">
   287→                <div>
   288→                  <h3 className="font-semibold">{selectedTxn.txRef}</h3>
   289→                  <p className="text-xs text-slate-500">{formatDate(selectedTxn.createdAt)}</p>
   290→                </div>
   291→                <Button variant="ghost" size="icon" onClick={() => setSelectedId(null)}><X className="h-4 w-4" /></Button>
   292→              </div>
   293→              <div className="p-6 space-y-6">
   294→                {/* Status & Amount */}
   295→                <div className="flex items-center justify-between">
   296→                  <Badge variant={getStatusBadgeVariant(selectedTxn.status)} className={getStatusColor(selectedTxn.status)}>{selectedTxn.status}</Badge>
   297→                  <p className="text-2xl font-bold">{formatCurrency(selectedTxn.amount, selectedTxn.currency)}</p>
   298→                </div>
   299→                <div className="grid grid-cols-2 gap-4 text-sm">
   300→                  <div><p className="text-slate-500">Buyer</p><p className="font-medium">{selectedTxn.buyer?.name}</p></div>
   301→                  <div><p className="text-slate-500">Seller</p><p className="font-medium">{selectedTxn.seller?.name}</p></div>
   302→                  <div><p className="text-slate-500">Funded</p><p className="font-medium text-blue-600">{formatCurrency(selectedTxn.fundedAmount, selectedTxn.currency)}</p></div>
   303→                  <div><p className="text-slate-500">Released</p><p className="font-medium text-emerald-600">{formatCurrency(selectedTxn.releasedAmount, selectedTxn.currency)}</p></div>
   304→                </div>
   305→
   306→                {/* Risk Score */}
   307→                <div>
   308→                  <p className="text-sm font-medium mb-2">AI Risk Assessment</p>
   309→                  <ScoreBar score={selectedTxn.aiRiskScore || 0} label="Risk Score" />
   310→                </div>
   311→
   312→                {/* Milestones */}
   313→                {selectedTxn.milestones && selectedTxn.milestones.length > 0 && (
   314→                  <div>
   315→                    <p className="text-sm font-medium mb-2">Milestones</p>
   316→                    <div className="space-y-2">
   317→                      {selectedTxn.milestones.map((m: any) => (
   318→                        <div key={m.id} className="flex items-center justify-between p-3 rounded-lg bg-slate-50">
   319→                          <div className="flex items-center gap-2">
   320→                            {m.status === 'released' ? <CheckCircle className="h-4 w-4 text-emerald-500" /> : <Clock className="h-4 w-4 text-slate-400" />}
   321→                            <span className="text-sm font-medium">{m.title}</span>
   322→                          </div>
   323→                          <span className="text-sm font-medium">{formatCurrency(m.amount, selectedTxn.currency)}</span>
   324→                        </div>
   325→                      ))}
   326→                    </div>
   327→                  </div>
   328→                )}
   329→
   330→                {/* Disputes */}
   331→                {selectedTxn.disputes && selectedTxn.disputes.length > 0 && (
   332→                  <div>
   333→                    <p className="text-sm font-medium mb-2">Disputes</p>
   334→                    {selectedTxn.disputes.map((d: any) => (
   335→                      <div key={d.id} className="p-3 rounded-lg bg-red-50 border border-red-100 mb-2">
   336→                        <div className="flex items-center justify-between mb-1">
   337→                          <Badge variant={d.status === 'open' ? 'destructive' : 'secondary'}>{d.status}</Badge>
   338→                          <span className="text-xs text-slate-500">{formatDate(d.createdAt)}</span>
   339→                        </div>
   340→                        <p className="text-sm font-medium">{d.reason}</p>
   341→                        {d.aiRecommendation && <p className="text-xs text-slate-500 mt-1">AI: {d.aiRecommendation}</p>}
   342→                        {d.resolution && <p className="text-xs text-emerald-600 mt-1">Resolution: {d.resolution}</p>}
   343→                      </div>
   344→                    ))}
   345→                  </div>
   346→                )}
   347→
   348→                {/* Description */}
   349→                {selectedTxn.description && (
   350→                  <div>
   351→                    <p className="text-sm font-medium mb-1">Description</p>
   352→                    <p className="text-sm text-slate-600">{selectedTxn.description}</p>
   353→                  </div>
   354→                )}
   355→
   356→                {/* Fees */}
   357→                <div className="grid grid-cols-2 gap-4 text-sm">
   358→                  <div><p className="text-slate-500">Fee</p><p>{formatCurrency(selectedTxn.feeAmount, selectedTxn.feeCurrency)}</p></div>
   359→                  <div><p className="text-slate-500">Refunded</p><p>{formatCurrency(selectedTxn.refundedAmount, selectedTxn.currency)}</p></div>
   360→                </div>
   361→              </div>
   362→            </motion.div>
   363→          </motion.div>
   364→        )}
   365→      </AnimatePresence>
   366→
   367→      {/* Fund Dialog */}
   368→      <Dialog open={fundOpen} onOpenChange={setFundOpen}>
   369→        <DialogContent className="max-w-md">
   370→          <DialogHeader>
   371→            <DialogTitle className="flex items-center gap-2"><CreditCard className="h-5 w-5" /> Fund Escrow</DialogTitle>
   372→            <DialogDescription>Pay via a payment provider to fund this escrow. You will be redirected to complete payment.</DialogDescription>
   373→          </DialogHeader>
   374→          <div className="space-y-4">
   375→            {fundProviders.length > 0 && (
   376→              <div className="space-y-2"><Label>Payment Provider</Label>
   377→                <Select value={fundProvider} onValueChange={setFundProvider}>
   378→                  <SelectTrigger><SelectValue placeholder="Select provider" /></SelectTrigger>
   379→                  <SelectContent>
   380→                    {fundProviders.map(p => <SelectItem key={p.code} value={p.code}>{p.name}</SelectItem>)}
   381→                  </SelectContent>
   382→                </Select>
   383→                <p className="text-xs text-slate-400">Auto-selected based on escrow currency. Fees apply per provider.</p>
   384→              </div>
   385→            )}
   386→            {fundProviders.length === 0 && (
   387→              <div className="p-3 rounded-lg bg-amber-50 border border-amber-200 text-sm text-amber-700">
   388→                No payment providers configured. Payments will use demo mode. Add provider keys in .env to enable real payments.
   389→              </div>
   390→            )}
   391→            <div className="space-y-2"><Label>Email *</Label><Input type="email" placeholder="payer@email.com" value={fundEmail} onChange={e => setFundEmail(e.target.value)} /></div>
   392→            <div className="grid grid-cols-2 gap-3">
   393→              <div className="space-y-2"><Label>First Name</Label><Input placeholder="John" value={fundFirstName} onChange={e => setFundFirstName(e.target.value)} /></div>
   394→              <div className="space-y-2"><Label>Last Name</Label><Input placeholder="Doe" value={fundLastName} onChange={e => setFundLastName(e.target.value)} /></div>
   395→            </div>
   396→          </div>
   397→          <DialogFooter>
   398→            <Button variant="outline" onClick={() => setFundOpen(false)}>Cancel</Button>
