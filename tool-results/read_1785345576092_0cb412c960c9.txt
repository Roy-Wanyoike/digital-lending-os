     1→/**
     2→ * Youngsend Saga Orchestrator — Payment Workflow
     3→ *
     4→ * Implements the Saga pattern for distributed transactions using a state machine approach.
     5→ *
     6→ * PaymentSaga orchestrates the flow:
     7→ *   payment.initiated → wallet.debit → escrow.create → notification.send
     8→ *
     9→ * On failure at any step, compensating (undo) transactions are triggered:
    10→ *   - If escrow.create fails → wallet.credit (refund)
    11→ *   - If wallet.debit fails → payment.cancel
    12→ *   - If notification.send fails → non-critical, logged but saga continues
    13→ *
    14→ * State Machine:
    15→ *
    16→ *   IDLE ──▶ INITIATED ──▶ DEBITING_WALLET ──▶ CREATING_ESCROW ──▶ SENDING_NOTIFICATION ──▶ COMPLETED
    17→ *              │                │                       │                      │
    18→ *              ▼                ▼                       ▼                      ▼
    19→ *           FAILED         COMPENSATING_WALLET    COMPENSATING_ESCROW       FAILED (non-critical)
    20→ *                              │                       │
    21→ *                              ▼                       ▼
    22→ *                           FAILED                  COMPENSATING_WALLET
    23→ *                                                      │
    24→ *                                                      ▼
    25→ *                                                   FAILED
    26→ */
    27→
    28→import { v4 as uuidv4 } from "uuid";
    29→import type { YoungsendProducer } from "./producer";
    30→import { buildEvent, type BaseEvent } from "./producer";
    31→import type { YoungsendConsumer } from "./consumer";
    32→import type { MessageMetadata } from "./consumer";
    33→import {
    34→  PaymentInitiatedEventSchema,
    35→  PaymentCompletedEventSchema,
    36→  PaymentFailedEventSchema,
    37→  EscrowCreatedEventSchema,
    38→  EscrowCancelledEventSchema,
    39→  WalletDepositedEventSchema,
    40→  WalletWithdrawnEventSchema,
    41→} from "./event-schemas";
    42→import type {
    43→  PaymentInitiatedPayload,
    44→  PaymentCompletedPayload,
    45→  PaymentFailedPayload,
    46→  EscrowCreatedPayload,
    47→  EscrowCancelledPayload,
    48→} from "./event-schemas";
    49→
    50→// ─── Types ────────────────────────────────────────────────────────────────────
    51→
    52→export type SagaState =
    53→  | "IDLE"
    54→  | "INITIATED"
    55→  | "DEBITING_WALLET"
    56→  | "WALLET_DEBITED"
    57→  | "CREATING_ESCROW"
    58→  | "ESCROW_CREATED"
    59→  | "SENDING_NOTIFICATION"
    60→  | "COMPLETED"
    61→  | "COMPENSATING_WALLET"
    62→  | "COMPENSATING_ESCROW"
    63→  | "COMPENSATING_PAYMENT"
    64→  | "FAILED";
    65→
    66→export type SagaStep =
    67→  | "PAYMENT_INITIATE"
    68→  | "WALLET_DEBIT"
    69→  | "ESCROW_CREATE"
    70→  | "NOTIFICATION_SEND"
    71→  | "COMPENSATE_WALLET_CREDIT"
    72→  | "COMPENSATE_ESCROW_CANCEL"
    73→  | "COMPENSATE_PAYMENT_CANCEL";
    74→
    75→export interface SagaStepResult {
    76→  step: SagaStep;
    77→  status: "success" | "failed";
    78→  error?: string;
    79→  durationMs: number;
    80→  timestamp: string;
    81→}
    82→
    83→export interface PaymentSagaData {
    84→  /** Unique saga instance ID */
    85→  sagaId: string;
    86→  /** Correlation ID linking all events in this saga */
    87→  correlationId: string;
    88→  /** Current state in the state machine */
    89→  state: SagaState;
    90→  /** Current step being executed */
    91→  currentStep: SagaStep | null;
    92→  /** Payment data from the initiated event */
    93→  payment: {
    94→    paymentId: string;
    95→    userId: string;
    96→    payeeId?: string;
    97→    amount: number;
    98→    currency: string;
    99→    paymentMethod: string;
   100→    provider: string;
   101→    reference: string;
   102→  } | null;
   103→  /** Wallet operation data */
   104→  wallet: {
   105→    walletId: string;
   106→    debitId?: string;
   107→    creditId?: string;
   108→    amount: number;
   109→    currency: string;
   110→  } | null;
   111→  /** Escrow data */
   112→  escrow: {
   113→    escrowId: string;
   114→    transactionId: string;
   115→    amount: number;
   116→    currency: string;
   117→  } | null;
   118→  /** Notification data */
   119→  notification: {
   120→    notificationId?: string;
   121→    channels: string[];
   122→  } | null;
   123→  /** History of all steps executed in this saga */
   124→  stepHistory: SagaStepResult[];
   125→  /** Error that caused the saga to fail */
   126→  failureReason: string | null;
   127→  /** Timestamps */
   128→  createdAt: string;
   129→  updatedAt: string;
   130→  completedAt: string | null;
   131→}
   132→
   133→// ─── State Machine Transitions ────────────────────────────────────────────────
   134→
   135→/**
   136→ * Valid state transitions for the PaymentSaga.
   137→ * Key = current state, Value = array of allowed next states.
   138→ */
   139→const STATE_TRANSITIONS: Record<SagaState, SagaState[]> = {
   140→  IDLE: ["INITIATED"],
   141→  INITIATED: ["DEBITING_WALLET", "COMPENSATING_PAYMENT", "FAILED"],
   142→  DEBITING_WALLET: ["WALLET_DEBITED", "COMPENSATING_PAYMENT", "FAILED"],
   143→  WALLET_DEBITED: ["CREATING_ESCROW", "COMPENSATING_WALLET", "FAILED"],
   144→  CREATING_ESCROW: ["ESCROW_CREATED", "COMPENSATING_WALLET", "FAILED"],
   145→  ESCROW_CREATED: ["SENDING_NOTIFICATION", "COMPENSATING_ESCROW", "FAILED"],
   146→  SENDING_NOTIFICATION: ["COMPLETED", "FAILED"], // Notification failure is non-critical
   147→  COMPLETED: [], // Terminal state
   148→  COMPENSATING_WALLET: ["FAILED", "COMPENSATING_PAYMENT"],
   149→  COMPENSATING_ESCROW: ["COMPENSATING_WALLET", "FAILED"],
   150→  COMPENSATING_PAYMENT: ["FAILED"], // Terminal state
   151→  FAILED: [], // Terminal state
   152→};
   153→
   154→// ─── Saga Orchestrator Class ──────────────────────────────────────────────────
   155→
   156→export class PaymentSaga {
   157→  private producer: YoungsendProducer;
   158→  private consumer: YoungsendConsumer;
   159→  private sagaInstances: Map<string, PaymentSagaData>;
   160→  private serviceName: string;
   161→  private correlationToSaga: Map<string, string>; // correlationId → sagaId
   162→
   163→  constructor(options: {
   164→    producer: YoungsendProducer;
   165→    consumer: YoungsendConsumer;
   166→    serviceName?: string;
   167→  }) {
   168→    this.producer = options.producer;
   169→    this.consumer = options.consumer;
   170→    this.serviceName = options.serviceName ?? "saga-orchestrator";
   171→    this.sagaInstances = new Map();
   172→    this.correlationToSaga = new Map();
   173→  }
   174→
   175→  // ── Start Listening ────────────────────────────────────────────────────────
   176→
   177→  /**
   178→   * Start the saga orchestrator. Registers handlers on the consumer
   179→   * for all relevant payment, wallet, escrow, and notification events.
   180→   */
   181→  async start(): Promise<void> {
   182→    console.info("[PaymentSaga] Starting saga orchestrator...");
   183→
   184→    this.consumer.onMessage(async (event: BaseEvent, metadata: MessageMetadata) => {
   185→      await this.routeEvent(event, metadata);
   186→    });
   187→
   188→    await this.consumer.start();
   189→    console.info("[PaymentSaga] Saga orchestrator started. Listening for events.");
   190→  }
   191→
   192→  /**
   193→   * Stop the saga orchestrator gracefully.
   194→   */
   195→  async stop(): Promise<void> {
   196→    console.info("[PaymentSaga] Stopping saga orchestrator...");
   197→    await this.consumer.stop();
   198→    console.info("[PaymentSaga] Saga orchestrator stopped.");
   199→  }
   200→
   201→  // ── Event Routing ─────────────────────────────────────────────────────────
   202→
   203→  private async routeEvent(event: BaseEvent, metadata: MessageMetadata): Promise<void> {
   204→    const { eventType, correlationId } = event;
   205→
   206→    // Find the saga instance by correlation ID
   207→    const sagaId = this.correlationToSaga.get(correlationId);
   208→
   209→    switch (eventType) {
   210→      // ── Saga-triggering events ──
   211→      case "payment_initiated":
   212→        await this.onPaymentInitiated(event, metadata);
   213→        break;
   214→
   215→      // ── Saga-continuation events ──
   216→      case "payment_completed":
   217→      case "wallet_withdrawn":
   218→      case "escrow_created":
   219→      case "escrow_funded":
   220→      case "escrow_released":
   221→        if (sagaId) {
   222→          await this.onStepSuccess(event, metadata, sagaId);
   223→        }
   224→        break;
   225→
   226→      // ── Saga-failure events ──
   227→      case "payment_failed":
   228→        if (sagaId) {
   229→          await this.onStepFailed(event, metadata, sagaId);
   230→        }
   231→        break;
   232→
   233→      // ── Escrow dispute/cancel events ──
   234→      case "escrow_disputed":
   235→      case "escrow_cancelled":
   236→        if (sagaId) {
   237→          await this.onEscrowDispute(event, metadata, sagaId);
   238→        }
   239→        break;
   240→
   241→      default:
   242→        // Not a saga-relevant event; skip
   243→        break;
   244→    }
   245→  }
   246→
   247→  // ── Saga Lifecycle Handlers ─────────────────────────────────────────────────
   248→
   249→  /**
   250→   * Handle payment.initiated — start a new saga instance.
   251→   */
   252→  private async onPaymentInitiated(event: BaseEvent, metadata: MessageMetadata): Promise<void> {
   253→    // Validate the event payload
   254→    const parsed = PaymentInitiatedEventSchema.safeParse(event);
   255→    if (!parsed.success) {
   256→      console.error("[PaymentSaga] Invalid payment_initiated event:", parsed.error);
   257→      return;
   258→    }
   259→
   260→    const payload = parsed.data.payload;
   261→
   262→    // Check if saga already exists for this payment (idempotency)
   263→    const existingSaga = this.findSagaByPaymentId(payload.paymentId);
   264→    if (existingSaga && existingSaga.state !== "FAILED") {
   265→      console.info(
   266→        `[PaymentSaga] Saga already exists for paymentId=${payload.paymentId} in state=${existingSaga.state}`,
   267→      );
   268→      return;
   269→    }
   270→
   271→    // Create new saga instance
   272→    const correlationId = event.correlationId;
   273→    const sagaId = uuidv4();
   274→    const saga = this.createSagaInstance(sagaId, correlationId, payload);
   275→
   276→    this.sagaInstances.set(sagaId, saga);
   277→    this.correlationToSaga.set(correlationId, sagaId);
   278→
   279→    console.info(
   280→      `[PaymentSaga] New saga created: sagaId=${sagaId} paymentId=${payload.paymentId} correlationId=${correlationId}`,
   281→    );
   282→
   283→    // Transition: IDLE → INITIATED → DEBITING_WALLET
   284→    this.transition(sagaId, "INITIATED");
   285→    await this.executeWalletDebit(saga);
   286→  }
   287→
   288→  /**
   289→   * Handle successful completion of a saga step.
   290→   */
   291→  private async onStepSuccess(event: BaseEvent, metadata: MessageMetadata, sagaId: string): Promise<void> {
   292→    const saga = this.sagaInstances.get(sagaId);
   293→    if (!saga || saga.state === "COMPLETED" || saga.state === "FAILED") return;
   294→
   295→    switch (event.eventType) {
   296→      case "payment_completed":
   297→        // Payment is confirmed; move to escrow creation
   298→        await this.executeEscrowCreation(sagaId);
   299→        break;
   300→
   301→      case "wallet_withdrawn":
   302→        // Wallet debit succeeded; move to escrow creation
   303→        this.transition(sagaId, "WALLET_DEBITED");
   304→        await this.executeEscrowCreation(sagaId);
   305→        break;
   306→
   307→      case "escrow_created":
   308→        // Escrow created; move to notification
   309→        this.transition(sagaId, "ESCROW_CREATED");
   310→        await this.executeNotificationSend(sagaId);
   311→        break;
   312→
   313→      case "escrow_funded":
   314→        // Escrow funded — proceed to notification if not already done
   315→        if (saga.state === "ESCROW_CREATED" || saga.state === "CREATING_ESCROW") {
   316→          this.transition(sagaId, "ESCROW_CREATED");
   317→          await this.executeNotificationSend(sagaId);
   318→        }
   319→        break;
   320→
   321→      default:
   322→        break;
   323→    }
   324→  }
   325→
   326→  /**
   327→   * Handle failure of a saga step.
   328→   */
   329→  private async onStepFailed(event: BaseEvent, metadata: MessageMetadata, sagaId: string): Promise<void> {
   330→    const saga = this.sagaInstances.get(sagaId);
   331→    if (!saga || saga.state === "COMPLETED" || saga.state === "FAILED") return;
   332→
   333→    let parsed;
   334→    if (event.eventType === "payment_failed") {
   335→      parsed = PaymentFailedEventSchema.safeParse(event);
   336→    }
   337→
   338→    const errorMessage = parsed?.success
   339→      ? parsed.data.payload.errorMessage
   340→      : "Unknown error";
   341→
   342→    await this.compensate(sagaId, errorMessage);
   343→  }
   344→
   345→  /**
   346→   * Handle escrow dispute or cancellation — triggers compensation.
   347→   */
   348→  private async onEscrowDispute(event: BaseEvent, metadata: MessageMetadata, sagaId: string): Promise<void> {
   349→    const saga = this.sagaInstances.get(sagaId);
   350→    if (!saga || saga.state === "COMPLETED" || saga.state === "FAILED") return;
   351→
   352→    const reason = event.eventType === "escrow_disputed"
   353→      ? "Escrow disputed"
   354→      : "Escrow cancelled";
   355→
   356→    await this.compensate(sagaId, reason);
   357→  }
   358→
   359→  // ── Saga Step Execution ────────────────────────────────────────────────────
   360→
   361→  /**
   362→   * Step 1: Debit the buyer's wallet for the payment amount.
   363→   * Emits wallet.events.wallet_withdrawn (via wallet service consuming payment events).
   364→   */
   365→  private async executeWalletDebit(saga: PaymentSagaData): Promise<void> {
   366→    if (saga.state !== "INITIATED") return;
   367→
   368→    this.transition(this.getSagaId(saga), "DEBITING_WALLET");
   369→    const start = Date.now();
   370→
   371→    try {
   372→      // Build the wallet debit event
   373→      const walletDebitEvent = buildEvent(this.serviceName, {
   374→        eventType: "wallet_withdrawn",
   375→        topic: "wallet.events.wallet_withdrawn",
   376→        key: saga.wallet!.walletId,
   377→        payload: {
   378→          walletId: saga.wallet!.walletId,
   379→          userId: saga.payment!.userId,
   380→          amount: saga.payment!.amount,
   381→          currency: saga.payment!.currency,
   382→          balanceBefore: 0, // Wallet service will compute
   383→          balanceAfter: 0, // Wallet service will compute
   384→          destination: "escrow_hold",
   385→          withdrawalId: uuidv4(),
   386→        },
   387→        correlationId: saga.correlationId,
   388→        causationId: saga.sagaId,
   389→      });
   390→
   391→      // Note: In a real implementation, this would call the wallet service directly
   392→      // or use a command topic. For event-driven, we emit a command event.
   393→      // The wallet service consumes this and publishes wallet.events.wallet_withdrawn.
   394→      const result = await this.producer.produce(
   395→        "wallet.commands.wallet_withdraw",
   396→        saga.wallet!.walletId,
   397→        walletDebitEvent,
   398→      );
   399→
   400→      if (result.success) {
   401→        this.recordStep(this.getSagaId(saga), {
   402→          step: "WALLET_DEBIT",
   403→          status: "success",
   404→          durationMs: Date.now() - start,
   405→          timestamp: new Date().toISOString(),
   406→        });
   407→        // The saga waits for wallet.events.wallet_withdrawn to continue
   408→      } else {
   409→        throw new Error(`Wallet debit command failed: ${result.error?.message}`);
   410→      }
   411→    } catch (error) {
   412→      const err = error instanceof Error ? error : new Error(String(error));
   413→      this.recordStep(this.getSagaId(saga), {
   414→        step: "WALLET_DEBIT",
   415→        status: "failed",
   416→        error: err.message,
   417→        durationMs: Date.now() - start,
   418→        timestamp: new Date().toISOString(),
   419→      });
   420→      await this.compensate(this.getSagaId(saga), `Wallet debit failed: ${err.message}`);
   421→    }
   422→  }
   423→
   424→  /**
   425→   * Step 2: Create the escrow transaction.
   426→   */
   427→  private async executeEscrowCreation(sagaId: string): Promise<void> {
   428→    const saga = this.sagaInstances.get(sagaId);
   429→    if (!saga || (saga.state !== "WALLET_DEBITED" && saga.state !== "DEBITING_WALLET")) return;
   430→
   431→    this.transition(sagaId, "CREATING_ESCROW");
   432→    const start = Date.now();
   433→
   434→    try {
   435→      const escrowCreateEvent = buildEvent(this.serviceName, {
   436→        eventType: "escrow_created",
   437→        topic: "escrow.events.escrow_created",
   438→        key: saga.escrow!.escrowId,
   439→        payload: {
   440→          escrowId: saga.escrow!.escrowId,
   441→          transactionId: saga.escrow!.transactionId,
   442→          sellerId: saga.payment!.payeeId ?? "",
   443→          buyerId: saga.payment!.userId,
   444→          amount: saga.payment!.amount,
   445→          currency: saga.payment!.currency,
   446→          description: `Payment ${saga.payment!.reference} - Escrow hold`,
   447→          expiresAt: new Date(Date.now() + 30 * 24 * 60 * 60 * 1000).toISOString(), // 30-day expiry
   448→        },
   449→        correlationId: saga.correlationId,
   450→        causationId: saga.sagaId,
   451→      });
   452→
   453→      const result = await this.producer.produce(
   454→        "escrow.commands.escrow_create",
   455→        saga.escrow!.escrowId,
   456→        escrowCreateEvent,
   457→      );
   458→
   459→      if (result.success) {
   460→        this.recordStep(sagaId, {
   461→          step: "ESCROW_CREATE",
   462→          status: "success",
   463→          durationMs: Date.now() - start,
   464→          timestamp: new Date().toISOString(),
   465→        });
   466→        // Wait for escrow.events.escrow_created
   467→      } else {
   468→        throw new Error(`Escrow creation command failed: ${result.error?.message}`);
   469→      }
   470→    } catch (error) {
   471→      const err = error instanceof Error ? error : new Error(String(error));
   472→      this.recordStep(sagaId, {
   473→        step: "ESCROW_CREATE",
   474→        status: "failed",
   475→        error: err.message,
   476→        durationMs: Date.now() - start,
   477→        timestamp: new Date().toISOString(),
   478→      });
   479→      await this.compensate(sagaId, `Escrow creation failed: ${err.message}`);
   480→    }
   481→  }
   482→
   483→  /**
   484→   * Step 3: Send notification to buyer and seller.
   485→   * This is non-critical — failure does not trigger compensation.
   486→   */
   487→  private async executeNotificationSend(sagaId: string): Promise<void> {
   488→    const saga = this.sagaInstances.get(sagaId);
   489→    if (!saga || saga.state !== "ESCROW_CREATED") return;
   490→
   491→    this.transition(sagaId, "SENDING_NOTIFICATION");
   492→    const start = Date.now();
   493→
   494→    try {
   495→      // Send notification to buyer
   496→      const buyerNotification = buildEvent(this.serviceName, {
   497→        eventType: "notification_in_app",
   498→        topic: "notification.events.notification_in_app",
   499→        key: saga.payment!.userId,
   500→        payload: {
   501→          userId: saga.payment!.userId,
   502→          notificationId: uuidv4(),
   503→          title: "Payment Confirmed",
   504→          body: `Your payment of ${saga.payment!.currency} ${saga.payment!.amount} has been confirmed and is held in escrow.`,
   505→          category: "payment",
   506→          actionUrl: `/payments/${saga.payment!.paymentId}`,
   507→          read: false,
   508→        },
   509→        correlationId: saga.correlationId,
   510→        causationId: saga.sagaId,
   511→      });
   512→
   513→      // Send notification to seller/payee if present
   514→      const sellerNotification = saga.payment!.payeeId
   515→        ? buildEvent(this.serviceName, {
   516→            eventType: "notification_in_app",
   517→            topic: "notification.events.notification_in_app",
   518→            key: saga.payment!.payeeId!,
   519→            payload: {
   520→              userId: saga.payment!.payeeId!,
   521→              notificationId: uuidv4(),
   522→              title: "Payment Received",
   523→              body: `You have received a payment of ${saga.payment!.currency} ${saga.payment!.amount} held in escrow.`,
   524→              category: "payment",
   525→              actionUrl: `/escrow/${saga.escrow!.escrowId}`,
   526→              read: false,
   527→            },
   528→            correlationId: saga.correlationId,
   529→            causationId: saga.sagaId,
   530→          })
   531→        : null;
   532→
   533→      await this.producer.produce(
   534→        "notification.commands.notification_send",
   535→        saga.payment!.userId,
   536→        buyerNotification,
   537→      );
   538→
   539→      if (sellerNotification) {
   540→        await this.producer.produce(
   541→          "notification.commands.notification_send",
   542→          saga.payment!.payeeId!,
   543→          sellerNotification,
   544→        );
   545→      }
   546→
   547→      this.recordStep(sagaId, {
   548→        step: "NOTIFICATION_SEND",
   549→        status: "success",
   550→        durationMs: Date.now() - start,
   551→        timestamp: new Date().toISOString(),
   552→      });
   553→
   554→      // Complete the saga
   555→      this.transition(sagaId, "COMPLETED");
   556→    } catch (error) {
   557→      const err = error instanceof Error ? error : new Error(String(error));
   558→      // Notification failure is non-critical — mark saga as completed anyway
   559→      console.warn(
   560→        `[PaymentSaga] Notification send failed (non-critical) for sagaId=${sagaId}: ${err.message}`,
   561→      );
   562→      this.recordStep(sagaId, {
   563→        step: "NOTIFICATION_SEND",
   564→        status: "failed",
   565→        error: err.message,
   566→        durationMs: Date.now() - start,
   567→        timestamp: new Date().toISOString(),
   568→      });
   569→      this.transition(sagaId, "COMPLETED");
   570→    }
   571→  }
   572→
   573→  // ── Compensating Transactions ──────────────────────────────────────────────
   574→
   575→  /**
   576→   * Execute compensating (undo) transactions based on how far the saga progressed.
   577→   */
   578→  private async compensate(sagaId: string, reason: string): Promise<void> {
   579→    const saga = this.sagaInstances.get(sagaId);
   580→    if (!saga || saga.state === "COMPLETED" || saga.state === "FAILED") return;
   581→
   582→    saga.failureReason = reason;
   583→    saga.updatedAt = new Date().toISOString();
   584→
   585→    console.warn(
   586→      `[PaymentSaga] Initiating compensation for sagaId=${sagaId} currentState=${saga.state} reason=${reason}`,
   587→    );
   588→
   589→    // Determine which compensations are needed based on current state
   590→    const needsEscrowCancel =
   591→      saga.state === "CREATING_ESCROW" ||
   592→      saga.state === "ESCROW_CREATED" ||
   593→      saga.state === "SENDING_NOTIFICATION";
   594→
   595→    const needsWalletCredit =
   596→      needsEscrowCancel ||
   597→      saga.state === "WALLET_DEBITED" ||
   598→      saga.state === "DEBITING_WALLET";
   599→
   600→    // Execute compensations in reverse order of original steps
   601→    if (needsEscrowCancel && saga.escrow) {
   602→      this.transition(sagaId, "COMPENSATING_ESCROW");
   603→      await this.compensateEscrowCancel(saga);
   604→    }
   605→
   606→    if (needsWalletCredit && saga.wallet) {
   607→      this.transition(sagaId, "COMPENSATING_WALLET");
   608→      await this.compensateWalletCredit(saga);
   609→    }
   610→
   611→    // Mark saga as failed
   612→    this.transition(sagaId, "FAILED");
   613→    console.error(
   614→      `[PaymentSaga] Saga FAILED: sagaId=${sagaId} paymentId=${saga.payment?.paymentId} reason=${reason}`,
   615→    );
   616→
   617→    // Emit saga completion event for observability
   618→    await this.emitSagaCompletedEvent(saga, false);
   619→  }

   621→  /**
   622→   * Compensate: Cancel the escrow transaction.
   623→   */
   624→  private async compensateEscrowCancel(saga: PaymentSagaData): Promise<void> {
   625→    const start = Date.now();
   626→
   627→    try {
   628→      const cancelEvent = buildEvent(this.serviceName, {
   629→        eventType: "escrow_cancelled",
   630→        topic: "escrow.events.escrow_cancelled",
   631→        key: saga.escrow!.escrowId,
   632→        payload: {
   633→          escrowId: saga.escrow!.escrowId,
   634→          transactionId: saga.escrow!.transactionId,
   635→          cancelledBy: this.serviceName,
   636→          reason: saga.failureReason ?? "Saga compensation",
   637→          refundAmount: saga.escrow!.amount,
   638→          currency: saga.escrow!.currency,
   639→          refundedTo: saga.payment!.userId,
   640→        },
   641→        correlationId: saga.correlationId,
   642→        causationId: saga.sagaId,
   643→      });
   644→
   645→      await this.producer.produce(
   646→        "escrow.commands.escrow_cancel",
   647→        saga.escrow!.escrowId,
   648→        cancelEvent,
   649→      );
   650→
   651→      this.recordStep(this.getSagaId(saga), {
   652→        step: "COMPENSATE_ESCROW_CANCEL",
   653→        status: "success",
   654→        durationMs: Date.now() - start,
   655→        timestamp: new Date().toISOString(),
   656→      });
   657→    } catch (error) {
   658→      const err = error instanceof Error ? error : new Error(String(error));
   659→      console.error(
   660→        `[PaymentSaga] Compensation failed (escrow cancel) for sagaId=${saga.sagaId}: ${err.message}`,
   661→      );
   662→      this.recordStep(this.getSagaId(saga), {
   663→        step: "COMPENSATE_ESCROW_CANCEL",
   664→        status: "failed",
   665→        error: err.message,
   666→        durationMs: Date.now() - start,
   667→        timestamp: new Date().toISOString(),
   668→      });
   669→    }
   670→  }

   673→   * Compensate: Credit the buyer's wallet (refund the debit).
   674→   */
   675→  private async compensateWalletCredit(saga: PaymentSagaData): Promise<void> {
   676→    const start = Date.now();
   677→
   678→    try {
   679→      const creditEvent = buildEvent(this.serviceName, {
   680→        eventType: "wallet_deposited",

   685→          userId: saga.payment!.userId,
   686→          amount: saga.wallet!.amount,
   687→          currency: saga.wallet!.currency,
   688→          balanceBefore: 0, // Wallet service computes
   689→          balanceAfter: 0,
   690→          source: "refund",
   691→          referenceId: saga.payment!.paymentId,
   692→        },
   693→        correlationId: saga.correlationId,
   694→        causationId: saga.sagaId,
   695→        metadata: {
   696→          reason: "Saga compensation",
   697→          sagaId: saga.sagaId,

   699→      });
   700→
   701→      await this.producer.produce(
   702→        "wallet.commands.wallet_deposit",

   707→      this.recordStep(this.getSagaId(saga), {

   730→  /**
   731→   * Transition the saga to a new state. Validates the transition is legal.
   732→   */
   733→  private transition(sagaId: string, newState: SagaState): void {
   734→    const saga = this.sagaInstances.get(sagaId);
   735→    if (!saga) {
   736→      throw new Error(`Saga not found: ${sagaId}`);
   737→    }
   738→
   739→    const allowedTransitions = STATE_TRANSITIONS[saga.state];
   740→    if (!allowedTransitions.includes(newState)) {
   741→      throw new Error(
   742→        `Invalid state transition: ${saga.state} → ${newState}. ` +
   743→        `Allowed: [${allowedTransitions.join(", ")}]`,
   744→      );
   745→    }
   746→
   747→    const oldState = saga.state;
   748→    saga.state = newState;
   749→    saga.updatedAt = new Date().toISOString();
   750→
   751→    if (newState === "COMPLETED" || newState === "FAILED") {
   752→      saga.completedAt = new Date().toISOString();
   753→    }
   754→
   755→    console.info(
   756→      `[PaymentSaga] State transition: sagaId=${sagaId} ${oldState} → ${newState}`,
   757→    );
   758→  }

   762→  private createSagaInstance(
   763→    sagaId: string,
   764→    correlationId: string,
   765→    payment: PaymentInitiatedPayload,
   766→  ): PaymentSagaData {
   767→    const now = new Date().toISOString();
   768→    return {
   769→      sagaId,
   770→      correlationId,
   771→      state: "IDLE",
   772→      currentStep: null,
   773→      payment: {
   774→        paymentId: payment.paymentId,
   775→        userId: payment.userId,
   776→        payeeId: payment.payeeId,
   777→        amount: payment.amount,
   778→        currency: payment.currency,
   779→        paymentMethod: payment.paymentMethod,
   780→        provider: payment.provider,
   781→        reference: payment.reference,
   782→      },
   783→      wallet: {
   784→        walletId: `wallet:${payment.userId}`,
   785→        amount: payment.amount,
   786→        currency: payment.currency,
   787→      },
   788→      escrow: {
   789→        escrowId: uuidv4(),
   790→        transactionId: uuidv4(),
   791→        amount: payment.amount,
   792→        currency: payment.currency,
   793→      },
   794→      notification: {
   795→        channels: ["in_app", "email"],
   796→      },
   797→      stepHistory: [],
   798→      failureReason: null,
   799→      createdAt: now,
   800→      updatedAt: now,
   801→      completedAt: null,
   802→    };

   809→  private findSagaByPaymentId(paymentId: string): PaymentSagaData | undefined {
   810→    for (const saga of this.sagaInstances.values()) {
   811→      if (saga.payment?.paymentId === paymentId) {
   812→        return saga;
   813→      }
   814→    }
   815→    return undefined;
   816→  }

   818→  private recordStep(sagaId: string, result: SagaStepResult): void {
   819→    const saga = this.sagaInstances.get(sagaId);
   820→    if (saga) {
   821→      saga.currentStep = result.step;
   822→      saga.stepHistory.push(result);
   823→      saga.updatedAt = new Date().toISOString();
   824→    }
   825→  }

   829→  private async emitSagaCompletedEvent(saga: PaymentSagaData, success: boolean): Promise<void> {
   830→    const event = buildEvent(this.serviceName, {
   831→      eventType: "saga_completed",
   832→      topic: "audit.events.audit_action_logged",
   833→      key: saga.payment!.userId,
   834→      payload: {
   835→        userId: saga.payment!.userId,
   836→        action: success ? "payment_saga_completed" : "payment_saga_failed",
   837→        resource: "payment_saga",
   838→        resourceId: saga.sagaId,
   839→        newState: {
   840→          sagaId: saga.sagaId,
   841→          correlationId: saga.correlationId,
   842→          paymentId: saga.payment!.paymentId,
   843→          state: saga.state,
   844→          steps: saga.stepHistory,
   845→          durationMs: saga.completedAt
   846→            ? new Date(saga.completedAt).getTime() - new Date(saga.createdAt).getTime()
   847→            : 0,
   848→        },
   849→        changes: {
   850→          success,
   851→          failureReason: saga.failureReason,
   852→        },
   853→      },
   854→      correlationId: saga.correlationId,
   855→      causationId: saga.sagaId,
   856→    });
   857→
   858→    await this.producer.produce(
   859→      "audit.events.audit_action_logged",
   860→      saga.payment!.userId,
   861→      event,
   862→    );
   863→  }

   910→  /**
   911→   * Purge completed/failed saga instances older than the given age.
   912→   * Useful for memory management in long-running processes.
   913→   */
   914→  purgeCompletedSagas(maxAgeMs: number = 24 * 60 * 60 * 1000): number {
   915→    const now = Date.now();
   916→    let purged = 0;
   917→
   918→    for (const [sagaId, saga] of this.sagaInstances) {
   919→      if (
   920→        (saga.state === "COMPLETED" || saga.state === "FAILED") &&
   921→        saga.completedAt &&
   922→        now - new Date(saga.completedAt).getTime() > maxAgeMs
   923→      ) {
   924→        this.sagaInstances.delete(sagaId);
   925→        this.correlationToSaga.delete(saga.correlationId);
   926→        purged++;
   927→      }
   928→    }
   929→
   930→    if (purged > 0) {
   931→      console.info(`[PaymentSaga] Purged ${purged} completed saga instances.`);
   932→    }
   933→
   934→    return purged;
   935→  }
