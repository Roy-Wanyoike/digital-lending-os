     1→'use client'
     2→
     3→import { useState, useCallback } from 'react'
     4→import { motion, AnimatePresence } from 'framer-motion'
     5→import { Shield, Plus, Eye, DollarSign, Play, ChevronRight, AlertTriangle, CheckCircle, Clock, X, ExternalLink, CreditCard } from 'lucide-react'
     6→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
     7→import { Badge } from '@/components/ui/badge'
     8→import { Button } from '@/components/ui/button'
     9→import { Input } from '@/components/ui/input'
    10→import { Table, TableBody, TableCell, TableHead, TableHeader, TableRow } from '@/components/ui/table'
    11→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
    12→import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter } from '@/components/ui/dialog'
    13→import { Label } from '@/components/ui/label'
    14→import { Separator } from '@/components/ui/separator'
    15→import { ScrollArea } from '@/components/ui/scroll-area'
    16→import {
    17→  useApi, formatCurrency, getStatusBadgeVariant, getStatusColor,
    18→  getRiskBg, getRiskColor, formatDate, ESCROW_STATUSES, PipelineCard,
    19→  LoadingSkeleton, ErrorState, KPICard, CURRENCY_FLAGS, ScoreBar, Toast,
    20→  type EscrowTransaction, type Business,
    21→} from '@/lib/dashboard-helpers'
    22→
    23→// ─── Types ────────────────────────────────────────────────────
    24→
    25→interface Milestone { id: string; sequence: number; title: string; amount: number; status: string; releasedAt?: string | null }
    26→interface Disbursement { id: string; amount: number; currency: string; status: string; paymentRef?: string | null }
    27→interface Dispute { id: string; raisedBy: string; reason: string; description?: string | null; status: string; resolution?: string | null; aiRecommendation?: string | null; createdAt: string }
    28→interface AuditEntry { id: string; action: string; actor?: string | null; details: string; createdAt: string }
    29→interface EscrowDetail extends EscrowTransaction {
    30→  milestones: Milestone[]
    31→  disbursements: Disbursement[]
    32→  disputes: Dispute[]
    33→  auditLog: AuditEntry[]
    34→}
    35→
    36→// ─── Component ────────────────────────────────────────────────
    37→
    38→export function EscrowTab() {
    39→  const [refreshKey, setRefreshKey] = useState(0)
    40→  const { data: transactions, loading, error: txError, refetch } = useApi<EscrowTransaction[]>(`/api/escrow/transactions?limit=50&k=${refreshKey}`)
    41→  const { data: businesses, error: bizError } = useApi<Business[]>('/api/businesses?limit=50')
    42→  const [statusFilter, setStatusFilter] = useState('all')
    43→  const [selectedId, setSelectedId] = useState<string | null>(null)
    44→  const [createOpen, setCreateOpen] = useState(false)
    45→  const [disputeOpen, setDisputeOpen] = useState(false)
    46→  const [fundOpen, setFundOpen] = useState(false)
    47→  const [fundEscrowId, setFundEscrowId] = useState<string | null>(null)
    48→  const [toastMsg, setToastMsg] = useState('')
    49→  const [toastVis, setToastVis] = useState(false)
    50→
    51→  // Fund form state
    52→  const [fundEmail, setFundEmail] = useState('')
    53→  const [fundFirstName, setFundFirstName] = useState('')
    54→  const [fundLastName, setFundLastName] = useState('')
    55→  const [fundProvider, setFundProvider] = useState('')
    56→  const [fundProviders, setFundProviders] = useState<Array<{code: string; name: string}>>([])
    57→  const [funding, setFunding] = useState(false)
    58→
    59→  // Create form state
    60→  const [formBuyer, setFormBuyer] = useState('')
    61→  const [formSeller, setFormSeller] = useState('')
    62→  const [formAmount, setFormAmount] = useState('')
    63→  const [formCurrency, setFormCurrency] = useState('USD')
    64→  const [formDesc, setFormDesc] = useState('')
    65→  const [formMilestones, setFormMilestones] = useState('1')
    66→  const [creating, setCreating] = useState(false)
    67→
    68→  // Dispute form state
    69→  const [disputeRaisedBy, setDisputeRaisedBy] = useState('buyer')
    70→  const [disputeReason, setDisputeReason] = useState('')
    71→  const [disputeDesc, setDisputeDesc] = useState('')
    72→  const [disputing, setDisputing] = useState(false)
    73→
    74→  const showToast = useCallback((msg: string) => { setToastMsg(msg); setToastVis(true); setTimeout(() => setToastVis(false), 3000) }, [])
    75→
    76→  const doAction = useCallback(async (id: string, action: string, body?: Record<string, string>) => {
    77→    try {
    78→      const res = await fetch(`/api/escrow/transactions/${id}/${action}`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: body ? JSON.stringify(body) : undefined })
    79→      const json = await res.json()
    80→      if (!res.ok) throw new Error(json.error || 'Action failed')
    81→      showToast(`${action} successful!`)
    82→      setRefreshKey(k => k + 1)
    83→      return json.data
    84→    } catch (e: any) { showToast(e.message); return null }
    85→  }, [showToast])
    86→
    87→  const openFundDialog = useCallback(async (escrowId: string, currency: string) => {
    88→    setFundEscrowId(escrowId)
    89→    setFundOpen(true)
    90→    try {
    91→      const res = await fetch(`/api/payments/providers?currency=${currency}`)
    92→      const json = await res.json()
    93→      const providers = (json.data || []).map((p: any) => ({ code: p.code, name: p.name }))
    94→      setFundProviders(providers)
    95→      if (providers.length > 0) setFundProvider(providers[0].code)
    96→    } catch { setFundProviders([]) }
    97→  }, [])
    98→
    99→  if (loading) return <LoadingSkeleton />
   100→  if (txError || bizError) return <ErrorState message={txError || bizError || ''} onRetry={refetch} />
   101→
   102→  const allTxns = transactions || []
   103→  const filtered = statusFilter === 'all' ? allTxns : allTxns.filter(t => t.status?.toLowerCase() === statusFilter.toLowerCase())
   104→
   105→  const pipelineCounts = ESCROW_STATUSES.map(s => ({
   106→    status: s,
   107→    count: allTxns.filter(t => t.status?.toLowerCase() === s.toLowerCase().replace(/\s/g, '_') || t.status?.toLowerCase().replace(/\s/g, '') === s.toLowerCase().replace(/\s/g, '')).length,
   108→  }))
   109→  const pipelineColors = ['#94a3b8', '#3b82f6', '#f59e0b', '#10b981', '#ef4444']
   110→
   111→  const totalVolume = allTxns.reduce((s, t) => s + t.amount, 0)
   112→  const activeEscrows = allTxns.filter(t => ['funded', 'in_escrow'].includes(t.status?.toLowerCase())).length
   113→  const disputed = allTxns.filter(t => t.status?.toLowerCase() === 'disputed').length
   114→
   115→  // ─── Derived data ────────────────────────────────────────
   116→
   117→  const handleFund = async () => {
   118→    if (!fundEscrowId || !fundEmail) return
   119→    setFunding(true)
   120→    try {
   121→      const body: Record<string, string> = { email: fundEmail }
   122→      if (fundFirstName) body.firstName = fundFirstName
   123→      if (fundLastName) body.lastName = fundLastName
   124→      if (fundProvider) body.provider = fundProvider
   125→      const res = await fetch(`/api/escrow/transactions/${fundEscrowId}/fund`, {
   126→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   127→        body: JSON.stringify(body),
   128→      })
   129→      const json = await res.json()
   130→      if (!res.ok) throw new Error(json.error || 'Funding failed')
   131→      const data = json.data
   132→      if (data?.checkoutUrl) {
   133→        window.open(data.checkoutUrl, '_blank', 'noopener,noreferrer')
   134→        showToast(`Redirecting to ${data.providerName}...`)
   135→      } else if (data?.status === 'awaiting_payment') {
   136→        showToast('Payment initiated! Complete it via the provider.')
   137→      } else {
   138→        showToast('Escrow funding initiated!')
   139→      }
   140→      setFundOpen(false)
   141→      setFundEmail(''); setFundFirstName(''); setFundLastName(''); setFundProvider('')
   142→      setRefreshKey(k => k + 1)
   143→    } catch (e: any) { showToast(e.message) } finally { setFunding(false) }
   144→  }
   145→
   146→  const handleCreate = async () => {
   147→    setCreating(true)
   148→    try {
   149→      const milestones = Array.from({ length: parseInt(formMilestones) }, (_, i) => ({
   150→        title: `Milestone ${i + 1}`,
   151→        amount: Math.round((parseFloat(formAmount) / parseInt(formMilestones)) * 100) / 100,
   152→      }))
   153→      const res = await fetch('/api/escrow/transactions', {
   154→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   155→        body: JSON.stringify({ buyerId: formBuyer, sellerId: formSeller, amount: parseFloat(formAmount), currency: formCurrency, description: formDesc || undefined, milestones }),
   156→      })
   157→      const json = await res.json()
   158→      if (!res.ok) throw new Error(json.error || json.details?.[0]?.message || 'Create failed')
   159→      showToast('Escrow created!')
   160→      setCreateOpen(false)
   161→      setFormBuyer(''); setFormSeller(''); setFormAmount(''); setFormDesc(''); setFormMilestones('1')
   162→      setRefreshKey(k => k + 1)
   163→    } catch (e: any) { showToast(e.message) } finally { setCreating(false) }
   164→  }
   165→
   166→  const handleDispute = async () => {
   167→    if (!selectedId || !disputeReason) return
   168→    setDisputing(true)
   169→    try {
   170→      const res = await fetch(`/api/escrow/transactions/${selectedId}/disputes`, {
   171→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   172→        body: JSON.stringify({ raisedBy: disputeRaisedBy, reason: disputeReason, description: disputeDesc || undefined }),
   173→      })
   174→      const json = await res.json()
   175→      if (!res.ok) throw new Error(json.error || 'Dispute failed')
   176→      showToast('Dispute raised!')
   177→      setDisputeOpen(false); setDisputeReason(''); setDisputeDesc('')
   178→      setRefreshKey(k => k + 1)
   179→    } catch (e: any) { showToast(e.message) } finally { setDisputing(false) }
   180→  }
   181→
   182→  const selectedTxn = allTxns.find(t => t.id === selectedId)
   183→
   184→  return (
   185→    <motion.div initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} className="space-y-6">
   186→      {/* KPIs */}
   187→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   188→        <KPICard title="Total Volume" value={formatCurrency(totalVolume)} icon={DollarSign} />
   189→        <KPICard title="Active Escrows" value={activeEscrows.toString()} icon={Shield} />
   190→        <KPICard title="Disputed" value={disputed.toString()} icon={AlertTriangle} />
   191→        <KPICard title="Total Deals" value={allTxns.length.toString()} icon={ChevronRight} />
   192→      </div>
   193→
   194→      {/* Pipeline */}
   195→      <div className="flex gap-2 sm:gap-3 overflow-x-auto pb-2">
   196→        {pipelineCounts.map((p, i) => (
   197→          <PipelineCard key={p.status} label={p.status} count={p.count} color={pipelineColors[i]} />
   198→        ))}
   199→      </div>
   200→
   201→      {/* Toolbar */}
   202→      <div className="flex flex-wrap items-center justify-between gap-3">
   203→        <div className="flex items-center gap-3">
   204→          <span className="text-sm text-muted-foreground">Status:</span>
   205→          <Select value={statusFilter} onValueChange={setStatusFilter}>
   206→            <SelectTrigger className="w-40"><SelectValue /></SelectTrigger>
   207→            <SelectContent>
   208→              <SelectItem value="all">All Statuses</SelectItem>
   209→              {ESCROW_STATUSES.map(s => <SelectItem key={s} value={s.toLowerCase().replace(/\s/g, '_')}>{s}</SelectItem>)}
   210→            </SelectContent>
   211→          </Select>
   212→        </div>
   213→        <Button size="sm" onClick={() => setCreateOpen(true)}><Plus className="h-4 w-4 mr-1" /> New Escrow</Button>
   214→      </div>
   215→
   216→      {/* Table */}
   217→      <Card>
   218→        <CardContent className="p-0">
   219→          <div className="overflow-x-auto">
   220→            <Table>
   221→              <TableHeader>
   222→                <TableRow>
   223→                  <TableHead>Reference</TableHead>
   224→                  <TableHead>Buyer → Seller</TableHead>
   225→                  <TableHead className="text-right">Amount</TableHead>
   226→                  <TableHead>Status</TableHead>
   227→                  <TableHead>AI Risk</TableHead>
   228→                  <TableHead>Milestones</TableHead>
   229→                  <TableHead>Actions</TableHead>
   230→                  <TableHead>Date</TableHead>
   231→                </TableRow>
   232→              </TableHeader>
   233→              <TableBody>
   234→                {filtered.slice(0, 20).map(txn => {
   235→                  const st = txn.status?.toLowerCase() || ''
   236→                  const canFund = st === 'created'
   237→                  const canActivate = st === 'funded'
   238→                  const canRelease = st === 'in_escrow'
   239→                  const canDispute = ['funded', 'in_escrow'].includes(st)
   240→                  return (
   241→                    <TableRow key={txn.id} className="even:bg-muted/50">
   242→                      <TableCell className="font-mono text-xs">{txn.txRef}</TableCell>
   243→                      <TableCell className="max-w-[160px] truncate text-xs">
   244→                        <span className="text-foreground">{txn.buyer?.name}</span>
   245→                        <span className="text-muted-foreground mx-1">→</span>
   246→                        <span className="text-foreground">{txn.seller?.name}</span>
   247→                      </TableCell>
   248→                      <TableCell className="text-right font-medium">{formatCurrency(txn.amount, txn.currency)}</TableCell>
   249→                      <TableCell><Badge variant={getStatusBadgeVariant(txn.status)} className={getStatusColor(txn.status)}>{txn.status}</Badge></TableCell>
   250→                      <TableCell>
   251→                        <div className="flex items-center gap-2">
   252→                          <div className="w-10 h-1.5 bg-muted rounded-full overflow-hidden">
   253→                            <div className={`h-full rounded-full ${getRiskBg(txn.aiRiskScore || 0)}`} style={{ width: `${txn.aiRiskScore || 0}%` }} />
   254→                          </div>
   255→                          <span className={`text-xs font-medium ${getRiskColor(txn.aiRiskScore || 0)}`}>{txn.aiRiskLevel}</span>
   256→                        </div>
   257→                      </TableCell>
   258→                      <TableCell className="text-xs">{txn.currentMilestone}/{txn.totalMilestones}</TableCell>
   259→                      <TableCell>
   260→                        <div className="flex items-center gap-1">
   261→                          <Button variant="ghost" size="icon" className="h-7 w-7" onClick={() => setSelectedId(txn.id)} title="View details"><Eye className="h-3.5 w-3.5" /></Button>
   262→                          {canFund && <Button variant="outline" size="sm" className="h-7 text-xs" onClick={() => openFundDialog(txn.id, txn.currency)}><CreditCard className="h-3 w-3 mr-1" />Fund</Button>}
   263→                          {canActivate && <Button variant="outline" size="sm" className="h-7 text-xs" onClick={() => doAction(txn.id, 'activate')}><Play className="h-3 w-3 mr-1" />Activate</Button>}
   264→                          {canRelease && <Button variant="outline" size="sm" className="h-7 text-xs" onClick={() => {
   265→                            const ms = txn.milestones?.find((m: any) => m.status === 'ready' || m.status === 'pending')
   266→                            if (ms) doAction(txn.id, 'release', { milestoneId: ms.id })
   267→                            else showToast('No pending milestones')
   268→                          }}><CheckCircle className="h-3 w-3 mr-1" />Release</Button>}
   269→                          {canDispute && <Button variant="outline" size="sm" className="h-7 text-xs text-red-600" onClick={() => { setSelectedId(txn.id); setDisputeOpen(true) }}><AlertTriangle className="h-3 w-3 mr-1" />Dispute</Button>}
   270→                        </div>
   271→                      </TableCell>
   272→                      <TableCell className="text-xs text-muted-foreground whitespace-nowrap">{formatDate(txn.createdAt)}</TableCell>
   273→                    </TableRow>
   274→                  )
   275→                })}
   276→              </TableBody>
   277→            </Table>
   278→          </div>
   279→        </CardContent>
   280→      </Card>
   281→
   282→      {/* Detail Drawer */}
   283→      <AnimatePresence>
   284→        {selectedTxn && (
   285→          <motion.div initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }} className="fixed inset-0 z-50 flex justify-end">
   286→            <div className="absolute inset-0 bg-black/30" onClick={() => setSelectedId(null)} />
   287→            <motion.div initial={{ x: '100%' }} animate={{ x: 0 }} exit={{ x: '100%' }} transition={{ type: 'spring', damping: 25, stiffness: 300 }} className="relative w-full max-w-xl bg-background shadow-xl h-full overflow-y-auto border-l">
   288→              <div className="sticky top-0 z-10 bg-background border-b px-6 py-4 flex items-center justify-between">
   289→                <div>
   290→                  <h3 className="font-semibold">{selectedTxn.txRef}</h3>
   291→                  <p className="text-xs text-muted-foreground">{formatDate(selectedTxn.createdAt)}</p>
   292→                </div>
   293→                <Button variant="ghost" size="icon" onClick={() => setSelectedId(null)}><X className="h-4 w-4" /></Button>
   294→              </div>
   295→              <div className="p-6 space-y-6">
   296→                {/* Status & Amount */}
   297→                <div className="flex items-center justify-between">
   298→                  <Badge variant={getStatusBadgeVariant(selectedTxn.status)} className={getStatusColor(selectedTxn.status)}>{selectedTxn.status}</Badge>
   299→                  <p className="text-2xl font-bold">{formatCurrency(selectedTxn.amount, selectedTxn.currency)}</p>
   300→                </div>
   301→                <div className="grid grid-cols-2 gap-4 text-sm">
   302→                  <div><p className="text-muted-foreground">Buyer</p><p className="font-medium">{selectedTxn.buyer?.name}</p></div>
   303→                  <div><p className="text-muted-foreground">Seller</p><p className="font-medium">{selectedTxn.seller?.name}</p></div>
   304→                  <div><p className="text-muted-foreground">Funded</p><p className="font-medium text-blue-600 dark:text-blue-400">{formatCurrency(selectedTxn.fundedAmount, selectedTxn.currency)}</p></div>
   305→                  <div><p className="text-muted-foreground">Released</p><p className="font-medium text-emerald-600 dark:text-emerald-400">{formatCurrency(selectedTxn.releasedAmount, selectedTxn.currency)}</p></div>
   306→                </div>
   307→
   308→                {/* Risk Score */}
   309→                <div>
   310→                  <p className="text-sm font-medium mb-2">AI Risk Assessment</p>
   311→                  <ScoreBar score={selectedTxn.aiRiskScore || 0} label="Risk Score" />
   312→                </div>
   313→
   314→                {/* Milestones */}
   315→                {selectedTxn.milestones && selectedTxn.milestones.length > 0 && (
   316→                  <div>
   317→                    <p className="text-sm font-medium mb-2">Milestones</p>
   318→                    <div className="space-y-2">
   319→                      {selectedTxn.milestones.map((m: any) => (
   320→                        <div key={m.id} className="flex items-center justify-between p-3 rounded-lg bg-muted">
   321→                          <div className="flex items-center gap-2">
   322→                            {m.status === 'released' ? <CheckCircle className="h-4 w-4 text-emerald-500" /> : <Clock className="h-4 w-4 text-muted-foreground" />}
   323→                            <span className="text-sm font-medium">{m.title}</span>
   324→                          </div>
   325→                          <span className="text-sm font-medium">{formatCurrency(m.amount, selectedTxn.currency)}</span>
   326→                        </div>
   327→                      ))}
   328→                    </div>
   329→                  </div>
   330→                )}
   331→
   332→                {/* Disputes */}
   333→                {selectedTxn.disputes && selectedTxn.disputes.length > 0 && (
   334→                  <div>
   335→                    <p className="text-sm font-medium mb-2">Disputes</p>
   336→                    {selectedTxn.disputes.map((d: any) => (
   337→                      <div key={d.id} className="p-3 rounded-lg bg-red-50 dark:bg-red-950/30 border border-red-100 dark:border-red-900 mb-2">
   338→                        <div className="flex items-center justify-between mb-1">
   339→                          <Badge variant={d.status === 'open' ? 'destructive' : 'secondary'}>{d.status}</Badge>
   340→                          <span className="text-xs text-muted-foreground">{formatDate(d.createdAt)}</span>
   341→                        </div>
   342→                        <p className="text-sm font-medium">{d.reason}</p>
   343→                        {d.aiRecommendation && <p className="text-xs text-muted-foreground mt-1">AI: {d.aiRecommendation}</p>}
   344→                        {d.resolution && <p className="text-xs text-emerald-600 mt-1">Resolution: {d.resolution}</p>}
   345→                      </div>
   346→                    ))}
   347→                  </div>
   348→                )}
   349→
   350→                {/* Description */}
   351→                {selectedTxn.description && (
   352→                  <div>
   353→                    <p className="text-sm font-medium mb-1">Description</p>
   354→                    <p className="text-sm text-muted-foreground">{selectedTxn.description}</p>
   355→                  </div>
   356→                )}
   357→
   358→                {/* Fees */}
   359→                <div className="grid grid-cols-2 gap-4 text-sm">
   360→                  <div><p className="text-muted-foreground">Fee</p><p>{formatCurrency(selectedTxn.feeAmount, selectedTxn.feeCurrency)}</p></div>
   361→                  <div><p className="text-muted-foreground">Refunded</p><p>{formatCurrency(selectedTxn.refundedAmount, selectedTxn.currency)}</p></div>
   362→                </div>
   363→              </div>
   364→            </motion.div>
   365→          </motion.div>
   366→        )}
   367→      </AnimatePresence>
   368→
   369→      {/* Fund Dialog */}
   370→      <Dialog open={fundOpen} onOpenChange={setFundOpen}>
   371→        <DialogContent className="max-w-md">
   372→          <DialogHeader>
   373→            <DialogTitle className="flex items-center gap-2"><CreditCard className="h-5 w-5" /> Fund Escrow</DialogTitle>
   374→            <DialogDescription>Pay via a payment provider to fund this escrow. You will be redirected to complete payment.</DialogDescription>
   375→          </DialogHeader>
   376→          <div className="space-y-4">
   377→            {fundProviders.length > 0 && (
   378→              <div className="space-y-2"><Label>Payment Provider</Label>
   379→                <Select value={fundProvider} onValueChange={setFundProvider}>
   380→                  <SelectTrigger><SelectValue placeholder="Select provider" /></SelectTrigger>
   381→                  <SelectContent>
   382→                    {fundProviders.map(p => <SelectItem key={p.code} value={p.code}>{p.name}</SelectItem>)}
   383→                  </SelectContent>
   384→                </Select>
   385→                <p className="text-xs text-muted-foreground">Auto-selected based on escrow currency. Fees apply per provider.</p>
   386→              </div>
   387→            )}
   388→            {fundProviders.length === 0 && (
   389→              <div className="p-3 rounded-lg bg-amber-50 dark:bg-amber-950/30 border border-amber-200 dark:border-amber-800 text-sm text-amber-700 dark:text-amber-300">
   390→                No payment providers configured. Payments will use demo mode. Add provider keys in .env to enable real payments.
   391→              </div>
   392→            )}
   393→            <div className="space-y-2"><Label>Email *</Label><Input type="email" placeholder="payer@email.com" value={fundEmail} onChange={e => setFundEmail(e.target.value)} /></div>
   394→            <div className="grid grid-cols-2 gap-3">
   395→              <div className="space-y-2"><Label>First Name</Label><Input placeholder="John" value={fundFirstName} onChange={e => setFundFirstName(e.target.value)} /></div>
