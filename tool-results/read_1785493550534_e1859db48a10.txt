     1→/**
     2→ * Youngsend Kafka Event Schemas
     3→ *
     4→ * Strict Zod schemas for every event type. Each event includes:
     5→ * - eventId: UUID v4 — unique identifier for this event
     6→ * - eventType: String discriminator matching the topic event type
     7→ * - timestamp: ISO 8601 datetime of event creation
     8→ * - version: Semantic version of the event schema (e.g. "1.0.0")
     9→ * - correlationId: UUID linking all events in a single business transaction
    10→ * - causationId: UUID linking this event to the parent event that triggered it
    11→ * - sourceService: Name of the producing microservice
    12→ * - payload: Domain-specific data (strictly typed per event type)
    13→ * - metadata: Optional key-value pairs for tracing, feature flags, etc.
    14→ */
    15→
    16→import { z } from "zod";
    17→
    18→// ─── Base Event Envelope ───────────────────────────────────────────────────────
    19→
    20→export const EventMetadataSchema = z
    21→  .record(z.string(), z.unknown())
    22→  .optional()
    23→  .default({});
    24→
    25→export const BaseEventSchema = z.object({
    26→  /** UUID v4 — globally unique event identifier */
    27→  eventId: z.string().uuid(),
    28→  /** Discriminator matching the event type (e.g. "payment_initiated") */
    29→  eventType: z.string(),
    30→  /** ISO 8601 datetime string */
    31→  timestamp: z.string().datetime({ offset: true }),
    32→  /** Semantic version of this event schema */
    33→  version: z.string(),
    34→  /** Links all events belonging to the same business transaction */
    35→  correlationId: z.string().uuid(),
    36→  /** Links this event to the parent event that caused it */
    37→  causationId: z.string().uuid().optional().default("00000000-0000-0000-0000-000000000000"),
    38→  /** Name of the originating microservice */
    39→  sourceService: z.string(),
    40→  /** Domain-specific payload */
    41→  payload: z.unknown(),
    42→  /** Optional metadata for tracing, feature flags, etc. */
    43→  metadata: EventMetadataSchema,
    44→});
    45→
    46→export type BaseEvent = z.infer<typeof BaseEventSchema>;

   691→// ─── Validation Helpers ────────────────────────────────────────────────────────
   692→
   693→/**
   694→ * Parse and validate a raw event against a specific schema.
   695→ * Returns { success: true, data } or { success: false, errors }.
   696→ */
   697→export function parseEvent<T>(schema: z.ZodSchema<T>, raw: unknown): {
   698→  success: boolean;
   699→  data?: T;
   700→  errors?: z.ZodError;
   701→} {
   702→  const result = schema.safeParse(raw);
   703→  if (result.success) {
   704→    return { success: true, data: result.data };
   705→  }
   706→  return { success: false, errors: result.error };
   707→}
   708→
   709→/**
   710→ * Parse an event against its eventType discriminator.
   711→ * Routes to the correct schema based on the eventType field.
   712→ */
   713→export function parseEventByType(raw: unknown): {
   714→  success: boolean;
   715→  data?: AnyYoungsendEvent;
   716→  errors?: z.ZodError;
   717→} {
   718→  return parseEvent(AnyYoungsendEventSchema, raw);
   719→}
