   231→const REMIND_INTERVAL_DAYS: Record<string, number> = {
   232→  friendly: 7,
   233→  firm: 3,
   234→  final: 1,
   235→  legal: 0,
   236→}
   237→
   238→const remindSchema = z.object({
   239→  channel: z.enum(['email', 'sms', 'whatsapp', 'in_app'] as const),
   240→  template: z.enum(['friendly', 'firm', 'final', 'legal'] as const),
   241→})
   242→
   243→export async function POST(
   244→  request: NextRequest,
   245→  { params }: { params: Promise<{ id: string }> }
   246→) {
   247→  try {
   248→    const user = await getApiUser(request)
   249→    const { id } = await params
   250→    const body = await request.json()
   251→    const parsed = remindSchema.safeParse(body)
   252→
   253→    if (!parsed.success) {
   254→      return NextResponse.json(
   255→        { error: parsed.error.issues.map((i) => i.message).join(', ') },
   256→        { status: 400 }
   257→      )
   258→    }
   259→
   260→    const data = parsed.data
   261→
   262→    const collectionCase = await db.collectionCase.findUnique({ where: { id } })
   263→    if (!collectionCase) {
   264→      return NextResponse.json({ error: 'Collection case not found' }, { status: 404 })
   265→    }
   266→
   267→    // Verify tenant access
   268→    const biz = await db.business.findUnique({
   269→      where: { id: collectionCase.businessId },
   270→      select: { tenantId: true },
   271→    })
   272→    if (!biz || biz.tenantId !== user.tenantId) {
   273→      return NextResponse.json({ error: 'Collection case not found' }, { status: 404 })
   274→    }
   275→
   276→    const now = new Date()
   277→    const intervalDays = REMIND_INTERVAL_DAYS[data.template]
   278→    const nextReminderDue = intervalDays === 0
   279→      ? now
   280→      : new Date(now.getTime() + intervalDays * 24 * 60 * 60 * 1000)
   281→
   282→    const reminder = await db.$transaction(async (tx) => {
   283→      const newReminder = await tx.collectionReminder.create({
   284→        data: {
   285→          caseId: id,
   286→          channel: data.channel,
   287→          template: data.template,
   288→          status: 'sent',
   289→          sentAt: now,
   290→        },
   291→      })
   292→
   293→      await tx.collectionCase.update({
   294→        where: { id },
   295→        data: {
   296→          reminderCount: { increment: 1 },
   297→          lastReminderAt: now,
   298→          nextReminderDue,
   299→          ...(data.template === 'legal' ? { status: 'escalated', priority: 'urgent' } : {}),
   300→        },
   301→      })
   302→
   303→      return newReminder
   304→    })
   305→
   306→    return NextResponse.json({ data: reminder }, { status: 201 })
   307→  } catch (error) {
   308→    console.error('Error sending collection reminder:', error)
   309→    if (error instanceof AuthError) return NextResponse.json({ error: error.message }, { status: error.status })
   310→    return NextResponse.json({ error: 'Failed to send collection reminder' }, { status: 500 })
   311→  }
   312→}
