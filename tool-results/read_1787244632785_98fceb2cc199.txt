   224→interface OverdueLoansTableProps {
   225→  onLoanSelect?: (loan: OverdueLoan) => void
   226→  filterCollector?: string | null // 'mine', 'unassigned', or agent ID
   227→  filterBucket?: string | null // bucket name to filter by
   228→  highPriorityOnly?: boolean
   229→  agents?: CollectionAgent[]
   230→}
   231→
   232→type SortField = 'loanNumber' | 'customerName' | 'daysInArrears' | 'outstandingBalance' | 'status' | 'collectorName'
   233→type SortDirection = 'asc' | 'desc'
   234→
   235→export function OverdueLoansTable({
   236→  onLoanSelect,
   237→  filterCollector = null,
   238→  filterBucket = null,
   239→  highPriorityOnly = false,
   240→  agents = []
   241→}: OverdueLoansTableProps) {
   242→  const [searchQuery, setSearchQuery] = useState('')
   243→  const [sortField, setSortField] = useState<SortField>('daysInArrears')
   244→  const [sortDirection, setSortDirection] = useState<SortDirection>('desc')
   245→  const [selectedLoans, setSelectedLoans] = useState<Set<string>>(new Set())
   246→  const [currentPage, setCurrentPage] = useState(1)
   247→  const [isLoading, setIsLoading] = useState(false)
   248→  const itemsPerPage = 10
   249→
   250→  // Filter and sort data
   251→  const filteredAndSortedData = useMemo(() => {
   252→    let result = [...mockLoans]
   253→
   254→    // Apply search filter
   255→    if (searchQuery) {
   256→      const query = searchQuery.toLowerCase()
   257→      result = result.filter(loan =>
   258→        loan.loanNumber.toLowerCase().includes(query) ||
   259→        loan.customerName.toLowerCase().includes(query) ||
   260→        loan.customerPhone.includes(query)
   261→      )
   262→    }
   263→
   264→    // Apply collector filter
   265→    if (filterCollector === 'unassigned') {
   266→      result = result.filter(loan => !loan.assignedCollectorId)
   267→    } else if (filterCollector === 'mine') {
   268→      // In production, this would use actual user ID
   269→      result = result.filter(loan => loan.assignedCollectorId === 'agent-001')
   270→    } else if (filterCollector && filterCollector !== 'all') {
   271→      result = result.filter(loan => loan.assignedCollectorId === filterCollector)
   272→    }
   273→
   274→    // Apply bucket filter (days in arrears range)
   275→    if (filterBucket) {
   276→      switch (filterBucket) {
   277→        case '1-7 days':
   278→          result = result.filter(loan => loan.daysInArrears >= 1 && loan.daysInArrears <= 7)
   279→          break
   280→        case '8-30 days':
   281→          result = result.filter(loan => loan.daysInArrears >= 8 && loan.daysInArrears <= 30)
   282→          break
   283→        case '31-60 days':
   284→          result = result.filter(loan => loan.daysInArrears >= 31 && loan.daysInArrears <= 60)
   285→          break
   286→        case '61-90 days':
   287→          result = result.filter(loan => loan.daysInArrears >= 61 && loan.daysInArrears <= 90)
   288→          break
   289→        case '90+ days':
   290→          result = result.filter(loan => loan.daysInArrears > 90)
   291→          break
   292→      }
   293→    }
   294→
   295→    // Apply high priority filter (loans with 30+ days in arrears or very high risk)
   296→    if (highPriorityOnly) {
   297→      result = result.filter(loan =>
   298→        loan.daysInArrears >= 30 || loan.riskLevel === 'VERY_HIGH' || loan.riskLevel === 'HIGH'
   299→      )
   300→    }
   301→
   302→    // Apply sorting
   303→    result.sort((a, b) => {
   304→      let aVal: string | number = a[sortField] || ''
   305→      let bVal: string | number = b[sortField] || ''
   306→      
   307→      if (typeof aVal === 'string') aVal = aVal.toLowerCase()
   308→      if (typeof bVal === 'string') bVal = bVal.toLowerCase()
   309→      
   310→      if (aVal < bVal) return sortDirection === 'asc' ? -1 : 1
   311→      if (aVal > bVal) return sortDirection === 'asc' ? 1 : -1
   312→      return 0
   313→    })
   314→
   315→    return result
   316→  }, [searchQuery, sortField, sortDirection, filterCollector, filterBucket, highPriorityOnly])
   317→
   318→  // Paginate data
   319→  const paginatedData = useMemo(() => {
   320→    const startIndex = (currentPage - 1) * itemsPerPage
   321→    return filteredAndSortedData.slice(startIndex, startIndex + itemsPerPage)
   322→  }, [filteredAndSortedData, currentPage])
   323→
   324→  const totalPages = Math.ceil(filteredAndSortedData.length / itemsPerPage)
   325→
   326→  // Handle sort toggle
   327→  const handleSort = (field: SortField) => {
   328→    if (sortField === field) {
   329→      setSortDirection(prev => prev === 'asc' ? 'desc' : 'asc')
   330→    } else {
   331→      setSortField(field)
   332→      setSortDirection('desc')
   333→    }
   334→  }
   335→
   336→  // Handle selection
   337→  const handleSelectAll = () => {
   338→    if (selectedLoans.size === paginatedData.length) {
   339→      setSelectedLoans(new Set())
   340→    } else {
   341→      setSelectedLoans(new Set(paginatedData.map(l => l.id)))
   342→    }
   343→  }
   344→
   345→  const handleSelectOne = (id: string) => {
   346→    const newSet = new Set(selectedLoans)
   347→    if (newSet.has(id)) {
   348→      newSet.delete(id)
   349→    } else {
   350→      newSet.add(id)
   351→    }
   352→    setSelectedLoans(newSet)
   353→  }
   354→
   355→  // Handle bulk assignment
   356→  const handleBulkAssign = async (agentId: string) => {
   357→    setIsLoading(true)
   358→    try {
   359→      // Simulate API call
   360→      await new Promise(resolve => setTimeout(resolve, 500))
   361→      toast.success(`Assigned ${selectedLoans.size} loans to agent`)
   362→      setSelectedLoans(new Set())
   363→    } catch (error) {
   364→      toast.error('Failed to assign loans')
   365→    } finally {
   366→      setIsLoading(false)
   367→    }
   368→  }
   369→
   370→  // Handle contact action
   371→  const handleContactAction = (method: string, loan: OverdueLoan) => {
   372→    toast.success(`Initiating ${method.toUpperCase()} to ${loan.customerName}`)
   373→  }
   374→
   375→  // Format currency
   376→  const formatCurrency = (amount: number): string => `KSh ${amount.toLocaleString()}`
   377→
   378→  // Get row color based on arrears status
   379→  const getRowClassName = (daysInArrears: number): string => {
   380→    if (daysInArrears > 90) return 'bg-red-50/50 dark:bg-red-950/20'
   381→    if (daysInArrears > 60) return 'bg-orange-50/50 dark:bg-orange-950/20'
   382→    if (daysInArrears > 30) return 'bg-amber-50/50 dark:bg-amber-950/20'
   383→    if (daysInArrears > 7) return 'bg-yellow-50/50 dark:bg-yellow-950/20'
   384→    return ''
   385→  }
   386→
   387→  // Get severity badge
   388→  const getSeverityBadge = (daysInArrears: number) => {
   389→    if (daysInArrears > 90) return <Badge className="bg-rose-100 text-rose-800 dark:bg-rose-900/40 dark:text-rose-400 border-0">Severe</Badge>
   390→    if (daysInArrears > 60) return <Badge className="bg-red-100 text-red-800 dark:bg-red-900/40 dark:text-red-400 border-0">Critical</Badge>
   391→    if (daysInArrears > 30) return <Badge className="bg-orange-100 text-orange-800 dark:bg-orange-900/40 dark:text-orange-400 border-0">High</Badge>
   392→    if (daysInArrears > 7) return <Badge className="bg-amber-100 text-amber-800 dark:bg-amber-900/40 dark:text-amber-400 border-0">Medium</Badge>
   393→    return <Badge className="bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-400 border-0">Low</Badge>
   394→  }
   395→
   396→  // Sort icon component
   397→  const SortIcon = ({ field }: { field: SortField }) => {
   398→    if (sortField !== field) return <ArrowUpDown className="w-4 h-4 ml-1 opacity-50" />
   399→    return sortDirection === 'asc' 
   400→      ? <ArrowUp className="w-4 h-4 ml-1" />
   401→      : <ArrowDown className="w-4 h-4 ml-1" />
   402→  }
   403→
   404→  return (
   405→    <div className="space-y-4">
   406→      {/* Search and Actions Bar */}
   407→      <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4">
   408→        <div className="relative w-full sm:w-auto sm:flex-1 max-w-sm">
   409→          <Search className="absolute left-3 top-1/2 transform -translate-y-1/2 w-4 h-4 text-slate-400" />
   410→          <Input
   411→            placeholder="Search loans, customers..."
   412→            value={searchQuery}
   413→            onChange={(e) => setSearchQuery(e.target.value)}
   414→            className="pl-10 bg-white dark:bg-slate-800"
   415→          />
   416→        </div>
   417→        
   418→        <div className="flex gap-2 w-full sm:w-auto">
   419→          {selectedLoans.size > 0 && (
   420→            <DropdownMenu>
   421→              <DropdownMenuTrigger asChild>
   422→                <Button variant="outline" size="sm" className="dark:border-slate-700">
   423→                  <UserPlus className="w-4 h-4 mr-2" />
   424→                  Assign ({selectedLoans.size})
   425→                </Button>
   426→              </DropdownMenuTrigger>
   427→              <DropdownMenuContent align="end">
   428→                {agents.map(agent => (
   429→                  <DropdownMenuItem key={agent.id} onClick={() => handleBulkAssign(agent.id)}>
   430→                    {agent.name}
   431→                  </DropdownMenuItem>
   432→                ))}
   433→              </DropdownMenuContent>
   434→            </DropdownMenu>
   435→          )}
   436→          
   437→          <Button variant="outline" size="sm" className="dark:border-slate-700">
   438→            Export CSV
   439→          </Button>
   440→        </div>
   441→      </div>
   442→
   443→      {/* Table */}
   444→      <div className="rounded-lg border border-slate-200 dark:border-slate-700 overflow-hidden">
   445→        <Table>
   446→          <TableHeader>
   447→            <TableRow className="bg-slate-50 dark:bg-slate-900 hover:bg-slate-50 dark:hover:bg-slate-900">
   448→              <TableHead className="w-12">
   449→                <Checkbox
   450→                  checked={selectedLoans.size === paginatedData.length && paginatedData.length > 0}
   451→                  onCheckedChange={handleSelectAll}
   452→                />
   453→              </TableHead>
   454→              <TableHead 
   455→                className="cursor-pointer select-none"
   456→                onClick={() => handleSort('loanNumber')}
   457→              >
   458→                <div className="flex items-center">
   459→                  Loan #
   460→                  <SortIcon field="loanNumber" />
   461→                </div>
   462→              </TableHead>
   463→              <TableHead 
   464→                className="cursor-pointer select-none"
   465→                onClick={() => handleSort('customerName')}
   466→              >
   467→                <div className="flex items-center">
   468→                  Customer
   469→                  <SortIcon field="customerName" />
   470→                </div>
   471→              </TableHead>
   472→              <TableHead 
   473→                className="cursor-pointer select-none text-center"
   474→                onClick={() => handleSort('daysInArrears')}
   475→              >
   476→                <div className="flex items-center justify-center">
   477→                  Days Overdue
   478→                  <SortIcon field="daysInArrears" />
   479→                </div>
   480→              </TableHead>
   481→              <TableHead 
   482→                className="cursor-pointer select-none text-right"
   483→                onClick={() => handleSort('outstandingBalance')}
   484→              >
   485→                <div className="flex items-center justify-end">
   486→                  Amount
   487→                  <SortIcon field="outstandingBalance" />
   488→                </div>
   489→              </TableHead>
   490→              <TableHead>Agent</TableHead>
   491→              <TableHead>Status</TableHead>
   492→              <TableHead className="text-right">Actions</TableHead>
   493→            </TableRow>
   494→          </TableHeader>
   495→          <TableBody>
   496→            {isLoading ? (
   497→              Array.from({ length: 5 }).map((_, i) => (
   498→                <TableRow key={i}>
   499→                  {Array.from({ length: 8 }).map((_, j) => (
   500→                    <TableCell key={j}>
   501→                      <Skeleton className="h-5 w-full" />
   502→                    </TableCell>
   503→                  ))}
   504→                </TableRow>
   505→              ))
   506→            ) : paginatedData.length === 0 ? (
   507→              <TableRow>
   508→                <TableCell colSpan={8} className="text-center py-12 text-slate-500">
   509→                  <div className="space-y-2">
   510→                    <p className="font-medium">No overdue loans found</p>
   511→                    <p className="text-sm">Try adjusting your filters or search query</p>
   512→                  </div>
   513→                </TableCell>
   514→              </TableRow>
   515→            ) : (
   516→              paginatedData.map((loan) => (
   517→                <TableRow 
   518→                  key={loan.id}
   519→                  className={cn(
   520→                    "cursor-pointer transition-colors hover:bg-slate-50 dark:hover:bg-slate-800/50",
   521→                    getRowClassName(loan.daysInArrears)
   522→                  )}
   523→                  onClick={() => onLoanSelect?.(loan)}
   524→                >
   525→                  <TableCell onClick={(e) => e.stopPropagation()}>
   526→                    <Checkbox
   527→                      checked={selectedLoans.has(loan.id)}
   528→                      onCheckedChange={() => handleSelectOne(loan.id)}
   529→                    />
   530→                  </TableCell>
   531→                  <TableCell className="font-mono text-sm font-medium">
   532→                    {loan.loanNumber.replace('LN-', '').replace(/^0+/, '')}
   533→                  </TableCell>
   534→                  <TableCell>
   535→                    <div>
   536→                      <p className="font-medium">{loan.customerName}</p>
   537→                      <p className="text-xs text-slate-500 dark:text-slate-400">{loan.customerPhone}</p>
   538→                    </div>
   539→                  </TableCell>
   540→                  <TableCell className="text-center">
   541→                    <div className="flex flex-col items-center gap-1">
   542→                      <span className="font-bold text-lg">{loan.daysInArrears}</span>
   543→                      {getSeverityBadge(loan.daysInArrears)}
   544→                    </div>
   545→                  </TableCell>
   546→                  <TableCell className="text-right">
   547→                    <div>
   548→                      <p className="font-medium">{formatCurrency(loan.outstandingBalance)}</p>
   549→                      <p className="text-xs text-slate-500 dark:text-slate-400">
   550→                        of {formatCurrency(loan.principal)}
   551→                      </p>
   552→                    </div>
   553→                  </TableCell>
   554→                  <TableCell>
   555→                    {loan.collectorName ? (

   565→                        loan.status === 'DEFAULTED' 
   566→                          ? 'bg-red-100 text-red-800 dark:bg-red-900/40 dark:text-red-400 border-0'
   567→                          : loan.status === 'IN_ARREARS'
   568→                          ? 'bg-amber-100 text-amber-800 dark:bg-amber-900/40 dark:text-amber-400 border-0'
   569→                          : ''
   570→                      }
   571→                    >
   572→                      {loan.status.replace('_', ' ')}
   573→                    </Badge>
   574→                  </TableCell>
   575→                  <TableCell className="text-right" onClick={(e) => e.stopPropagation()}>
   576→                    <div className="flex items-center justify-end gap-1">
   577→                      <Button
   578→                        variant="ghost"
   579→                        size="icon"
   580→                        className="h-8 w-8"
   581→                        onClick={() => handleContactAction('call', loan)}
   582→                        title="Call customer"
   583→                      >
   584→                        <Phone className="w-4 h-4 text-blue-600" />

   586→                      <Button
   587→                        variant="ghost"
   588→                        size="icon"

   590→                        onClick={() => handleContactAction('sms', loan)}

   594→                      </Button>
   595→                      <Button
   596→                        variant="ghost"
   597→                        size="icon"
   598→                        className="h-8 w-8"
   599→                        onClick={() => handleContactAction('whatsapp', loan)}
   600→                        title="WhatsApp"
   601→                      >

   609→                        title="Send Email"
   610→                      >
   611→                        <Mail className="w-4 h-4 text-purple-600" />
   612→                      </Button>
   613→                      <DropdownMenu>
   614→                        <DropdownMenuTrigger asChild>
   615→                          <Button variant="ghost" size="icon" className="h-8 w-8">

   640→          </TableBody>
   641→        </Table>
   642→      </div>
   643→
   644→      {/* Pagination */}
   645→      {totalPages > 1 && (
   646→        <div className="flex items-center justify-between">
   647→          <p className="text-sm text-slate-500 dark:text-slate-400">
   648→            Showing {(currentPage - 1) * itemsPerPage + 1} to{' '}
   649→            {Math.min(currentPage * itemsPerPage, filteredAndSortedData.length)} of{' '}
   650→            {filteredAndSortedData.length} loans
   651→          </p>
   652→          <div className="flex items-center gap-2">
   653→            <Button
   654→              variant="outline"
   655→              size="sm"
   656→              disabled={currentPage === 1}

   660→              <ChevronLeft className="w-4 h-4" />
   661→              Previous
   662→            </Button>
   663→            <div className="flex gap-1">
   664→              {Array.from({ length: Math.min(5, totalPages) }).map((_, i) => {

   666→                if (totalPages <= 5) {
   667→                  pageNum = i + 1
   668→                } else if (currentPage <= 3) {
   669→                  pageNum = i + 1
   670→                } else if (currentPage >= totalPages - 2) {
   671→                  pageNum = totalPages - 4 + i
   672→                } else {
   673→                  pageNum = currentPage - 2 + i
   674→                }
   675→                
   676→                return (

   678→                    key={pageNum}
   679→                    variant={currentPage === pageNum ? "default" : "outline"}

   689→                )
   690→              })}
   691→            </div>
   692→            <Button
   693→              variant="outline"
   694→              size="sm"
   695→              disabled={currentPage === totalPages}
   696→              onClick={() => setCurrentPage(p => p + 1)}
   697→              className="dark:border-slate-700"
   698→            >
   699→              Next
   700→              <ChevronRight className="w-4 h-4" />
   701→            </Button>
   702→          </div>
   703→        </div>
