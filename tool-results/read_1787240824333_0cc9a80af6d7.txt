   567→// Helper Components
   568→function ApiEndpoint({ 
   569→  method, 
   570→  path, 
   571→  description, 
   572→  body, 
   573→  params, 
   574→  response 
   575→}: { 
   576→  method: string; 
   577→  path: string; 
   578→  description: string;
   579→  body?: Record<string, string>;
   580→  params?: Record<string, string>;
   581→  response?: Record<string, string>;
   582→}) {
   583→  const getMethodColor = (m: string) => {
   584→    if (m.includes('POST')) return 'bg-emerald-500'
   585→    if (m.includes('PUT')) return 'bg-blue-500'
   586→    if (m.includes('DELETE')) return 'bg-red-500'
   587→    return 'bg-amber-500'
   588→  }
   589→
   590→  return (
   591→    <div className="space-y-3">
   592→      <div className="flex items-center gap-3">
   593→        <Badge className={`${getMethodColor(method)} text-white`}>
   594→          {method}
   595→        </Badge>
   596→        <code className="text-sm bg-muted px-3 py-1 rounded font-mono">{path}</code>
   597→      </div>
   598→      <p className="text-sm text-muted-foreground ml-20">{description}</p>
   599→      
   600→      {body && (
   601→        <div className="ml-20 space-y-1">
   602→          <p className="text-xs font-medium text-muted-foreground uppercase">Request Body:</p>
   603→          <pre className="text-xs bg-muted p-3 rounded overflow-x-auto">
   604→            {JSON.stringify(body, null, 2)}
   605→          </pre>
   606→        </div>
   607→      )}
   608→      
   609→      {params && (
   610→        <div className="ml-20 space-y-1">
   611→          <p className="text-xs font-medium text-muted-foreground uppercase">Query Params:</p>
   612→          <pre className="text-xs bg-muted p-3 rounded overflow-x-auto">
   613→            {JSON.stringify(params, null, 2)}
   614→          </pre>
   615→        </div>
   616→      )}
   617→      
   618→      {response && (
   619→        <div className="ml-20 space-y-1">
   620→          <p className="text-xs font-medium text-muted-foreground uppercase">Response:</p>
   621→          <pre className="text-xs bg-muted p-3 rounded overflow-x-auto">
   622→            {JSON.stringify(response, null, 2)}
   623→          </pre>
   624→        </div>
   625→      )}
   626→    </div>
   627→  )
   628→}
