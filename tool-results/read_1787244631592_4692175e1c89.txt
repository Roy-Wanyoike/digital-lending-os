   250→// Generate realistic latency based on time of day
   251→function generateTimeBasedLatency(
   252→  baseLatency: number,
   253→  variance: number
   254→): LatencyMetrics {
   255→  const hour = new Date().getHours();
   256→  const isPeakHour = hour >= 8 && hour <= 11 || hour >= 14 && hour <= 17;
   257→  const peakMultiplier = isPeakHour ? 1.3 + Math.random() * 0.4 : 1;
   258→  
   259→  const p50 = Math.round(baseLatency * peakMultiplier + (Math.random() - 0.5) * variance);
   260→  const p95 = Math.round(p50 * (1.8 + Math.random() * 0.4));
   261→  const p99 = Math.round(p95 * (1.5 + Math.random() * 0.5));
   262→  
   263→  return { p50, p95, p99 };
   264→}

   266→// Generate request volume with realistic patterns
   267→function generateRequestVolume(baseVolume: number): RequestVolume {
   268→  const hour = new Date().getHours();
   269→  const dayOfWeek = new Date().getDay();
   270→  const isWeekend = dayOfWeek === 0 || dayOfWeek === 6;
   271→  
   272→  let timeMultiplier = 1;
   273→  if (isWeekend) {
   274→    timeMultiplier = 0.4 + Math.random() * 0.2;
   275→  } else if (hour >= 6 && hour <= 9) {
   276→    timeMultiplier = 1.5 + Math.random() * 0.3; // Morning peak
   277→  } else if (hour >= 12 && hour <= 14) {
   278→    timeMultiplier = 1.2 + Math.random() * 0.2; // Lunch
   279→  } else if (hour >= 17 && hour <= 20) {
   280→    timeMultiplier = 1.4 + Math.random() * 0.3; // Evening
   281→  } else if (hour >= 0 && hour <= 5) {
   282→    timeMultiplier = 0.1 + Math.random() * 0.1; // Night
   283→  }
   284→  
   285→  const total = Math.round(baseVolume * timeMultiplier);
   286→  const errorRate = 0.01 + Math.random() * 0.09; // 1-10% base error rate varies by provider
   287→  const failed = Math.max(0, Math.round(total * errorRate / 100));
   288→  const successful = total - failed;
   289→  
   290→  return { total, successful, failed };
   291→}

   404→// Generate alerts based on provider statuses
   405→function generateAlerts(providers: ProviderStatus[]): Alert[] {
   406→  const alerts: Alert[] = [];
   407→  const alertTemplates: Record<ProviderStatusType, { severity: AlertSeverity; messages: string[] }[]> = {
   408→    healthy: [],
   409→    degraded: [
   410→      { severity: "warning", messages: [
   411→        "Elevated response times detected",
   412→        "Success rate below threshold",
   413→        "Intermittent timeouts occurring",
   414→        "Performance degradation observed",
   415→      ]},
   416→      { severity: "info", messages: [
   417→        "Approaching rate limits",
   418→        "Cache hit ratio decreased",
   419→      ]},
   420→    ],
   421→    down: [
   422→      { severity: "critical", messages: [
   423→        "Service unreachable",
   424→        "Connection refused",
   425→        "All health checks failing",
   426→        "Complete service outage",
   427→      ]},
   428→      { severity: "warning", messages: [
   429→        "Failover initiated",
   430→        "Circuit breaker open",
   431→      ]},
   432→    ],
   433→    unknown: [
   434→      { severity: "warning", messages: [
   435→        "Unable to determine service status",
   436→        "Health check timeout",
   437→        "Status check pending",
   438→      ]},
   439→    ],
   440→  };
   441→
   442→  providers.forEach((provider, index) => {
   443→    const templates = alertTemplates[provider.status];
   444→    templates?.forEach(({ severity, messages }) => {
   445→      if (Math.random() > 0.5) {
   446→        const message = messages[Math.floor(Math.random() * messages.length)];
   447→        alerts.push({
   448→          id: `alert-${provider.id}-${index}-${Date.now()}`,
   449→          severity,
   450→          provider: provider.displayName,
   451→          providerId: provider.id,
   452→          message: `${provider.displayName}: ${message}`,
   453→          createdAt: new Date(Date.now() - Math.random() * 3600000),
   454→          acknowledged: Math.random() > 0.7,
   455→          ...(Math.random() > 0.7 ? { acknowledgedBy: "ops-team@lendingos.co.ke" } : {}),
   456→          resolved: false,
   457→          type: severity === "critical" ? "outage" : severity === "warning" ? "degradation" : "info",
   458→        });
   459→      }
   460→    });
   461→  });
   462→
   463→  return alerts.sort((a, b) => b.createdAt.getTime() - a.createdAt.getTime());
   464→}

   554→// Main health check function
   555→export function performHealthCheck(tenantId?: string): HealthCheckResult {
   556→  const now = new Date();
   557→  const providers = PROVIDERS_CONFIG.map(generateProviderStatus);
   558→  const alerts = [...generateAlerts(providers), ...alertsStore.filter(a => !a.resolved)];
   559→  
   560→  const healthyCount = providers.filter(p => p.status === "healthy").length;
   561→  const degradedCount = providers.filter(p => p.status === "degraded").length;
   562→  const downCount = providers.filter(p => p.status === "down").length;
   563→  const unknownCount = providers.filter(p => p.status === "unknown").length;
   564→  
   565→  let overallHealth: "healthy" | "degraded" | "critical";
   566→  if (downCount > 0) {
   567→    overallHealth = "critical";
   568→  } else if (degradedCount > 2 || unknownCount > 0) {
   569→    overallHealth = "degraded";
   570→  } else if (degradedCount > 0) {
   571→    overallHealth = "degraded";
   572→  } else {
   573→    overallHealth = "healthy";
   574→  }
   575→
   576→  return {
   577→    timestamp: now,
   578→    providers,
   579→    overallHealth,
   580→    alerts: alerts.slice(0, 20), // Limit to recent 20 alerts
   581→    summary: {
   582→      totalProviders: providers.length,
   583→      healthyCount,
   584→      degradedCount,
   585→      downCount,
   586→      unknownCount,
   587→      activeIncidents: incidentsStore.filter(i => i.status !== "resolved").length,
   588→      activeAlerts: alerts.filter(a => !a.resolved).length,
   589→    },
   590→  };
   591→}

   684→// Generate historical data for charts
   685→export function generateHistoricalData(
   686→  providerId: string, 
   687→  period: "1h" | "6h" | "24h" | "7d" | "30d" = "24h"
   688→): ProviderHistory {
   689→  const config = PROVIDERS_CONFIG.find(p => p.id === providerId);
   690→  if (!config) {
   691→    throw new Error(`Provider ${providerId} not found`);
   692→  }
   693→
   694→  const now = new Date();
   695→  let points: number;
   696→  let intervalMs: number;
   697→
   698→  switch (period) {
   699→    case "1h":
   700→      points = 60;
   701→      intervalMs = 60 * 1000; // 1 minute
   702→      break;
   703→    case "6h":
   704→      points = 72;
   705→      intervalMs = 5 * 60 * 1000; // 5 minutes
   706→      break;
   707→    case "24h":
   708→      points = 96;
   709→      intervalMs = 15 * 60 * 1000; // 15 minutes
   710→      break;
   711→    case "7d":
   712→      points = 168;
   713→      intervalMs = 60 * 60 * 1000; // 1 hour
   714→      break;
   715→    case "30d":
   716→      points = 120;
   717→      intervalMs = 6 * 60 * 60 * 1000; // 6 hours
   718→      break;
   719→  }
   720→
   721→  const configs: Record<string, { baseLatency: number; baseErrorRate: number }> = {
   722→    "mpesa": { baseLatency: 420, baseErrorRate: 0.06 },
   723→    "pesalink": { baseLatency: 180, baseErrorRate: 0.02 },
   724→    "card-gateway": { baseLatency: 320, baseErrorRate: 0.05 },
   725→    "bank-transfer": { baseLatency: 1200, baseErrorRate: 0.01 },
   726→    "id-verification": { baseLatency: 850, baseErrorRate: 0.08 },
   727→    "crb-check": { baseLatency: 1100, baseErrorRate: 0.05 },
   728→    "biometrics": { baseLatency: 1500, baseErrorRate: 0.20 },
   729→    "sms-gateway": { baseLatency: 150, baseErrorRate: 0.03 },
   730→    "email-service": { baseLatency: 220, baseErrorRate: 0.01 },
   731→    "whatsapp-api": { baseLatency: 380, baseErrorRate: 0.05 },
   732→  };
   733→
   734→  const pc = configs[config.id] || { baseLatency: 500, baseErrorRate: 0.05 };
   735→  const data: HistoricalDataPoint[] = [];
   736→  
   737→  let totalRequests = 0;
   738→  let totalErrors = 0;
   739→  let minLatency = { p50: Infinity, p95: Infinity, p99: Infinity };
   740→  let maxLatency = { p50: 0, p95: 0, p99: 0 };
   741→  let sumLatency = { p50: 0, p95: 0, p99: 0 };
   742→
   743→  for (let i = points - 1; i >= 0; i--) {
   744→    const timestamp = new Date(now.getTime() - i * intervalMs);
   745→    const hour = timestamp.getHours();
   746→    const isPeakHour = hour >= 8 && hour <= 11 || hour >= 14 && hour <= 17;
   747→    const peakMultiplier = isPeakHour ? 1.3 + Math.random() * 0.4 : 1;
   748→    
   749→    // Add some random spikes
   750→    const spikeChance = Math.random();
   751→    const spikeMultiplier = spikeChance > 0.98 ? 3 : spikeChance > 0.95 ? 2 : 1;
   752→    
   753→    const p50 = Math.round(pc.baseLatency * peakMultiplier * spikeMultiplier + (Math.random() - 0.5) * pc.baseLatency * 0.3);
   754→    const p95 = Math.round(p50 * (1.8 + Math.random() * 0.4));
   755→    const p99 = Math.round(p95 * (1.5 + Math.random() * 0.5));
   756→
   757→    const requestCount = Math.round(100 + Math.random() * 500);
   758→    const errorRate = pc.baseErrorRate * spikeMultiplier + (Math.random() - 0.5) * 0.02;
   759→    const errorCount = Math.max(0, Math.round(requestCount * Math.max(0, errorRate) / 100));
   760→
   761→    const point: HistoricalDataPoint = {
   762→      timestamp,
   763→      latency: { p50: Math.max(1, p50), p95: Math.max(1, p95), p99: Math.max(1, p99) },
   764→      successRate: Math.round((100 - errorRate) * 100) / 100,
   765→      errorRate: Math.round(Math.max(0, errorRate) * 100) / 100,
   766→      requestCount,
   767→      errorCount,
   768→    };
   769→
   770→    data.push(point);
   771→
   772→    // Accumulate stats
   773→    totalRequests += requestCount;
   774→    totalErrors += errorCount;
   775→    
   776→    if (point.latency.p50 < minLatency.p50) minLatency.p50 = point.latency.p50;
   777→    if (point.latency.p95 < minLatency.p95) minLatency.p95 = point.latency.p95;
   778→    if (point.latency.p99 < minLatency.p99) minLatency.p99 = point.latency.p99;
   779→    
   780→    if (point.latency.p50 > maxLatency.p50) maxLatency.p50 = point.latency.p50;
   781→    if (point.latency.p95 > maxLatency.p95) maxLatency.p95 = point.latency.p95;
   782→    if (point.latency.p99 > maxLatency.p99) maxLatency.p99 = point.latency.p99;
   783→    
   784→    sumLatency.p50 += point.latency.p50;
   785→    sumLatency.p95 += point.latency.p95;
   786→    sumLatency.p99 += point.latency.p99;
   787→  }
   788→
   789→  const pointCount = data.length || 1;
   790→  const overallUptime = ((totalRequests - totalErrors) / totalRequests) * 100;
   791→
   792→  return {
   793→    providerId: config.id,
   794→    period,
   795→    data,
   796→    summary: {
   797→      avgLatency: {
   798→        p50: Math.round(sumLatency.p50 / pointCount),
   799→        p95: Math.round(sumLatency.p95 / pointCount),
   800→        p99: Math.round(sumLatency.p99 / pointCount),
   801→      },
   802→      maxLatency,
   803→      minLatency: {
   804→        p50: minLatency.p50 === Infinity ? 0 : minLatency.p50,
   805→        p95: minLatency.p95 === Infinity ? 0 : minLatency.p95,
   806→        p99: minLatency.p99 === Infinity ? 0 : minLatency.p99,
   807→      },
   808→      totalRequests,
   809→      totalErrors,
   810→      overallUptime: Math.round(overallUptime * 100) / 100,
   811→    },
   812→  };
   813→}

   815→// Get dependency graph data
   816→export function getDependencyGraph(): { nodes: Array<{ id: string; label: string; type: string; status: string }>; edges: Array<{ from: string; to: string }> } {
   817→  const providers = PROVIDERS_CONFIG.map(generateProviderStatus);
   818→  
   819→  const nodes = providers.map(p => ({
   820→    id: p.id,
   821→    label: `${p.icon} ${p.displayName}`,
   822→    type: p.type,
   823→    status: p.status,
   824→  }));
   825→
   826→  const edges: Array<{ from: string; to: string }> = [];
   827→  providers.forEach(p => {
   828→    p.dependencies.forEach(dep => {
   829→      edges.push({ from: p.id, to: dep });
   830→    });
   831→  });
   832→
   833→  return { nodes, edges };
   834→}
